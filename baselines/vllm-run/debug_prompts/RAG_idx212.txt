struct Scanner {

    idx: usize,
}
impl Scanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self {
            input: input.into_bytes(),
            idx: 0,
        }
    }
    fn next_i64(&mut self) -> i64 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i64;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i64;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.input[self.idx] - b'0') as i64;
            self.idx += 1;
        }
        num * sign
    }
    fn next_usize(&mut self) -> usize {
        self.next_i64() as usize
    }
}

    input: Vec<u8>,

        let mut data = Vec::new();
        if let Ok(mut f) = File::open("group.in") {
            let _ = f.read_to_end(&mut data);
        } else {
            let mut stdin = io::stdin();
            let _ = stdin.read_to_end(&mut data);
        }
        Self { input: data, idx: 0 }
    }

    fn next_i64(&mut self) -> i64 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i64;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i64;

            num = num * 10 + (self.input[self.idx] - b'0') as i64;
            self.idx += 1;
        }
        num * sign
    }

    fn next_i32(&mut self) -> i32 {

    }
}

fn main() {
    let mut sc = Scanner::new();
    let mut out = String::new();

    let n = sc.next_i32() as usize;
    let mut a: Vec<i64> = vec![0; n];
    for i in 0..n {
        a[i] = sc.next_i32() as i64;
    }
    a.sort_unstable();

        a[i] += a[i - 1];

        while idx >= 0 {
            b[i] += a[idx as usize];
