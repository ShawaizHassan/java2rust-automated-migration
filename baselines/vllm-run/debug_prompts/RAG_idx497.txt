fn get_zarr(s: &[u8], z: &mut [i32]) {
    let n = s.len();
    let mut l: usize = 0;
    let mut r: usize = 0;
    for i in 1..n {
        if i > r {
            l = i;
            r = i;
            while r < n && s[r - l] == s[r] {
                r += 1;
            }
            z[i] = (r - l) as i32;
            if r > 0 {
                r -= 1;
            }
        } else {
            let k = i - l;
            let bound = (r - i + 1) as i32;
            if z[k] < bound {
                z[i] = z[k];
            } else {
                l = i;
                while r < n && s[r - l] == s[r] {
                    r += 1;
                }
                z[i] = (r - l) as i32;
                if r > 0 {
                    r -= 1;
                }
            }
        }
    }
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = match it.next() {
        Some(x) => x.parse().unwrap(),
        None => return,
    };

    let first = it.next().unwrap_or("");
    let mut s: Vec<u8> = first.as_bytes().to_vec();

    let mut z: Vec<i32> = vec![0; 1_000_001];

    for _ in 1..n {
        let p_str = it.next().unwrap_or("");
        let p = p_str.as_bytes();
        let p_len = p.len();

        let start = if s.len() > p_len { s.len() - p_len } else { 0 };
        let tail = &s[start..];

        let mut str_bytes: Vec<u8> = Vec::with_capacity(p_len + 1 + tail.len());
        str_bytes.extend_from_slice(p);
        str_bytes.push(b'$');
        str_bytes.extend_from_slice(tail);

        if str_bytes.len() > z.len() {
            z.resize(str_bytes.len(), 0);

            }

                nb = zi;
                break;
            }
