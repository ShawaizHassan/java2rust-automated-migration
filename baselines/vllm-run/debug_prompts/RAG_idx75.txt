    };

    let mut x: Vec<i32> = vec![0; n];

    let mut d: Vec<i32> = vec![0; n];

    for i in 0..n {
        let val: i32 = it.next().unwrap().parse().unwrap();
        x[i] = val;
        y[i] = val;
        d[i] = val;
    }

    y.sort();

    let mut z: Vec<i32> = Vec::new();
    let mut f: Vec<i32> = Vec::new();

    if n >= 2 {
        let mut j: usize = n - 1;
        for i in 0..(n - 1) {
            if x[i] > x[i + 1] {
                z.push((i + 1) as i32);
            }
            if x[j] < x[j - 1] {
                f.push((j + 1) as i32);
            }
            j -= 1;
        }
    }

    if z.is_empty() || f.is_empty() {
        println!("yes");
        println!("1 1");
        return;
    }

    let mut i: usize = (z[0] - 1) as usize;
    let mut j: usize = (f[0] - 1) as usize;
    while i < j {
        d.swap(i, j);
        i += 1;
        j -= 1;

            break;
        }

fn main() {
    let mut input = String::new();

    let mut it = input.split_whitespace();

    let t: i32 = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };

    let mut out = String::new();

    for _ in 0..t {
        let n: usize = it.next().unwrap().parse::<i32>().unwrap() as usize;
        let s = it.next().unwrap();

        let bytes = s.as_bytes();

        let mut x1 = 0;
        let mut x2 = 0;
        let mut y1 = 0;
        let mut y2 = 0;

        for i in 0..n {
            let d = (bytes[i] - b'0') as i32;

                if d % 2 == 1 {
                    x1 = 1;
                } else {
                    y1 = 1;
                }
            } else {
                if d % 2 == 1 {
                    x2 = 1;
                } else {
                    y2 = 1;
                }
            }
        }

        let ans = if n % 2 == 1 {

        };
