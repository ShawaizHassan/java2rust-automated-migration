fn go(n: i32, memo: &mut Vec<i32>) -> i32 {
    if memo[n as usize] != -1 {
        return memo[n as usize];
    }
    if n < 10 {
        memo[n as usize] = n;
        return n;
    }

    let mut x = n;
    let mut prod: i32 = 1;
    while x > 0 {
        let d = x % 10;
        if d != 0 {
            prod *= d;
        }
        x /= 10;
    }

    let res = go(prod, memo);
    memo[n as usize] = res;
    res
}
