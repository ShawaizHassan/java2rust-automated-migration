const MOD: i64 = 1_000_000_007;
const MAX_POW: usize = 200_001;

fn add(a: i64, b: i64) -> i64 {
    (a + b) % MOD
}

fn mul(a: i64, b: i64) -> i64 {
    (a * b) % MOD
}

fn main() {

    for i in 1..MAX_POW {
