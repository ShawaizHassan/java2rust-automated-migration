use std::io::{self, Read};

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: i32 = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };
    let mut total: i64 = it.next().unwrap().parse().unwrap();
    let mut distress: i32 = 0;

    for _ in 0..n {
        let op = it.next().unwrap();
        let kid: i64 = it.next().unwrap().parse().unwrap();
        if op == "+" {
            total += kid;
        } else if total >= kid {
            total -= kid;
        } else {
            distress += 1;
        }
    }

    print!("{} {}", total, distress);
}
