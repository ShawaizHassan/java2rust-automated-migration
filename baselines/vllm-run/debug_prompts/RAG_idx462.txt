fn combine(s: &str, t: &str) -> String {
    let mut w = String::with_capacity(t.len() + 1 + s.len());
    w.push_str(t);
    w.push('#');
    w.push_str(s);

    let wb = w.as_bytes();
    let mut pref = vec![0i32; wb.len()];
    pref[0] = 0;
    let mut j: i32 = 0;

    for i in 1..wb.len() {
        while j > 0 && wb[i] != wb[j as usize] {
            j = pref[j as usize - 1];
        }
        if wb[i] == wb[j as usize] {
            j += 1;
        }
        pref[i] = j;
    }

    let tlen = t.len();
    for i in (tlen + 1)..wb.len() {
        if pref[i] == tlen as i32 {
            return s.to_string();
        }
    }

    let x = pref[wb.len() - 1] as usize;
    let mut ans = String::with_capacity(s.len() + (tlen - x));
    ans.push_str(s);
    let tb = t.as_bytes();

        ans.push(tb[i] as char);
    }
    ans
}

struct FastScanner {

impl FastScanner {
    fn new() -> Self {
        let mut input = Vec::new();
        io::stdin().read_to_end(&mut input).unwrap();
        Self { input, idx: 0 }
    }

    fn next_token(&mut self) -> Option<Vec<u8>> {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        if self.idx >= self.input.len() {
            return None;
        }
        let start = self.idx;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        Some(self.input[start..self.idx].to_vec())
    }

    fn next_string(&mut self) -> String {
        String::from_utf8(self.next_token().unwrap()).unwrap()
    }

    fn next_i32(&mut self) -> i32 {
        let tok = self.next_token().unwrap();
        let s = std::str::from_utf8(&tok).unwrap();
        s.parse::<i32>().unwrap()
    }
}
