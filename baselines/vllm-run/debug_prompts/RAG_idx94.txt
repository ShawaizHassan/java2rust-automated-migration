    let total_no: usize = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };
    let destiny: usize = it.next().unwrap().parse().unwrap();

    let mut total = vec![0i32; total_no];
    for i in 0..total_no.saturating_sub(1) {
        if let Some(v) = it.next() {
            total[i] = v.parse().unwrap();
        }
    }

    let mut i: usize = 0;
    while i < destiny {
        i = i + total[i] as usize;
        if i == destiny - 1 {
            result = true;
            break;
        }
    }

struct Scanner {
    input: Vec<u8>,
    idx: usize,
}
impl Scanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self {
            input: input.into_bytes(),
            idx: 0,
        }
    }
    fn next_i32(&mut self) -> i32 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
}
