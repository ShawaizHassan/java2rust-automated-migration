    let n: usize = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };

    let mut x: Vec<i32> = vec![0; n];
    let mut y: Vec<i32> = vec![0; n];
    let mut d: Vec<i32> = vec![0; n];

    for i in 0..n {
        let val: i32 = it.next().unwrap().parse().unwrap();
        x[i] = val;
        y[i] = val;
        d[i] = val;
    }

    y.sort();

    let mut z: Vec<i32> = Vec::new();
    let mut f: Vec<i32> = Vec::new();

    if n >= 2 {
        let mut j: usize = n - 1;
        for i in 0..(n - 1) {
            if x[i] > x[i + 1] {
                z.push((i + 1) as i32);
            }
            if x[j] < x[j - 1] {
                f.push((j + 1) as i32);
            }
            j -= 1;
        }
    }

    if z.is_empty() || f.is_empty() {
        println!("yes");
        println!("1 1");
        return;
    }

    let mut i: usize = (z[0] - 1) as usize;
    let mut j: usize = (f[0] - 1) as usize;
    while i < j {
        d.swap(i, j);
        i += 1;
        j -= 1;
    }

    let mut tt = true;

        if y[i] != d[i] {
            tt = false;
            break;
        }
    }

    if tt {
        println!("yes");
