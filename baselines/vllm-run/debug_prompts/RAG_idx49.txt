    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let t: i32 = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };

    let mut out = String::new();

    for _ in 0..t {
        let n: usize = it.next().unwrap().parse::<i32>().unwrap() as usize;
        let k: usize = it.next().unwrap().parse::<i32>().unwrap() as usize;

        let mut arr = vec![0i32; n];
        for i in 0..n {
            arr[i] = it.next().unwrap().parse::<i32>().unwrap();
        }

        let mut hm: HashMap<i32, i32> = HashMap::new();
        for i in 0..(n / 2) {
            let temp = arr[i] + arr[n - i - 1];
            *hm.entry(temp).or_insert(0) += 1;
        }

        let mut pre = vec![0i32; 2 * k + 2];
        for i in 0..(n / 2) {
            let a = arr[i];
            let b = arr[n - i - 1];
            let minv = (a.min(b) as usize) + 1;
            let maxv = (a.max(b) as usize) + k;
            pre[minv] += 1;
            if maxv + 1 < pre.len() {
                pre[maxv + 1] -= 1;
            }
        }

        for i in 3..(2 * k + 1) {
            pre[i] = pre[i - 1] + pre[i];
        }

        let mut ans: i32 = i32::MAX;
        let half = (n / 2) as i32;

        for i in 2..(2 * k + 1) {
            let temp = *hm.get(&(i as i32)).unwrap_or(&0);
            let c1 = pre[i] - temp;
            let c2 = half - c1 - temp;
            let cur = c1 + c2 * 2;
            if cur < ans {
                ans = cur;
            }
        }

        out.push_str(&format!("{}\n", ans));
    }

            res[idx][1] = (i + 1) as i32;
            idx += 1;
            idxes[pos] = -1;
