use std::io::{self, Read, Write};

struct InputReader {
    input: Vec<u8>,
    idx: usize,
}

impl InputReader {
    fn new() -> Self {
        let mut input = Vec::new();
        io::stdin().read_to_end(&mut input).unwrap();
        Self { input, idx: 0 }
    }

    fn next(&mut self) -> Option<String> {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        if self.idx >= self.input.len() {
            return None;
        }
        let start = self.idx;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        Some(String::from_utf8_lossy(&self.input[start..self.idx]).to_string())
    }

    fn next_i32(&mut self) -> i32 {
        self.next().unwrap().parse::<i32>().unwrap()
    }
}

fn main() {
    let mut inr = InputReader::new();
    let n: i32 = inr.next_i32();
    let k: i32 = inr.next_i32();

    let mut out = String::new();
    let nn = n as i64;
    let kk = k as i64;

    if nn * (nn - 1) / 2 <= kk {
        out.push_str("no solution\n");
    } else {
        for i in 0..n {
            out.push_str(&format!("0 {}\n", i));
        }
    }

    let mut stdout = io::BufWriter::new(io::stdout());
    stdout.write_all(out.as_bytes()).unwrap();
}
