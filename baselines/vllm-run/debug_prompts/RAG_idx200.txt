        tok = new StringTokenizer("");
    }

    private String tryReadNextLine() {
        try {
            return in.readLine();
        } catch (Exception e) {
            throw new InputMismatchException();
        }
    }

    public String nextToken() {
        while (!tok.hasMoreTokens()) {
            tok = new StringTokenizer(next());
        }
        return tok.nextToken();
    }

    public String next() {
        String newLine = tryReadNextLine();
        if (newLine == null)
            throw new InputMismatchException();
        return newLine;
    }

    public int nextInt() {
        return Integer.parseInt(nextToken());
    }

fn binary_search(al: &Vec<Pair>, key: i32, mut low: i32, mut high: i32) -> i32 {
    while low <= high {
        let mid = ((low + high) as u32 >> 1) as i32; // emulate Java >>> 1 on int
        let p = &al[mid as usize];
        if p.x == key {
            return p.y;
        } else if p.x > key {
            high = mid - 1;
        } else {
            low = mid + 1;
        }
    }
    -1
}
