use std::io::{self, Read};

struct Scanner {
    input: Vec<u8>,
    idx: usize,
}

impl Scanner {
    fn new() -> Self {
        let mut s = String::new();
        io::stdin().read_to_string(&mut s).unwrap();
        Self { input: s.into_bytes(), idx: 0 }
    }

    fn next_int(&mut self) -> i32 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut val: i32 = 0;
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_digit() {
            val = val * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        sign * val
    }

    fn next_line(&mut self) -> String {
        if self.idx >= self.input.len() {
            return String::new();
        }
        if self.input[self.idx] == b'\n' {
            self.idx += 1;
            return String::new();
        }
        let start = self.idx;
        while self.idx < self.input.len() && self.input[self.idx] != b'\n' {
            self.idx += 1;
        }
        let end = self.idx;
        if self.idx < self.input.len() && self.input[self.idx] == b'\n' {
            self.idx += 1;
        }
        let mut slice = &self.input[start..end];
        if !slice.is_empty() && slice[slice.len() - 1] == b'\r' {
            slice = &slice[..slice.len() - 1];
        }
        String::from_utf8_lossy(slice).to_string()
    }
}
