struct Scanner {

}

    fn new() -> Self {

        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }
    fn next_i32(&mut self) -> i32 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;

            num = num * 10 + (self.input[self.idx] - b'0') as i32;

        num * sign
    }
}

use std::io::{self, Read};

fn dfs(u: usize, adj: &Vec<Vec<usize>>, visited: &mut Vec<bool>, price: &Vec<i32>) -> i64 {
    visited[u] = true;
    let mut minv: i64 = price[u] as i64;
    for &v in &adj[u] {
        if !visited[v] {
            let child_min = dfs(v, adj, visited, price);
            if child_min < minv {
                minv = child_min;
            }
        }
    }
    minv
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = match it.next() {
        Some(x) => x.parse().unwrap(),
        None => return,
    };
    let mut m: i32 = it.next().unwrap().parse().unwrap();

    let mut price = vec![0i32; n + 1];
    for i in 1..=n {
        price[i] = it.next().unwrap().parse().unwrap();
    }

    let mut adj: Vec<Vec<usize>> = vec![Vec::new(); n + 1];
    while m > 0 {
        let x: usize = it.next().unwrap().parse().unwrap();
        let y: usize = it.next().unwrap().parse().unwrap();
        adj[x].push(y);
        adj[y].push(x);
        m -= 1;
    }

    let mut visited = vec![false; n + 1];
    let mut ans: i64 = 0;

    for i in 1..=n {
        if !visited[i] {
            ans += dfs(i, &adj, &mut visited, &price);
        }
    }
