        let mut data = Vec::new();
        input.read_to_end(&mut data).unwrap();
        Self { data, idx: 0 }
    }

    fn next_int(&mut self) -> i32 {
        let mut negative = false;
        loop {
            if self.idx >= self.data.len() {
                panic!("EOF");
            }
            let b = self.data[self.idx];
            self.idx += 1;
            if b == b'-' {
                negative = true;
                continue;
            }
            if b >= b'0' && b <= b'9' {
                let mut r: i32 = (b - b'0') as i32;

                    } else {
                        self.idx += 1;
                        break;
                    }
                }
                return if negative { -r } else { r };
            }
        }
    }
}

impl FastScanner {
    fn new<R: Read>(mut reader: R) -> Self {
        let mut data = Vec::new();
        reader.read_to_end(&mut data).unwrap();
        Self { data, idx: 0 }
    }

    fn next_i32(&mut self) -> i32 {
        while self.idx < self.data.len() && self.data[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        if self.idx >= self.data.len() {
            panic!("EOF");
        }
        let mut neg = false;
        if self.data[self.idx] == b'-' {
            neg = true;
            self.idx += 1;
        }
        let mut val: i32 = 0;
        while self.idx < self.data.len() {
            let c = self.data[self.idx];
            if !(b'0'..=b'9').contains(&c) {
                break;
            }
            val = val * 10 + (c - b'0') as i32;
            self.idx += 1;
        }
        if neg { -val } else { val }
    }
}
