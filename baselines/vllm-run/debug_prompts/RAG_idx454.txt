use std::io::{self, Read};

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };
    let m: usize = it.next().unwrap().parse().unwrap();

    let mut ar = vec![vec![0i32; m]; n];
    for i in 0..n {
        for j in 0..m {
            ar[i][j] = it.next().unwrap().parse().unwrap();
        }
    }

    let mut cur: i32 = 0;
    for i in 0..n {
        cur ^= ar[i][0];
    }

    let mut ans = vec![1i32; n];
    let mut anss = false;
    let mut done = false;

    if cur == 0 {
        for i in 0..n {
            if done {
                break;
            }
            for j in 0..m {
                if ar[i][j] != ar[i][0] {
                    ans[i] = (j + 1) as i32;
                    anss = true;
                    done = true;
                    break;
                }
            }
        }
    } else {
        anss = true;
    }

    if !anss {
        print!("NIE");
    } else {

        }
    }
