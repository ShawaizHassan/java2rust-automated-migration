fn can_make_at_least(has: &Vec<i32>, need: &Vec<i32>, free: i32, guess: i64) -> Option<bool> {
    if guess < 0 {
        panic!("IllegalArgumentException");
    }
    let mut needed: i64 = 0;
    for i in 0..need.len() {
        let req = (need[i] as i64) * guess;
        let h = has[i] as i64;
        if h >= req {
            continue;
        }
        needed += req - h;
        if needed > free as i64 {
            return Some(false);
        }
    }
    if needed == free as i64 {
        None
    } else {
        Some(true)
    }
}
