struct Scanner {
    input: Vec<u8>,
    idx: usize,
}

impl Scanner {
    fn new() -> Self {
        let mut input = Vec::new();
        io::stdin().read_to_end(&mut input).unwrap();
        Self { input, idx: 0 }
    }

    fn next_i32(&mut self) -> i32 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
}

fn main() {
    let mut sc = Scanner::new();
    let n = sc.next_i32() as usize;
    let mut a = vec![0i32; n];
    for i in 0..n {
        a[i] = sc.next_i32();
    }
    a.sort();

        print!("NO\n");
        return;

    for &v in &a {
        let idx = (v - base) as usize;
        cnt[idx] += 1;
    }

    for i in 1..(m as usize) {

            return;

        l = cnt[i];
