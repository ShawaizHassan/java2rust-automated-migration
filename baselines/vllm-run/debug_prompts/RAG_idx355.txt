struct Scanner {
    input: Vec<u8>,
    idx: usize,
}
impl Scanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }
    fn next_i32(&mut self) -> i32 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
}

use std::io::{self, Read};

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let s = input.lines().next().unwrap_or("");

    let mut x: i32 = 0;
    let mut y: i32 = 0;

    for ch in s.chars() {
        if ch == 'x' {
            x += 1;
        } else {
            y += 1;
        }
    }

    let diff: i32 = (x - y).abs();
    let mut out = String::new();

    if x > y {
        out.extend(std::iter::repeat('x').take(diff as usize));
    } else {
        out.extend(std::iter::repeat('y').take(diff as usize));
    }

    println!("{}", out);
}
