const MOD: i64 = 1_000_000_007;

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();

    let mut lines = input.lines();
    let input1 = lines.next().unwrap_or("").trim().to_string();
    let input2 = lines.next().unwrap_or("").to_string();

    let n: usize = input1.parse::<i32>().unwrap_or(0).max(0) as usize;

    let mut xs = vec![0i32; n];
    let parts: Vec<&str> = input2.split(' ').collect();
    for i in 0..parts.len() {
        if i >= n {
            break;
        }
        let t = parts[i].trim();
        if t.is_empty() {
            continue;
        }
        xs[i] = t.parse::<i32>().unwrap();
    }

    let mut pow2 = vec![0i64; n + 1];
    pow2[0] = 1;
    for i in 1..=n {
        pow2[i] = (pow2[i - 1] * 2) % MOD;
    }

    let size = 1usize << 20;
    let mut dp = vec![0i32; size];
    for &x in &xs {
        let idx = x as usize;
        if idx < size {
            dp[idx] += 1;
        }
    }

    for i in 0..20 {
        for j in 0..size {
            if ((j >> i) & 1) == 0 {
                let add = dp[j | (1usize << i)];
                dp[j] += add;
            }
        }
    }

    let mut ans: i64 = 0;
    for i in 0..size {
        let parity = (i as u32).count_ones() & 1;
        let sign: i64 = if parity == 0 { 1 } else { -1 };
        let cnt = dp[i] as usize;
        let term = (sign * pow2[cnt]) % MOD;
        ans += term;
    }

    let out = ((ans % MOD) + MOD) % MOD;
    println!("{}", out);
}
