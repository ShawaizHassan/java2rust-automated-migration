fn get(mut c: Vec<u8>, k: usize) -> Option<Vec<u8>> {
    if k == 0 || c.len() % k != 0 {
        return None;
    }
    c.sort_unstable();
    let n = c.len();
    let block = n / k;
    let mut res = vec![0u8; n];
    let mut cnt = 0usize;
    for i in 0..block {
        for j in 0..k {
            res[j * block + i] = c[cnt];
            cnt += 1;
        }
    }
    for i in 0..k {
        if res[i * block..(i + 1) * block] != res[0..block] {
            return None;
        }
    }
    Some(res)
}

        Self { input, idx: 0 }

    fn next_token(&mut self) -> Option<String> {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;

        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {

    }

}
