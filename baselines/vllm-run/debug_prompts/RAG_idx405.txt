fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };
    let m: usize = it.next().unwrap().parse().unwrap();

    let mut a = vec![0i32; n + 1];
    let mut b = vec![0i32; m + 1];

    for i in 1..=n {
        a[i] = it.next().unwrap().parse().unwrap();
    }
    for i in 1..=m {
        b[i] = it.next().unwrap().parse().unwrap();
    }

    let mut sol: i32 = 0;
    let max_sol: i32 = 1 << 9;

    while sol < max_sol {
        let mut possible = true;

        for i in 1..=n {
            let mut found = false;
            for j in 1..=m {
                if (sol | (a[i] & b[j])) == sol {
                    found = true;
                    break;
                }
            }
            if !found {
                possible = false;
                break;
            }
        }

        if possible {
            break;
        }
        sol += 1;
    }

    print!("{}", sol);
}
