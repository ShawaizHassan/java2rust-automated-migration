        io::stdin().read_to_string(&mut input).unwrap();
        Self { buf: input.into_bytes(), idx: 0 }
    }

    fn is_whitespace(c: u8) -> bool {
        c <= 32
    }

    fn next_i32(&mut self) -> i32 {
        while self.idx < self.buf.len() && Self::is_whitespace(self.buf[self.idx]) {
            self.idx += 1;

            sgn = -1;
            self.idx += 1;
