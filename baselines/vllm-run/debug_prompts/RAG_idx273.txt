use std::io::{self, Read};

struct Scanner {
    input: Vec<u8>,
    idx: usize,
}
impl Scanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }
    fn next<T: std::str::FromStr>(&mut self) -> T {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let start = self.idx;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        std::str::from_utf8(&self.input[start..self.idx])
            .unwrap()
            .parse::<T>()
            .ok()
            .unwrap()
    }
}

fn main() {
    let mut sc = Scanner::new();

    let n: usize = sc.next::<i32>() as usize;
    let mut names: Vec<[String; 2]> = Vec::with_capacity(n);

    for _ in 0..n {
        let a: String = sc.next();
        let b: String = sc.next();
        let mut pair = [a, b];
        pair.sort(); // lexicographic ascending, like Arrays.sort on String[]
        names.push(pair);
