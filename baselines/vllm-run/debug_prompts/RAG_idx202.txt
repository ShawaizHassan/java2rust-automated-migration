        Some(v) => v.parse().unwrap(),
        None => return,
    };

        arr_int[i] = it.next().unwrap().parse::<i64>().unwrap();
    }

    let mut out = String::new();

        let aux2: i64 = it.next().unwrap().parse().unwrap();

            aumento += aux2;
