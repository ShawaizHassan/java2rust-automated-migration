    fn new() -> Self {

        io::stdin().read_to_string(&mut input).unwrap();
        Self {
            input: input.into_bytes(),
            idx: 0,
        }
    }
    fn next_i32(&mut self) -> i32 {
        self.next_string().parse::<i32>().unwrap()
    }
    fn next_usize(&mut self) -> usize {
        self.next_string().parse::<usize>().unwrap()
    }
    fn next_string(&mut self) -> String {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let start = self.idx;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        String::from_utf8(self.input[start..self.idx].to_vec()).unwrap()
    }
}

use std::io::{self, Read};

struct FastScanner {
    input: Vec<u8>,
    idx: usize,
}

impl FastScanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }

    fn next_i64(&mut self) -> i64 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i64;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i64;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.input[self.idx] - b'0') as i64;
            self.idx += 1;
        }
        num * sign
    }

    fn next_usize(&mut self) -> usize {
        self.next_i64() as usize
    }
}

fn main() {
    let mut sc = FastScanner::new();
    let t = sc.next_usize();
    let mut out = String::new();

    for _ in 0..t {
        let n = sc.next_usize();
        let m = sc.next_usize();

        let mut arr: Vec<Vec<i64>> = vec![vec![0; m]; n];
        for i in 0..n {
            for j in 0..m {
                arr[i][j] = sc.next_i64();
            }
        }

        let len = n + m - 1;
        let mut arr1: Vec<i64> = vec![0; len];
        let mut arr2: Vec<i64> = vec![0; len];
        let mut ans: i64 = 0;

        for i in 0..len {
            let mut k: i64 = 0;
            let start = if i >= m - 1 { i - (m - 1) } else { 0 };
            let end = if i <= n - 1 { i } else { n - 1 };

            for j in start..=end {
                let base_col = std::cmp::min(i, m - 1) as i64;
                let col = (base_col - k) as usize;
                arr1[i] += arr[j][col];
                arr2[i] += 1;
                k += 1;
            }
            out.push('\n');
        }

        for i in 0..(len / 2) {
            let other = len - 1 - i;
            let option1 = (arr2[i] - arr1[i]) + (arr2[other] - arr1[other]);
            let option2 = arr1[i] + arr1[other];
            ans += if option1 < option2 { option1 } else { option2 };
