    fn new() -> Self {

        io::stdin().read_to_string(&mut input).unwrap();
        Self { data: input.into_bytes(), idx: 0 }
    }

    fn next_i32(&mut self) -> i32 {
        while self.idx < self.data.len() && self.data[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.data.len() && self.data[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.data.len() && !self.data[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.data[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
}

use std::io::{self, Read};

struct InputReader {
    data: Vec<u8>,
    idx: usize,
}
impl InputReader {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self {
            data: input.into_bytes(),
            idx: 0,
        }
    }
    fn next_token(&mut self) -> String {
        while self.idx < self.data.len() && self.data[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let start = self.idx;
        while self.idx < self.data.len() && !self.data[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        String::from_utf8(self.data[start..self.idx].to_vec()).unwrap()
    }
    fn next_i32(&mut self) -> i32 {
        self.next_token().parse::<i32>().unwrap()
    }
    fn next_f32(&mut self) -> f32 {
        self.next_token().parse::<f32>().unwrap()
    }
}

fn main() {
    let mut inr = InputReader::new();
    let n = inr.next_i32() as usize;

    let mut fnv: Vec<f32> = vec![0.0; n];
    let mut dnv: Vec<i32> = vec![0; n];
    let mut ds: i32 = 0;

    for i in 0..n {
        let f = inr.next_f32();
        fnv[i] = f;
        let d = f as i32; // truncates toward zero, like Java
        dnv[i] = d;
        ds += d;
    }

    for i in 0..n {
        if ds < 0 && fnv[i] > dnv[i] as f32 {
            if dnv[i] >= 0 {
                dnv[i] += 1;
                ds += 1;
            }
        } else if ds > 0 && fnv[i] < dnv[i] as f32 {
            if dnv[i] <= 0 {
                dnv[i] -= 1;
                ds -= 1;
            }
        }
    }

    let mut out = String::new();
    for v in dnv {
        out.push_str(&format!("{}\n", v));
    }
    print!("{out}");
}
