use std::io::{self, Read};

fn to_decimal(base26: &str) -> i32 {
    let mut res: i32 = 0;
    let mut base: i32 = 1;
    for b in base26.as_bytes().iter().rev() {
        let val = (*b as i32) - 64; // 'A' => 1
        res += val * base;
        base *= 26;
    }
    res
}

fn to_base26(mut decimal: i32) -> String {
    let mut res: Vec<u8> = Vec::new();
    while decimal != 0 {
        let k = decimal % 26;
        if k == 0 {
            decimal = decimal / 26 - 1;
            res.push(b'Z');
        } else {
            decimal /= 26;
            res.push((k as u8) + 64); // 1->'A'
        }
    }
    res.reverse();
    String::from_utf8(res).unwrap()
}

fn is_digits(s: &str) -> bool {
    !s.is_empty() && s.as_bytes().iter().all(|c| c.is_ascii_digit())
}

fn is_upper_letters(s: &str) -> bool {
    !s.is_empty() && s.as_bytes().iter().all(|c| c.is_ascii_uppercase())
}

fn parse_rc(s: &str) -> Option<(String, String)> {
    // Matches: R<rowdigits>C<columndigits>
    if !s.starts_with('R') {
        return None;
    }
    let after_r = &s[1..];
    let c_pos = after_r.find('C')?;
    let row = &after_r[..c_pos];
    let col = &after_r[c_pos + 1..];
    if is_digits(row) && is_digits(col) {
        Some((row.to_string(), col.to_string()))
    } else {
        None
    }
}

fn parse_a1(s: &str) -> Option<(String, String)> {
    // Matches: <A-Z>+<digits>+
    let bytes = s.as_bytes();
    let mut i = 0usize;
    while i < bytes.len() && bytes[i].is_ascii_uppercase() {
        i += 1;
    }
    if i == 0 || i == bytes.len() {
        return None;
    }
    let col = &s[..i];
    let row = &s[i..];
    if is_upper_letters(col) && is_digits(row) {
        Some((col.to_string(), row.to_string()))
    } else {
        None
    }
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.lines();

    let first = it.next().unwrap_or("");
    let mut n: usize = first.trim().parse::<usize>().unwrap_or(0);

    let mut lines: Vec<String> = Vec::with_capacity(n);
    while n > 0 {
        lines.push(it.next().unwrap_or("").to_string());
