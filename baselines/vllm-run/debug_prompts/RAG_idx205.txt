        let idx = (pos - 1) as usize;

struct FastScanner {
    buf: Vec<u8>,
    idx: usize,
}
impl FastScanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { buf: input.into_bytes(), idx: 0 }
    }
    fn next_i32(&mut self) -> i32 {
        while self.idx < self.buf.len() && self.buf[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.buf.len() && self.buf[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.buf.len() && !self.buf[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.buf[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
}

fn main() {
    let mut fs = FastScanner::new();
    let n = fs.next_i32() as usize;

    let mut p = vec![0usize; n];
    let mut pos = vec![0usize; n];
    for i in 0..n {
        let v = (fs.next_i32() - 1) as usize;
        p[i] = v;
        pos[v] = i;
    }

    let mut dp = vec![0usize; n];
    for i in 0..n {
        dp[i] = i;
    }

    if n >= 2 {
        for i in (0..=n - 2).rev() {
            if pos[i] < pos[i + 1] {
                dp[i] = dp[i + 1];
            }
        }
    }

    let mut ans = n as i32;
    for i in 0..n {
        let len = (dp[i] - i + 1) as i32;
        let cand = n as i32 - len;
        if cand < ans {
            ans = cand;
        }
    }

    print!("{}", ans);
    println!();
