impl InputReader {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { data: input.into_bytes(), idx: 0 }
    }

    fn next_i32(&mut self) -> i32 {
        while self.idx < self.data.len() && self.data[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.data.len() && self.data[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.data.len() && !self.data[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.data[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
}

use std::collections::HashMap;
use std::io::{self, Read};

struct Scanner {
    input: Vec<u8>,
    idx: usize,
}
impl Scanner {
    fn new() -> Self {
        let mut input = Vec::new();
        io::stdin().read_to_end(&mut input).unwrap();
        Self { input, idx: 0 }
    }
    fn next_token(&mut self) -> Option<Vec<u8>> {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        if self.idx >= self.input.len() {
            return None;
        }
        let start = self.idx;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        Some(self.input[start..self.idx].to_vec())
    }
    fn next_i32(&mut self) -> i32 {
        let tok = self.next_token().unwrap();
        let s = std::str::from_utf8(&tok).unwrap();
        s.parse::<i32>().unwrap()
    }
    fn next_char(&mut self) -> char {
        let tok = self.next_token().unwrap();
        tok[0] as char
    }
}

fn main() {
    let mut sc = Scanner::new();

    let n = sc.next_i32() as usize;
    let m = sc.next_i32() as usize;
    let k = sc.next_i32() as usize;

    let mut table: Vec<Vec<i32>> = vec![vec![0; m]; n];
    let mut cols: HashMap<usize, usize> = HashMap::new();

    for i in 0..n {
        for j in 0..m {
            table[i][j] = sc.next_i32();
        }
    }

    let mut out = String::new();

    for _ in 0..k {
        let c = sc.next_char();
        let x = sc.next_i32() as usize;
        let y = sc.next_i32() as usize;

        if c == 'r' {
            table.swap(x - 1, y - 1);
        } else if c == 'c' {
            let y_idx = y - 1;
            let x_idx = x - 1;
            let y1 = *cols.get(&y_idx).unwrap_or(&y_idx);
            let x1 = *cols.get(&x_idx).unwrap_or(&x_idx);
            cols.insert(x_idx, y1);
            cols.insert(y_idx, x1);
        } else {
            let row = x - 1;
            let col0 = y - 1;
            let col = *cols.get(&col0).unwrap_or(&col0);
            out.push_str(&format!("{}\n", table[row][col]));
        }
    }

    print!("{}", out);
