use std::io::{self, Read};

struct Scanner {
    input: Vec<u8>,
    idx: usize,
}
impl Scanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }
    fn next_i32(&mut self) -> i32 {
        self.next_string().parse::<i32>().unwrap()
    }
    fn next_usize(&mut self) -> usize {
        self.next_string().parse::<usize>().unwrap()
    }
    fn next_string(&mut self) -> String {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let start = self.idx;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        String::from_utf8(self.input[start..self.idx].to_vec()).unwrap()
    }
}

fn compare(ans: &[u8], arr: &[Vec<u8>]) -> bool {
    let m = arr[0].len();

            }

        if diff > 1 {

    }

}

fn helper(arr: &[Vec<u8>]) -> Option<Vec<u8>> {
