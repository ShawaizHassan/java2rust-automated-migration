use std::io::{self, Read};

fn gcd(mut x: i32, mut y: i32) -> i32 {
    if x < y {
        std::mem::swap(&mut x, &mut y);
    }
    while y != 0 {
        let r = x % y;
        x = y;
        y = r;
    }
    x
}

fn main() {
    let mut input = String::new();

        None => return,
    };
    let mut m: i32 = it.next().unwrap().parse().unwrap();

    let mut edges: Vec<(i32, i32)> = Vec::new();

    'outer: for i in 1..=n {
        for j in (i + 1)..=n {
            if gcd(i, j) == 1 {
                edges.push((i, j));
            }

            break 'outer;
        }
