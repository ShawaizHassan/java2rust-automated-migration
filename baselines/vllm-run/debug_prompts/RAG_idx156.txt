use std::io::{self, Read};

struct FastInput {
    data: Vec<u8>,
    idx: usize,
}
impl FastInput {
    fn new() -> Self {
        let mut data = Vec::new();
        io::stdin().read_to_end(&mut data).unwrap();
        Self { data, idx: 0 }
    }
    fn skip_whitespace(&mut self) {
        while self.idx < self.data.len() && self.data[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
    }
    fn next_token(&mut self) -> Vec<u8> {
        self.skip_whitespace();
        let start = self.idx;
        while self.idx < self.data.len() && !self.data[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        self.data[start..self.idx].to_vec()
    }
    fn next_string(&mut self) -> String {
        String::from_utf8(self.next_token()).unwrap()
    }
    fn next_i32(&mut self) -> i32 {
        let tok = self.next_token();
        let s = std::str::from_utf8(&tok).unwrap();
        s.parse::<i32>().unwrap()
    }
}

struct FastScanner {
    input: Vec<u8>,
    idx: usize,

impl FastScanner {

        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }
    fn next_i32(&mut self) -> i32 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }

        if self.idx < self.input.len() && self.input[self.idx] == b'-' {

            self.idx += 1;

    }
