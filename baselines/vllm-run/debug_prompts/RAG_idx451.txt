    to: usize,
    next: i32,

static mut USED: [bool; MAXN] = [false; MAXN];
static mut ADJ: [i32; MAXN] = [-1; MAXN];
static mut EDGE: [Edge; MAXN * MAXN] = [Edge { to: 0, next: -1 }; MAXN * MAXN];
static mut SIZE: usize = 0;

unsafe fn init() {
    ADJ.fill(-1);
    SIZE = 0;
}

unsafe fn add_edge(u: usize, v: usize) {
    EDGE[SIZE] = Edge { to: v, next: ADJ[u] };
    ADJ[u] = SIZE as i32;
    SIZE += 1;
}

unsafe fn dfs(u: usize) -> bool {
    let mut i = ADJ[u];
    while i != -1 {
        let e = EDGE[i as usize];
        let v = e.to;
        if !USED[v] {
            USED[v] = true;

                LINKER[v] = u as i32;
                return true;
            }
        }
        i = e.next;
    }
    false
}

fn gcd(a: i32, b: i32) -> i32 {
    if b == 0 { a } else { gcd(b, a % b) }
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: i32 = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };
    let m: i32 = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };

    let mut edges: Vec<(i32, i32)> = Vec::new();

    if m < n - 1 {
        print!("Impossible\n");
        return;
    }

    for i in 1..n {
        edges.push((i, i + 1));
    }

    'outer: for i in 1..n {
        if edges.len() >= m as usize {
            break;
        }
        let mut j = i + 2;
        while edges.len() < m as usize && j <= n {
            if i == 1 || gcd(i, j) == 1 {
                edges.push((i, j));
            }
            j += 1;
        }
        if edges.len() >= m as usize {
            break 'outer;
        }
    }

    if edges.len() < m as usize {
        print!("Impossible\n");
        return;
    }

    let mut out = String::new();
    out.push_str("Possible\n");
    for (a, b) in edges.iter() {
        out.push_str(&format!("{} {}\n", a, b));
    }
    print!("{}", out);
