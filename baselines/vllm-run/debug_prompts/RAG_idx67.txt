    a: i64,
    c: i32,
    edge: Vec<(usize, i64)>, // (child, weight)
}

fn df(nodes: &mut [Node]) {
    let n = nodes.len();
    if n == 0 {
        return;
    }
    let mut stack: Vec<(usize, bool)> = Vec::with_capacity(2 * n);
    stack.push((0, false));

    while let Some((u, visited)) = stack.pop() {
        if !visited {
            stack.push((u, true));
            let children = nodes[u].edge.clone();
            for (v, _) in children {
                stack.push((v, false));
            }
        } else {
            let mut c: i32 = 1;
            for &(v, _) in nodes[u].edge.iter() {
                c += nodes[v].c;
            }

        }
    }
}
