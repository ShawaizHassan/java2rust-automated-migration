impl FastInput {
    fn new() -> Self {

        io::stdin().read_to_string(&mut input).unwrap();
        Self {
            data: input.into_bytes(),
            idx: 0,
        }
    }
    fn next_i32(&mut self) -> i32 {
        while self.idx < self.data.len() && self.data[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.data.len() && self.data[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut val = 0i32;
        while self.idx < self.data.len() && !self.data[self.idx].is_ascii_whitespace() {
            val = val * 10 + (self.data[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        val * sign
    }
    fn next_string(&mut self) -> String {
        while self.idx < self.data.len() && self.data[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let start = self.idx;
        while self.idx < self.data.len() && !self.data[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        String::from_utf8_lossy(&self.data[start..self.idx]).to_string()
    }
}
