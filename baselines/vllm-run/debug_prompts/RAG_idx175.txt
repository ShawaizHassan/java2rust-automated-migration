use std::io::{self, Read};

struct Scanner {
    input: Vec<u8>,
    idx: usize,
}

impl Scanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }

    fn skip_whitespace(&mut self) {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
    }

    fn next_i32(&mut self) -> i32 {
        self.skip_whitespace();
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }

    fn next_string(&mut self) -> String {
        self.skip_whitespace();
        let start = self.idx;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        String::from_utf8_lossy(&self.input[start..self.idx]).to_string()
    }
}

use std::io::{self, Read};

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();

    // Java's Scanner.nextLine() reads the first line (may be empty).
    // Then nextInt() reads the next integer token (possibly on later lines).
    let mut parts = input.splitn(2, '\n');
    let s_line = parts.next().unwrap_or("");
    let rest = parts.next().unwrap_or("");

    // Java code calls s.toLowerCase() but does NOT assign it back, so it has no effect.
    let s = s_line.to_string();

    let k: i32 = rest.split_whitespace().next().unwrap().parse().unwrap();

    let l: i32 = s.len() as i32;

    if k > l {
        println!("impossible");
    } else {
        let mut count: i32 = 0;
        let mut a = vec![0i32; 26];

        for ch in s.bytes() {
            // Matches Java logic: a[((int)ch)-97]++ (assumes 'a'..'z')
            let idx = (ch as i32) - 97;
            a[idx as usize] += 1;
        }

        for i in 0..26 {
            if a[i] >= 1 {
                count += 1;
            }
        }

        if count >= k {
            println!("0");
        } else {
            println!("{}", k - count);
        }
    }
}
