fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();
    let str_in = match it.next() {
        Some(s) => s,
        None => return,
    };

    let ch: Vec<char> = str_in.chars().collect();
    let mut sb = String::new();

    let mut cc: i32 = 1;
    let mut pc: i32 = 0;
    let mut cch: char = ' ';

    for &cur in &ch {
        if cur != cch {
            cch = cur;
            pc = cc;
            cc = 1;
        } else {
            cc += 1;
            if cc > 2 || (cc == 2 && cc == pc) {
                cc -= 1;
                continue;
            }
        }
        sb.push(cch);
    }

    println!("{}", sb);
