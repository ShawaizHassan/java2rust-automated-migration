use std::io::{self, Read};

fn ans(a: &Vec<i32>, b: &Vec<i32>, n: usize) -> i64 {
    let mut c: Vec<i32> = Vec::with_capacity(n);
    for i in 0..n {
        c.push(a[i] - b[i]);
    }
    c.sort();

    let mut res: i64 = 0;
    let mut start: usize = 0;
    let mut end: usize = n - 1;

    while start < end {
        let sum = c[start] as i64 + c[end] as i64;
        if sum > 0 {
            res += (end - start) as i64;
            if end == 0 { break; }
            end -= 1;
        } else {
            start += 1;
        }
    }
    res
}
