public class Solution {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        PrintWriter out = new PrintWriter(System.out);

        int n = in.nextInt();

        int count = 0;
        while (true) {
            if (count * 4 > n) {
                out.print(-1);
                break;
            } else if ((n - count * 4) % 7 == 0) {
                for (int i = 0; i < count; i++)
                    out.print(4);
                for (int i = 0; i < (n - count * 4) / 7; i++)
                    out.print(7);
                break;
            }
            count++;
        }

        out.close();
    }
}

struct Scanner {

impl Scanner {
    fn new() -> Self {

        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }
    fn next_i32(&mut self) -> i32 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
}
