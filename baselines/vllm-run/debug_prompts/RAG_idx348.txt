        Self { input: input.into_bytes(), idx: 0 }
    }

    fn next_i32(&mut self) -> i32 {
        self.skip_whitespace();
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.input.len() {
            let c = self.input[self.idx];
            if c.is_ascii_digit() {
                num = num * 10 + (c - b'0') as i32;
                self.idx += 1;
            } else {
                break;
            }
        }
        num * sign
    }

    fn next_string(&mut self) -> String {
        self.skip_whitespace();
        let start = self.idx;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        String::from_utf8(self.input[start..self.idx].to_vec()).unwrap()
    }

    fn skip_whitespace(&mut self) {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
    }

use std::io::{self, Read};

struct Scanner {
    input: Vec<u8>,
    idx: usize,
}
impl Scanner {
    fn new() -> Self {
        let mut s = String::new();
        io::stdin().read_to_string(&mut s).unwrap();
        Self { input: s.into_bytes(), idx: 0 }
    }
    fn next_i32(&mut self) -> i32 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
    fn next_line(&mut self) -> String {
        if self.idx >= self.input.len() {
            return String::new();
        }
        // Read until '\n' (like Java Scanner.nextLine())
        let start = self.idx;
        while self.idx < self.input.len() && self.input[self.idx] != b'\n' {
            self.idx += 1;
        }
        let end = self.idx;
        if self.idx < self.input.len() && self.input[self.idx] == b'\n' {
            self.idx += 1;
        }
        // Strip trailing '\r' (Windows line endings) similar to typical handling
        let mut slice = &self.input[start..end];
        if !slice.is_empty() && slice[slice.len() - 1] == b'\r' {
            slice = &slice[..slice.len() - 1];
        }
        String::from_utf8_lossy(slice).into_owned()
    }
}

fn main() {
    let mut scan = Scanner::new();
    let t: i32 = scan.next_i32();

    let mut o = String::new();
    for _i in 0..=(t as i32) {

        let n = x.len();
        if n > 10 {

            o.push(bytes[0] as char);
            o.push_str(&(n as i32 - 2).to_string());
            o.push(bytes[n - 1] as char);
            o.push('\n');
        } else {
            o.push_str(&x);
            o.push('\n');
        }
    }
    print!("{}", o);
}
