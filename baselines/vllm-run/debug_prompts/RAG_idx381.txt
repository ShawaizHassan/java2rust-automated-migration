        let mut data = Vec::new();
        input.read_to_end(&mut data).unwrap();
        Self { data, idx: 0 }
    }

    fn next_int(&mut self) -> i32 {
        let mut negative = false;
        loop {
            if self.idx >= self.data.len() {
                panic!("EOF");
            }
            let b = self.data[self.idx];
            self.idx += 1;
            if b == b'-' {
                negative = true;
                continue;
            }
            if b >= b'0' && b <= b'9' {
                let mut r: i32 = (b - b'0') as i32;

                    let d = self.data[self.idx];
                    if d >= b'0' && d <= b'9' {
                        self.idx += 1;
                        r = r * 10 + (d - b'0') as i32;
                    } else {
                        self.idx += 1;
                        break;
                    }
                }
                return if negative { -r } else { r };
            }
        }
    }
