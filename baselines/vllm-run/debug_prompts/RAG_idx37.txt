use std::io::{self, Read};

struct FastScanner {
    buf: Vec<u8>,
    idx: usize,
}

impl FastScanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { buf: input.into_bytes(), idx: 0 }
    }

    fn next_token(&mut self) -> Option<String> {
        while self.idx < self.buf.len() && self.buf[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        if self.idx >= self.buf.len() {
            return None;
        }
        let start = self.idx;
        while self.idx < self.buf.len() && !self.buf[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        Some(String::from_utf8_lossy(&self.buf[start..self.idx]).into_owned())
    }

    fn next_usize(&mut self) -> usize {
        self.next_token().unwrap().parse::<usize>().unwrap()
    }
}

fn total_str(s: &str) -> i32 {
    let mut t = s.trim();
    if let Some(rest) = t.strip_prefix('+') {
        t = rest;
    } else if let Some(rest) = t.strip_prefix('-') {
        t = rest;
    }
    let trimmed = t.trim_start_matches('0');
    if trimmed.is_empty() {
        1
    } else {
        trimmed.len() as i32
    }
}

fn main() {
    let mut sc = FastScanner::new();
    let t = sc.next_usize();
    let mut out = String::new();

    for i in 0..t {
        let tok = sc.next_token().unwrap_or_else(|| "0".to_string());
        let s = total_str(&tok);
        out.push_str(&s.to_string());
        if i + 1 < t {
            out.push('\n');
        }
    }

    print!("{out}");
}
