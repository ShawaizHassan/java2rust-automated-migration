fn is_prime(x: i32) -> bool {
    if x == 1 {
        return false;
    }
    if x == 2 {
        return true;
    }
    if x % 2 == 0 {
        return false;
    }
    let sqrtx = (x as f64).sqrt() as i32;
    let mut i = 3;
    while i <= sqrtx {
        if x % i == 0 {
            return false;
        }
        i += 2;
    }
    true
}

fn next_prime(mut x: i32, dp: &mut Vec<i32>) -> i32 {
    if x < 0 || x as usize >= dp.len() {

        // Fallback: compute directly without memoization.
        while !is_prime(x) {
            x += 1;

    loop {

        if dp[idx] != -1 {
            let res = dp[idx];

                dp[v as usize] = res;

            return res;
