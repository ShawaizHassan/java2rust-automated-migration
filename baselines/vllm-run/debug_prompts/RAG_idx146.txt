static mut TOTAL: i32 = 0;
static mut CNT: i32 = 0;

fn reverse_range(i: usize, j: usize, p: &mut [i32]) {
    let mut l = i;
    let mut r = j;
    while l < r {
        p.swap(l, r);
        l += 1;
        r -= 1;
    }
}

fn inversion_count(p: &[i32]) -> i32 {
    let mut c: i32 = 0;
    for i in 0..p.len() {
        for j in (i + 1)..p.len() {
            if p[i] > p[j] {
                c += 1;
            }
        }
    }
    c
}

fn solve(k: i32, p: &mut Vec<i32>) {
    if k == 0 {
        unsafe {
            TOTAL += inversion_count(p);
            CNT += 1;
        }
        return;
    }
    let n = p.len();
    for i in 0..n {
        for j in i..n {
            reverse_range(i, j, p);
            solve(k - 1, p);
            reverse_range(i, j, p);
        }
    }
}
