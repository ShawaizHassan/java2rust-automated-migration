use std::collections::HashSet;
use std::io::{self, Read};

struct Scanner {
    input: Vec<u8>,
    idx: usize,
}

impl Scanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self {
            input: input.into_bytes(),
            idx: 0,
        }
    }

    fn next(&mut self) -> String {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let start = self.idx;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        String::from_utf8(self.input[start..self.idx].to_vec()).unwrap()
    }

    fn next_i32(&mut self) -> i32 {
        self.next().parse::<i32>().unwrap()
    }
}

fn main() {
    let mut sc = Scanner::new();
    let mut n = sc.next_i32();
    let mut set: HashSet<i32> = HashSet::new();

    while n > 0 {
        let current = sc.next();
        let mut mask: i32 = 0;
        for b in current.bytes() {
            let idx = (b - b'a') as i32;
            mask |= 1i32 << idx;
        }
        set.insert(mask);
        n -= 1;
    }

    print!("{}", set.len());
}
