fn dp(idx: usize, won_so_far: usize, rem_cap: usize) -> f64 {
    unsafe {
        if idx == N {
            if won_so_far < MIN_TO_WIN || rem_cap < won_so_far {
                return 0.0;
            } else {
                return 1.0;
            }
        }

        let v = MEMO[idx][won_so_far][rem_cap];
        if v != -1.0 {
            return v;
        }

        let mut rem_c = rem_cap as i32 + CAPACITY[idx] + 1;
        if rem_c > 200 {
            rem_c = 200;

            rem_c = 0;

        let ans = take + leave;
        MEMO[idx][won_so_far][rem_cap] = ans;
        ans
    }
}
