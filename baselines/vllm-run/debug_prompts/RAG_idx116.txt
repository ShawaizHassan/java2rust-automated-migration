use std::io::{self, Read};

#[derive(Clone, Copy)]
struct Node {
    ind: i32,
    val: i32,
}

fn cmp(a: &Node, b: &Node) -> i32 {
    if a.val > b.val {
        -1
    } else if a.val < b.val {
        1
    } else {
        0
    }
}

// Java-like PriorityQueue with comparator; min-heap by cmp; no swaps on cmp == 0.
struct JPriorityQueue {
    data: Vec<Node>,
}

impl JPriorityQueue {
    fn with_capacity(cap: usize) -> Self {
        Self {
            data: Vec::with_capacity(cap),
        }
    }

    fn is_empty(&self) -> bool {
        self.data.is_empty()
    }

    fn add(&mut self, x: Node) {
        self.data.push(x);
        self.sift_up(self.data.len() - 1);
    }

    fn poll(&mut self) -> Node {
        let n = self.data.len();
        let res = self.data[0];
        if n == 1 {
            self.data.pop();
            return res;
        }
        let last = self.data.pop().unwrap();
        self.data[0] = last;
        self.sift_down(0);
        res
    }

    fn sift_up(&mut self, mut idx: usize) {
        let x = self.data[idx];
        while idx > 0 {
            let parent = (idx - 1) / 2;
            if cmp(&x, &self.data[parent]) >= 0 {
                break;
            }
            self.data[idx] = self.data[parent];
            idx = parent;
        }
        self.data[idx] = x;
    }

    fn sift_down(&mut self, mut idx: usize) {
        let n = self.data.len();
        let x = self.data[idx];
        loop {
            let left = idx * 2 + 1;
            if left >= n {
                break;
            }
            let mut child = left;
            let right = left + 1;
            if right < n && cmp(&self.data[right], &self.data[left]) < 0 {
                child = right;
            }
            if cmp(&x, &self.data[child]) <= 0 {
                break;
            }
            self.data[idx] = self.data[child];
            idx = child;
        }
        self.data[idx] = x;
    }
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = match it.next() {
        Some(s) => s.parse().unwrap(),
        None => return,
    };

    let mut q = JPriorityQueue::with_capacity(n);
    for i in 0..n {
        let v: i32 = it.next().unwrap().parse().unwrap();
        q.add(Node {
            ind: (i as i32) + 1,
            val: v,
        });
    }

    let mut team1: Vec<i32> = Vec::new();
    let mut team2: Vec<i32> = Vec::new();

    let mut temp: i32 = 1;
    while !q.is_empty() {
        if temp > 0 {
            team1.push(q.poll().ind);
            temp *= -1;
        } else {
            team2.push(q.poll().ind);
            temp *= -1;
        }
    }

    let mut out = String::new();
    out.push_str(&format!("{}\n", team1.len()));
    for &x in &team1 {
        out.push_str(&format!("{} ", x));
    }

    out.push_str(&format!("{}\n", team2.len()));
    for &x in &team2 {
        out.push_str(&format!("{} ", x));
    }
    out.push('\n');

    print!("{}", out);
}
