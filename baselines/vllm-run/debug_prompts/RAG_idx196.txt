fn solve(dp: &mut Vec<Vec<i64>>, a: &Vec<i32>, n: usize, prev: usize, _sum: i64) -> i64 {
    if n == 0 {
        return 0;
    }
    if dp[n][prev] != -1 {
        return dp[n][prev];
    }
    let res = if prev == 0 {
        let take = a[n - 1] as i64 + solve(dp, a, n - 1, 1, _sum);
        let skip = solve(dp, a, n - 1, 0, _sum);
        take.max(skip)

    res
}
