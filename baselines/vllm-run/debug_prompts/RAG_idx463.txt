struct Scanner {
    input: Vec<u8>,
    idx: usize,
}
impl Scanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }
    fn skip_whitespace(&mut self) {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
    }
    fn next_i32(&mut self) -> i32 {
        self.skip_whitespace();
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
    fn next_line(&mut self) -> String {
        if self.idx < self.input.len() && self.input[self.idx] == b'\n' {
            self.idx += 1;
        }
        let start = self.idx;
        while self.idx < self.input.len() && self.input[self.idx] != b'\n' {
            self.idx += 1;
        }
        let end = self.idx;
        if self.idx < self.input.len() && self.input[self.idx] == b'\n' {
            self.idx += 1;
        }
        String::from_utf8_lossy(&self.input[start..end]).to_string()
    }
}

use std::io::{self, Read};

fn solve(s: &[u8], l: usize, r: usize, cur: u8) -> i32 {
    if l == r {
        return if s[l] == cur { 0 } else { 1 };
    }
    let mid = (l + r) / 2;

    let mut c1: i32 = 0;
    let mut c2: i32 = 0;

    for &ch in &s[l..=mid] {
        if ch != cur {
            c1 += 1;
        }
    }
    for &ch in &s[mid + 1..=r] {
        if ch != cur {
            c2 += 1;
        }
    }

    let tmp1 = c1 + solve(s, mid + 1, r, cur + 1);
    let tmp2 = c2 + solve(s, l, mid, cur + 1);

    if tmp1 < tmp2 { tmp1 } else { tmp2 }
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let t: i32 = it.next().unwrap().parse().unwrap();
    let mut out = String::new();

    for _ in 0..t {
        let n: usize = it.next().unwrap().parse().unwrap();
        let line = it.next().unwrap();
        let bytes = line.as_bytes();
        let ans = solve(bytes, 0, n - 1, b'a');
