use std::io::{self, Read};

struct LineReader {
    parts: Vec<String>,
    idx: usize,
}

impl LineReader {
    fn new(input: String) -> Self {
        let mut parts: Vec<String> = input
            .split('\n')
            .map(|s| {
                let s = if s.ends_with('\r') { &s[..s.len() - 1] } else { s };
                s.to_string()
            })
            .collect();

        // If input ends with '\n', split will include a final empty segment already.
        // If not, behavior matches typical line reading as well.
        LineReader { parts, idx: 0 }
    }

    fn next_line(&mut self) -> String {
        if self.idx >= self.parts.len() {
            return String::new();
        }
        let s = self.parts[self.idx].clone();
        self.idx += 1;
        s
    }
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut lr = LineReader::new(input);

    let first_line = lr.next_line();
    let n: usize = first_line
        .split_whitespace()
        .next()
        .unwrap_or("0")
        .parse::<usize>()
        .unwrap_or(0);

    let mut x: Vec<String> = Vec::with_capacity(n);
    for _ in 0..n {
        x.push(lr.next_line());
    }

use std::io::{self, Read};

struct Scanner {
    input: Vec<u8>,
    idx: usize,
}

impl Scanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }

    fn next_i32(&mut self) -> i32 {
        self.skip_whitespace();
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.input.len() {
            let c = self.input[self.idx];
            if c.is_ascii_digit() {
                num = num * 10 + (c - b'0') as i32;
                self.idx += 1;
            } else {
                break;
            }
        }
        num * sign
    }

    fn next_string(&mut self) -> String {
        self.skip_whitespace();
        let start = self.idx;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        String::from_utf8(self.input[start..self.idx].to_vec()).unwrap()
    }

    fn skip_whitespace(&mut self) {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
    }
}

fn main() {
    let mut sc = Scanner::new();
    let t = sc.next_i32();
    let mut out = String::new();

    for _ in 0..t {
        let n = sc.next_i32() as usize;
        let mut arr = [0i32; 26];

        for _ in 0..n {
            let st = sc.next_string();
            for &b in st.as_bytes() {
                let idx = (b - b'a') as usize;
                arr[idx] += 1;
            }
        }

        let mut ok = true;
        for i in 0..26 {
            if (arr[i] as usize) % n != 0 {
                ok = false;
                break;
            }
        }

        if ok {
            out.push_str("YES\n");
        } else {
            out.push_str("NO\n");
        }
    }

    print!("{}", out);
}
