const NMAX: usize = 1_000_006;

fn sieve() -> Vec<bool> {
    let mut f = vec![true; NMAX + 1];
    if NMAX >= 0 {
        f[0] = false;
    }
    if NMAX >= 1 {
        f[1] = false;
    }
    for i in 2..=NMAX {
        if f[i] {
            let mut j = i * 2;
            while j <= NMAX {
                f[j] = false;
                j += i;
            }
        }
    }
    f
}
