fn main() {
    // Fast input
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    // Sieve primes up to MAX-1
    let mut prime = vec![true; MAX];
    if MAX > 0 {
        prime[0] = false;
    }
    if MAX > 1 {
        prime[1] = false;
    }
    for i in 2..MAX {
        if prime[i] {
            let mut j = i * 2;
            while j < MAX {
                prime[j] = false;
                j += i;
            }
        }
    }

    // Precompute next prime >= v for all v in [0, MAX)
    let mut next_prime = vec![0usize; MAX];
    let mut next = None::<usize>;
    for v in (0..MAX).rev() {
        if prime[v] {
            next = Some(v);
        }
        next_prime[v] = next.unwrap_or(MAX); // MAX as sentinel (shouldn't be needed)
    }

    let n: usize = match it.next() {
        Some(x) => x.parse().unwrap(),
        None => return,

                (np as i32) - (e as i32)
            };
            rowsum[i] += add;
            clmsum[j] += add;
