#[derive(Clone)]
struct X {
    s: i32,
    d: i32,
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };
    let k: usize = it.next().unwrap().parse().unwrap();

    let mut arr: Vec<X> = Vec::with_capacity(n);
    for _ in 0..n {
        let s: i32 = it.next().unwrap().parse().unwrap();
        let d: i32 = it.next().unwrap().parse().unwrap();
        arr.push(X { s, d });
    }

    arr.sort_by(|a, b| {
        match b.s.cmp(&a.s) {
            std::cmp::Ordering::Equal => a.d.cmp(&b.d),
            other => other,
        }
    });

    let a1 = arr[k - 1].s;
    let a2 = arr[k - 1].d;

    let mut ans: i32 = 0;
    for x in &arr {
        if x.s == a1 && x.d == a2 {
            ans += 1;
        }
    }

    print!("{ans}\n");
}
