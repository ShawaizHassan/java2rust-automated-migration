use std::io::{self, Read};

fn seq_left(mut i: usize, tab: &Vec<i32>) -> i32 {
    let mut seq: i32 = 0;
    while i > 0 && tab[i] >= tab[i - 1] {
        seq += 1;
        i -= 1;
    }
    seq
}

fn seq_right(mut i: usize, tab: &Vec<i32>) -> i32 {
    let mut seq: i32 = 0;
    while i + 1 < tab.len() && tab[i] >= tab[i + 1] {
        seq += 1;
        i += 1;
    }
    seq
}

fn max_seq(tab: &Vec<i32>) -> i32 {
    let mut maxv: i32 = 0;
    for i in 0..tab.len() {
        let tot_seq = seq_left(i, tab) + seq_right(i, tab) + 1;
        if tot_seq >= maxv {
            maxv = tot_seq;
        }
    }
    maxv
}
