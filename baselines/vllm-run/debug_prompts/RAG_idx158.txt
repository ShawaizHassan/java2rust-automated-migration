use std::io::{self, Read};

fn strip_cr(s: &str) -> &str {
    if s.ends_with('\r') {
        &s[..s.len() - 1]
    } else {
        s
    }
}

    let mut it = input.splitn(3, '\n');

    let t_line = it.next().unwrap_or("");

    let s = strip_cr(s_line);
    let t = strip_cr(t_line);

    let a: Vec<u16> = s.encode_utf16().collect();
    let b: Vec<u16> = t.encode_utf16().collect();

    let n = a.len();
    let m = b.len();

    let m_i32 = m as i32;
