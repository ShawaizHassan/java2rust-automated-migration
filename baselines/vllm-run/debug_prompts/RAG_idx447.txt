use std::cmp::Ordering;
use std::io::{self, Read};

#[derive(Clone, Copy)]
struct Point {
    x: i32,
    y: i32,
}

struct FastScanner {
    input: Vec<u8>,
    idx: usize,
}

impl FastScanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self {
            input: input.into_bytes(),
            idx: 0,
        }
    }

    fn next_i32(&mut self) -> i32 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
}

fn main() {
    let mut fs = FastScanner::new();
    let n = fs.next_i32() as usize;

    let mut array: Vec<Point> = Vec::with_capacity(n);
    for i in 0..n {
        let v = fs.next_i32();
        array.push(Point { x: v, y: (i + 1) as i32 });
    }

    array.sort_by(|a, b| {
        let d = a.x - b.x;
        if d < 0 {
            Ordering::Less
        } else if d > 0 {
            Ordering::Greater
        } else {
            Ordering::Equal
        }
    });

    let mut sum1x: i64 = 0;
    let mut sum1y: i32 = 0;
    let mut sum2x: i64 = 0;
    let mut sum2y: i32 = 0;

    for i in (0..n).rev() {
        if i % 2 == 0 {
            sum1y += 1;
            sum1x += array[i].x as i64;
        }
        if i % 2 == 1 {
            sum2y += 1;
            sum2x += array[i].x as i64;
        }
    }

    let mut out = String::new();

    out.push_str(&format!("{}\n", sum1y));
    for i in (0..n).rev() {
        if i % 2 == 0 {
            out.push_str(&format!("{} ", array[i].y));
