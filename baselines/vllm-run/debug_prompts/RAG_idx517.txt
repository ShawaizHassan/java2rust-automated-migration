fn binary_search(al: &Vec<Pair>, key: i32, mut low: i32, mut high: i32) -> i32 {
    while low <= high {
        let mid = ((low + high) as u32 >> 1) as i32; // emulate Java >>> 1 on int
        let p = &al[mid as usize];
        if p.x == key {
            return p.y;
        } else if p.x > key {
            high = mid - 1;
        } else {
            low = mid + 1;
        }
    }
    -1
}

use std::cmp::Ordering;
use std::io::{self, Read};

#[derive(Clone, Copy)]
struct Pair {
    p: i32,
    t: i32,
}

struct FastScanner {
    input: Vec<u8>,
    idx: usize,
}

impl FastScanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self {
            input: input.into_bytes(),
            idx: 0,
        }
    }

    fn next_i32(&mut self) -> i32 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
}

fn solve(fs: &mut FastScanner) -> i32 {
    let n = fs.next_i32() as usize;
    let k = fs.next_i32() as usize;
    let mut a: Vec<Pair> = Vec::with_capacity(n);
    for _ in 0..n {
        let p = fs.next_i32();
        let t = fs.next_i32();
        a.push(Pair { p, t });
    }

    a.sort_by(|x, y| match y.p.cmp(&x.p) {
        Ordering::Equal => x.t.cmp(&y.t),
        other => other,
    });

    let target = a[k - 1];
    let mut count = 0i32;
    for i in 0..n {
        if a[i].p == target.p && a[i].t == target.t {
            count += 1;
        }
    }
    count
}

fn main() {
    let mut fs = FastScanner::new();
    let ans = solve(&mut fs);
    print!("{}\n", ans);
}
