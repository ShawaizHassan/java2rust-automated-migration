use std::io::{self, Read};

struct FastScanner {
    input: Vec<u8>,
    idx: usize,
}
impl FastScanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }
    fn next_i32(&mut self) -> i32 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
}

fn main() {
    let mut fs = FastScanner::new();
    let t = fs.next_i32();
    let mut out = String::new();

    for _ in 0..t {
        let n = fs.next_i32();
        let m = fs.next_i32();
        let a = fs.next_i32();
        let b = fs.next_i32();

        if (n as i64) * (a as i64) == (m as i64) * (b as i64) {
            out.push_str("YES\n");

            let mut tmp = String::new();
            if a > 0 {
                tmp.extend(std::iter::repeat('1').take(a as usize));
            }
            // Java: String.join("0", Collections.nCopies(m-a+1, ""))
            // produces exactly m-a zeros.
            let zeros = m - a;
            if zeros > 0 {
                tmp.extend(std::iter::repeat('0').take(zeros as usize));
            }

            let mut s: i32 = 0;
            for cand in 1..m {
                if ((cand as i64) * (n as i64)) % (m as i64) == 0 {
                    s = cand;
                    break;
                }
            }

            let bytes = tmp.as_bytes();
            for i in 0..(n as usize) {
                let j = ((i as i64) * (s as i64)).rem_euclid(m as i64) as usize;
                let k = (m as usize).wrapping_sub(j);
                out.push_str(std::str::from_utf8(&bytes[k..]).unwrap());
                out.push_str(std::str::from_utf8(&bytes[..k]).unwrap());
                out.push('\n');
            }
        } else {
            out.push_str("NO\n");
        }
    }

    print!("{out}");
}
