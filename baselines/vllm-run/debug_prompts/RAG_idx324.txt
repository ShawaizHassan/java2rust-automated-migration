struct Scanner {
    input: Vec<u8>,
    idx: usize,
}
impl Scanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }
    fn next_i32(&mut self) -> Option<i32> {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        if self.idx >= self.input.len() {
            return None;
        }
        let mut sign = 1i32;
        if self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num: i32 = 0;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        Some(num * sign)
    }
}

fn fun(x: i32, y: i32, z: i32) -> i32 {
    if z <= x {
        return y - z;
    }
    if z >= y {
        return z - x;
    }
    std::cmp::min(z - x + y - x, y - z + y - x)
}

fn main() {
    let mut sc = Scanner::new();
    let mut out = String::new();

    while let Some(n) = sc.next_i32() {
        let a = match sc.next_i32() {
            Some(v) => v,
            None => break,
        };
        let n_usize = n as usize;
        let mut aa: Vec<i32> = vec![0; n_usize];
        for i in 0..n_usize {
            aa[i] = sc.next_i32().unwrap();
        }
        aa.sort();

        if n_usize == 1 {
            out.push_str("0\n");
            continue;
        }

        let mut ans: i32 = i32::MAX;
        if a <= aa[0] {
            ans = std::cmp::min(ans, aa[n_usize - 2] - a);
        } else if a >= aa[n_usize - 1] {
            ans = std::cmp::min(ans, a - aa[1]);

            ans = std::cmp::min(ans, fun(aa[0], aa[n_usize - 2], a));
        }
        out.push_str(&format!("{}\n", ans));
    }

    print!("{}", out);
}
