fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();

    // BufferedReader.readLine() strips '\n' and not '\r'. Handle possible CRLF.
    let mut lines = input.split('\n');

    let set = fill();
    let mut out = String::new();

    loop {
        let line_opt = lines.next();
        if line_opt.is_none() {
            break;
        }
        let mut line = line_opt.unwrap();
        if line.ends_with('\r') {
            line = &line[..line.len() - 1];
        }
        if line.is_empty() {
            break;
        }

        let times: i32 = line.parse().unwrap();
        let mut sum: i32 = 0;

        for _ in 0..(times as usize) {
            let mut s = lines.next().unwrap_or("");
            if s.ends_with('\r') {
                s = &s[..s.len() - 1];
            }
            let first = s.as_bytes()[0];
            if (first >= b'0' && first <= b'9' && s.parse::<i32>().unwrap() < 18)
                || set.contains(s)
            {
                sum += 1;
            }
        }

        out.push_str(&sum.to_string());
        out.push('\n');
    }

use std::collections::{HashMap, HashSet};
use std::io::{self, Read};

struct Scanner {
    input: Vec<u8>,
    idx: usize,
}
impl Scanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }
    fn next_i32(&mut self) -> i32 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
}

fn main() {
    let mut sc = Scanner::new();
    let n = sc.next_i32() as usize;
    let m = sc.next_i32() as usize;

    let mut p: Vec<i32> = vec![0; n];
    for i in 0..n {
        p[i] = sc.next_i32();
    }

    let mut map: HashMap<i32, HashSet<i32>> = HashMap::new();
    for _ in 0..m {
        let key = sc.next_i32();
        let value = sc.next_i32();
        map.entry(key).or_insert_with(HashSet::new).insert(value);
    }

    if n == 1 {
        print!("0");
        return;
    }

    let mut result: i32 = 0;
    let mut current_queue: HashSet<i32> = HashSet::new();
    current_queue.insert(p[n - 1]);

    for i in (0..=n - 2).rev() {
        let pi = p[i];
        if let Some(set) = map.get(&pi) {
            let mut ok = true;
            for v in current_queue.iter() {
                if !set.contains(v) {
                    ok = false;
                    break;
                }
            }
            if ok {
                result += 1;
            } else {
                current_queue.insert(pi);
            }
        } else {
            current_queue.insert(pi);
        }
    }

    print!("{}", result);
}
