use std::io::{self, Read};

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();
    let n: i64 = match it.next() {
        Some(v) => v.parse::<i64>().unwrap(),
        None => return,
    };
    let k: i64 = it.next().unwrap().parse::<i64>().unwrap();

    let mut out = String::new();

    if k < n / 2 || (n == 1 && k > 0) {
        out.push_str("-1\n");
    } else if k == n / 2 {
        for i in 1..=n {
            out.push_str(&format!("{} ", i));
        }
    } else if n == 1 && k == 0 {
        out.push_str("0\n");
    } else {
        let base = k - n / 2 + 1;
        out.push_str(&format!("{} {} ", base, 2 * base));
        if base > n - 2 {
            for i in 1..=(n - 2) {
                out.push_str(&format!("{} ", i));
            }
        } else {
            for i in 1..=(n - 2) {
                out.push_str(&format!("{} ", 2 * base + i));
            }
        }
    }

    print!("{}", out);
}
