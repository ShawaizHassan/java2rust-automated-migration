struct InputReader {
    buf: Vec<u8>,
    idx: usize,
}

impl InputReader {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { buf: input.into_bytes(), idx: 0 }
    }

    fn is_whitespace(c: u8) -> bool {
        c <= 32
    }

    fn next_i32(&mut self) -> i32 {
        while self.idx < self.buf.len() && Self::is_whitespace(self.buf[self.idx]) {
            self.idx += 1;
        }
        let mut sgn: i32 = 1;
        if self.idx < self.buf.len() && self.buf[self.idx] == b'-' {
            sgn = -1;
            self.idx += 1;
        }
        let mut ret: i32 = 0;
        while self.idx < self.buf.len() && !Self::is_whitespace(self.buf[self.idx]) {
            let c = self.buf[self.idx];
            if c < b'0' || c > b'9' {
                panic!("Invalid character");
            }
            ret = ret * 10 + (c - b'0') as i32;
            self.idx += 1;
        }
        ret * sgn
    }
}

use std::io::{self, Read, Write};

struct InputReader {
    buf: Vec<u8>,
    idx: usize,
}

impl InputReader {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self {
            buf: input.into_bytes(),
            idx: 0,
        }
    }

    fn next(&mut self) -> String {
        while self.idx < self.buf.len() && self.buf[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let start = self.idx;
        while self.idx < self.buf.len() && !self.buf[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        String::from_utf8(self.buf[start..self.idx].to_vec()).unwrap()
    }
}

fn main() {
    let mut inr = InputReader::new();
    let s = inr.next();

    let mut balance: i32 = 0;
    for b in s.bytes() {
        if b == b'x' {
            balance += 1;
        } else {
            balance -= 1;
        }
    }

    let mut out = String::new();
    if balance < 0 {
        out.extend(std::iter::repeat('y').take((-balance) as usize));
    } else {
        out.extend(std::iter::repeat('x').take(balance as usize));
    }
    out.push('\n');

    let mut stdout = io::BufWriter::new(io::stdout());
    stdout.write_all(out.as_bytes()).unwrap();
}
