    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

use std::io::{self, Read};

fn java_abs_i32(x: i32) -> i32 {
    if x == i32::MIN { i32::MIN } else { x.abs() }
}

fn gcd(mut a: i32, mut b: i32) -> i32 {
    // Mirrors the Java recursive version; assumes b != 0 as in the original.
    loop {
        if a % b == 0 {
            return b;
        } else {
            let r = a % b;
            a = b;
            b = r;
        }
    }
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: i32 = it.next().unwrap().parse().unwrap();
    let m: i32 = it.next().unwrap().parse().unwrap();

    let mut a0: i32 = 0;
    for i in 0..=n {
        let v: i32 = it.next().unwrap().parse().unwrap();
        if i == 0 { a0 = v; }
    }

    let mut b0: i32 = 0;
    for i in 0..=m {
        let v: i32 = it.next().unwrap().parse().unwrap();
        if i == 0 { b0 = v; }
    }

    if n > m {
        let prod = a0.wrapping_mul(b0);
        if prod > 0 {
            println!("Infinity");
        } else {
            println!("-Infinity");
        }
    } else if n < m {

        if a0.wrapping_mul(b0) < 0 {

        }
        s.push_str(&format!("{}/{}", java_abs_i32(a0), java_abs_i32(b0)));
