        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }

    fn next_i32(&mut self) -> i32 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;

            num = num * 10 + (self.input[self.idx] - b'0') as i32;

        num * sign
    }

use std::collections::HashSet;
use std::io::{self, Read};

struct FastScanner {
    input: Vec<u8>,
    idx: usize,
}

impl FastScanner {
    fn new() -> Self {
        let mut input = Vec::new();
        io::stdin().read_to_end(&mut input).unwrap();
        Self { input, idx: 0 }
    }

    fn skip_whitespace(&mut self) {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
    }

    fn next_i32(&mut self) -> i32 {
        self.skip_whitespace();
        let mut sign: i32 = 1;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num: i32 = 0;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
}

fn main() {
    let mut fs = FastScanner::new();
    let n = fs.next_i32() as usize;

    let mut all_set: HashSet<i32> = HashSet::new();
    let mut cur_set: HashSet<i32> = HashSet::new();

    for _ in 0..n {
        let x = fs.next_i32();
        let mut next_set: HashSet<i32> = HashSet::new();
        next_set.insert(x);
        for &z in cur_set.iter() {
            next_set.insert(x | z);
        }
        all_set.extend(next_set.iter().copied());
        cur_set = next_set;
    }
