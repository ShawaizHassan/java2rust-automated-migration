use std::collections::VecDeque;
use std::io::{self, Read};

struct Scanner {
    input: Vec<u8>,
    idx: usize,
}

impl Scanner {
    fn new() -> Self {
        let mut input = Vec::new();
        io::stdin().read_to_end(&mut input).unwrap();
        Self { input, idx: 0 }
    }

    fn next<T: std::str::FromStr>(&mut self) -> T {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let start = self.idx;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        std::str::from_utf8(&self.input[start..self.idx])
            .unwrap()
            .parse::<T>()
            .ok()
            .unwrap()
    }
}

fn main() {
    let mut sc = Scanner::new();
    let n: i32 = sc.next();
    let x: String = sc.next();

    let chars: Vec<char> = x.chars().collect();
    let mut out: VecDeque<char> = VecDeque::new();
    out.push_back(chars[0]);

    for i in 1..(n as usize) {
        if i % 2 == 0 {
            out.push_back(chars[i]);
        } else {
            out.push_front(chars[i]);
        }
    }

    let mut res = String::with_capacity(n as usize);
    if n % 2 == 0 {
        for &ch in out.iter().rev() {
            res.push(ch);
        }
    } else {

        }
