    input: Vec<u8>,
    idx: usize,
}
impl FastScanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }
    fn next_i32(&mut self) -> i32 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
}

use std::io::{self, Read};

struct FastReader {
    input: Vec<u8>,
    idx: usize,
}

impl FastReader {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self {
            input: input.into_bytes(),
            idx: 0,
        }
    }

    fn next(&mut self) -> String {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let start = self.idx;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        String::from_utf8_lossy(&self.input[start..self.idx]).to_string()
    }

    fn next_i32(&mut self) -> i32 {
        self.next().parse::<i32>().unwrap()
    }

    #[allow(dead_code)]
    fn next_i64(&mut self) -> i64 {
        self.next().parse::<i64>().unwrap()
    }

    #[allow(dead_code)]
    fn next_f64(&mut self) -> f64 {
        self.next().parse::<f64>().unwrap()
    }

    #[allow(dead_code)]
    fn next_line(&mut self) -> String {
        if self.idx >= self.input.len() {
            return String::new();
        }
        if self.idx < self.input.len() && self.input[self.idx] == b'\n' {
            self.idx += 1;
        }
        let start = self.idx;
        while self.idx < self.input.len() && self.input[self.idx] != b'\n' {
            self.idx += 1;
        }
        if self.idx < self.input.len() && self.input[self.idx] == b'\n' {
            self.idx += 1;
        }
        String::from_utf8_lossy(&self.input[start..self.idx]).to_string()
    }
