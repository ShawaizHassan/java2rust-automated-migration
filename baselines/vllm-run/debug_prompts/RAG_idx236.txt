    fn next_i32(&mut self) -> i32 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;

        let mut sign = 1i32;

            self.idx += 1;

            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
}

use std::io::{self, Read};

const N: usize = 100_000 + 10;

struct Scanner {
    input: Vec<u8>,
    idx: usize,
}

impl Scanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self {
            input: input.into_bytes(),
            idx: 0,
        }
    }

    fn next(&mut self) -> String {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let start = self.idx;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        String::from_utf8(self.input[start..self.idx].to_vec()).unwrap()
    }

    fn next_i32(&mut self) -> i32 {
        self.next().parse::<i32>().unwrap()
    }
}

fn main() {
    let mut sc = Scanner::new();
    let t = sc.next_i32() as usize;

    let mut a: Vec<char> = vec!['\0'; N];
    let mut out = String::new();

    for _ in 0..t {
        let mut s = sc.next();
        s = format!("0{}0", s);

        let chars: Vec<char> = s.chars().collect();
        let n = chars.len();
        let mut flag = true;

        for i in 1..n - 1 {
            a[i] = chars[i];
            if chars[i] == '?' {
                if a[i - 1] != 'a' && chars[i + 1] != 'a' {
                    a[i] = 'a';
                }
                if a[i - 1] != 'b' && chars[i + 1] != 'b' {
                    a[i] = 'b';
                }
                if a[i - 1] != 'c' && chars[i + 1] != 'c' {
                    a[i] = 'c';
                }
            } else if chars[i] == chars[i + 1] {
                flag = false;
                out.push_str("-1\n");
                break;
            }
        }

        if flag {
            for i in 1..n - 1 {
                out.push(a[i]);
            }
            out.push('\n');
        }
    }

    print!("{out}");
}
