        io::stdin().read_to_string(&mut input).unwrap();

    }

    fn next_i32(&mut self) -> i32 {

            self.idx += 1;

            return i32::MIN;

            self.idx += 1;

fn find(p: &mut Vec<usize>, x: usize) -> usize {
    if p[x] != x {
        let root = find(p, p[x]);
        p[x] = root;
    }
    p[x]
}
