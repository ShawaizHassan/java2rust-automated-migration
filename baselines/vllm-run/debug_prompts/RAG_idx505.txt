        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }
    fn next_i32(&mut self) -> i32 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
}

use std::io::{self, Read};

struct Scanner {
    input: Vec<u8>,
    idx: usize,
}
impl Scanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }
    fn next_i32(&mut self) -> i32 {
        self.next_token().parse::<i32>().unwrap()
    }
    fn next_i64(&mut self) -> i64 {
        self.next_token().parse::<i64>().unwrap()
    }
    fn next_token(&mut self) -> String {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let start = self.idx;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        String::from_utf8(self.input[start..self.idx].to_vec()).unwrap()
    }
}

fn main() {
    let mut sc = Scanner::new();
    let n = sc.next_i32() as usize;
    let m = sc.next_i32() as usize;

    let mut b: Vec<i64> = vec![0; n];
    let mut g: Vec<i64> = vec![0; m];

    for i in 0..n {
        b[i] = sc.next_i64();
    }
    for i in 0..m {
        g[i] = sc.next_i64();
    }

    b.sort();
    g.sort();

        print!("-1\n");

        }
        if maxv != g[0] {
