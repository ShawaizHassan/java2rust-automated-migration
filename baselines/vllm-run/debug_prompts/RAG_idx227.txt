fn pow2(exp: i64) -> i64 {
    // matches Math.pow(2, pow) cast to long for typical contest ranges
    1i64 << (exp as u32)
}

fn solver(n: i32) -> String {
    let mut sum: i64 = 0;
    let n64 = n as i64;

    let mut x: i64 = 1;
    let mut pow: i64 = 2;

        sum = pow2(pow) - 1;
        x = n64 / sum;

        }
    }
