use std::io::{self, Read};

fn palin(st: &str) -> bool {
    let bytes = st.as_bytes();
    let l = bytes.len();
    for i in 0..(l / 2) {
        if bytes[i] != bytes[l - i - 1] {
            return false;
        }
    }
    true
}

fn check(st: &str) -> bool {
    let bytes = st.as_bytes();
    let l = bytes.len();
    if l == 0 {
        return true;
    }
    let first = bytes[0];
    for i in 1..=(l / 2) {
        if bytes[i] != first {
            return false;
        }
    }
    true
}

fn min_palin_seq(st: &str) -> bool {
    let mut s = st.to_string();
    loop {
        let l = s.len();
        if l % 2 != 0 {
            return false;
        } else {
            s = s[..(l / 2)].to_string();
            let l2 = s.len();
            let fl = palin(&s);
            if fl && l2 > 1 {
                continue;
            } else if l2 == 1 {
                return false;
            } else if !fl && l2 > 1 {
                return true;
            }
        }
        return true;
    }
}

use std::io::{self, Read};

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };
    let m: usize = it.next().unwrap().parse().unwrap();

    let mut ch = vec![false; m + 1];

    for _ in 0..n {
        let k: usize = it.next().unwrap().parse().unwrap();
        for _ in 0..k {
            let x: usize = it.next().unwrap().parse().unwrap();
            if x <= m {
                ch[x] = true;
            }
        }
    }

    for i in 1..=m {
        if !ch[i] {
            print!("NO\n");
            return;
        }
    }
