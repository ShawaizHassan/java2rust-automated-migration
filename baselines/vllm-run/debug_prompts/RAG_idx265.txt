            sign = -1;

        }
        num * sign
    }

fn main() {
    let mut scan = Scanner::new();
    let n = scan.next_i32() as usize;
    let mut h: Vec<i32> = vec![0; n];
    for i in 0..n {
        h[i] = scan.next_i32();
    }

    let mut max_floors: i32 = 0;
    let mut stack: Vec<i32> = Vec::with_capacity(n);

    for i in (0..n).rev() {
