use std::io::{self, Read};

struct Scanner {
    input: Vec<u8>,
    idx: usize,
}

impl Scanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }

    fn next_i64(&mut self) -> i64 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i64;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i64;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.input[self.idx] - b'0') as i64;
            self.idx += 1;
        }
        num * sign
    }
}

fn main() {
    let mut sc = Scanner::new();
    let t = sc.next_i64() as i32;
    let mut out = String::new();

    for _ in 0..t {
        let n = sc.next_i64();
        let r = sc.next_i64();
        let m: i64;
        if n <= r {
            m = n - 1;
            let res = (m * (2 + (m - 1))) / 2 + 1;
            out.push_str(&format!("{}\n", res));
        } else {
            m = r;
            let res = (m * (2 + (m - 1))) / 2;
            out.push_str(&format!("{}\n", res));
        }
    }

    print!("{}", out);
