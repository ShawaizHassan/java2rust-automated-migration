use std::io::{self, Read};

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: i32 = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };
    let k: i32 = it.next().unwrap().parse().unwrap();

    let mut last_diag: i32 = 0;
    let mut rem: i32 = k;

    let nn = n as usize;
    let mut mat = vec![vec![0i32; nn]; nn];

    for i in 0..nn {
        for j in i..nn {
            if rem == 1 && mat[i][j] == 0 {
                if last_diag == n {
                    print!("-1");
                    return;
                } else {
                    let d = last_diag as usize;
                    mat[d][d] = 1;
                    last_diag += 1;
                    rem -= 1;
                }
            } else if i == j && rem > 0 {
                mat[i][i] = 1;
                rem -= 1;
                last_diag += 1;
            } else if rem > 0 {
                mat[i][j] = 1;
                mat[j][i] = 1;
                rem -= 2;
            }
        }
    }

    if rem != 0 {
        print!("-1");
        return;
    }

    let mut out = String::new();
    for i in 0..nn {
        for j in 0..nn {
            out.push_str(&mat[i][j].to_string());
            out.push(' ');
        }
        out.push('\n');
    }
    print!("{}", out);
}
