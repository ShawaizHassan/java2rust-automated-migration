    let t: i32 = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };

    let mut out = String::new();

        let mut a: Vec<i32> = Vec::with_capacity(n);

            a.push(it.next().unwrap().parse::<i32>().unwrap());
        }

        let mut hs: HashSet<i32> = HashSet::with_capacity(n * 2 + 1);
        for &x in &a {
            hs.insert(x);

        let mut printed = false;

                    f = 0;

use std::collections::HashSet;
use std::io::{self, Read};

struct Scanner {
    input: Vec<u8>,
    idx: usize,
}
impl Scanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }
    fn next_i32(&mut self) -> i32 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
}

fn main() {
    let mut scan = Scanner::new();
    let n: i32 = scan.next_i32();

    let mut hs: HashSet<i32> = HashSet::new();
    let mut max_distinct: i32 = 0;
    let mut distinct: i32 = 0;

    for _ in 0..(2 * n) {
        let sock: i32 = scan.next_i32();
        if hs.contains(&sock) {
            distinct -= 1;
        } else {
            hs.insert(sock);
            distinct += 1;
        }
        if distinct > max_distinct {
            max_distinct = distinct;
        }
    }

    println!("{}", max_distinct);
