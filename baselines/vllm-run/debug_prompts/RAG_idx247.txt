fn find_root(word: &str) -> String {
    let mut new_word = String::new();
    for c in word.chars() {
        if !new_word.contains(c) {
            new_word.push(c);
        }
    }
    new_word
}

fn has_this_root(root1: &str, roots: &Vec<String>) -> bool {
    let root1_len = root1.chars().count();
    for r in roots {
        if r.chars().count() != root1_len {
            continue;
        }
        let mut ok = true;
        let mut i = 0usize;
        for c in root1.chars() {
            if !r.contains(c) {
                ok = false;
                break;
            }
            if i == root1_len - 1 {
                return true;
            }
            i += 1;
        }
        if ok && root1_len == 0 {
            return true;
        }
    }
    false
}

    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: i32 = match it.next() {

    let mut roots: Vec<String> = Vec::new();

            if !has_this_root(&root, &roots) {

            }
        }
        remaining -= 1;
