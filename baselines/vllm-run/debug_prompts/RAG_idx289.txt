        // Default Java Random() seed based on current time, mixed similarly
        let nanos = std::time::SystemTime::now()
            .duration_since(std::time::UNIX_EPOCH)
            .unwrap()
            .as_nanos() as u64;
        let seed = (nanos ^ 0x5DEECE66Du64) & ((1u64 << 48) - 1);
        Self { seed }
    }
    fn next(&mut self, bits: u32) -> i32 {
        self.seed = (self.seed.wrapping_mul(0x5DEECE66Du64).wrapping_add(0xBu64)) & ((1u64 << 48) - 1);
        (self.seed >> (48 - bits)) as i32
    }
    fn next_int_bound(&mut self, bound: i32) -> i32 {
        if bound <= 0 {
            panic!("bound must be positive");
        }
        if (bound & -bound) == bound {
            return ((bound as i64 * self.next(31) as i64) >> 31) as i32;
        }
        loop {
            let bits = self.next(31);
            let val = bits % bound;
            if bits - val + (bound - 1) >= 0 {
                return val;
            }
        }
    }
}

fn shuffle_array(a: &mut [i32]) {
    let mut rnd = JavaRandom::new();
    if a.len() <= 1 {

    for i in (1..a.len()).rev() {
        let j = rnd.next_int_bound((i + 1) as i32) as usize;
        a.swap(i, j);
    }
}
