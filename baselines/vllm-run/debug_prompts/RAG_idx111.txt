    fn next_i32(&mut self) -> i32 {

            self.idx += 1;

            self.idx += 1;
        }
        let mut num = 0i32;

            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
}

fn search(arr: &Vec<i32>, num: i32, low: usize, high: usize) -> usize {
    if high == low {
        return low;
    }
    let mid = (low + high) / 2;
    if arr[mid] == num {
        mid
    } else if arr[mid] < num {
        search(arr, num, mid + 1, high)
    } else {
        search(arr, num, low, mid)
    }
}

use std::io::{self, Read};

fn upper_bound(arr: &[i64], target: i64) -> i32 {
    let mut low: i32 = 0;
    let mut high: i32 = arr.len() as i32 - 1;
    while low <= high {
        let mid = (low + high) / 2;
        if arr[mid as usize] <= target {
            low = mid + 1;
        } else {
            high = mid - 1;
        }
    }
    high + 1
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: i64 = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };
    let m: i64 = it.next().unwrap().parse().unwrap();

    let nn = n as usize;
    let mut arr: Vec<i64> = Vec::with_capacity(nn);
    for _ in 0..nn {
        arr.push(it.next().unwrap().parse().unwrap());
    }

    arr.sort_unstable();

    let mut out = String::new();
    for _ in 0..(m as usize) {
        let t: i64 = it.next().unwrap().parse().unwrap();
        let ans = upper_bound(&arr, t);
        out.push_str(&ans.to_string());
