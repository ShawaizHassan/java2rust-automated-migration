fn pow_mod(mut a: i64, mut b: i64, m: i64) -> i64 {
    let mut res: i64 = 1;
    a %= m;
    while b > 0 {
        if (b & 1) == 1 {
            res = (res * a) % m;
        }
        a = (a * a) % m;
        b >>= 1;
    }
    res
}
