use std::io::{self, Read};

struct FastScanner {
    buf: Vec<u8>,
    idx: usize,
}
impl FastScanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { buf: input.into_bytes(), idx: 0 }
    }
    fn next_i64(&mut self) -> i64 {
        while self.idx < self.buf.len() && self.buf[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i64;
        if self.idx < self.buf.len() && self.buf[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i64;
        while self.idx < self.buf.len() && !self.buf[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.buf[self.idx] - b'0') as i64;
            self.idx += 1;
        }
        num * sign
    }
    fn next_usize(&mut self) -> usize {
        self.next_i64() as usize
    }
}

fn main() {
    let mut fs = FastScanner::new();
    let n = fs.next_usize();
    let d = fs.next_i64();

    let mut arr = vec![0i64; n];
    for i in 0..n {
        arr[i] = fs.next_i64();
    }

    let mut pre = vec![0i64; n];
    pre[0] = arr[0];
    if pre[0] > d {
        println!("-1");
        return;
    }
    for i in 1..n {
        pre[i] = pre[i - 1] + arr[i];
        if pre[i] > d {
            println!("-1");
            return;
        }
    }

    let mut max_pre = vec![0i64; n];
    max_pre[n - 1] = pre[n - 1];
    for i in (0..=(n - 2)).rev() {
        max_pre[i] = if max_pre[i + 1] > pre[i] { max_pre[i + 1] } else { pre[i] };
    }

    let mut cur_sum: i64 = 0;
    let mut count: i32 = 0;
    for i in 0..n {
        cur_sum += arr[i];
        if arr[i] == 0 && cur_sum < 0 {
            let flood = max_pre[i] - pre[i];
            let add = d - (cur_sum + flood);
            if flood > d {
                println!("-1");
                return;
            }
            cur_sum += add;
            count += 1;
        }
    }

    println!("{}", count);
}
