use std::io::{self, Read};

struct FastScanner {
    input: Vec<u8>,
    idx: usize,
}
impl FastScanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }
    fn next(&mut self) -> Option<String> {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        if self.idx >= self.input.len() {
            return None;
        }
        let start = self.idx;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        Some(String::from_utf8(self.input[start..self.idx].to_vec()).unwrap())
    }
    fn next_i32(&mut self) -> i32 {
        self.next().unwrap().parse::<i32>().unwrap()
    }
}

fn main() {
    let mut fs = FastScanner::new();
    let height = fs.next_i32() as usize;
    let width = fs.next_i32() as usize;

    let mut cs: Vec<i64> = vec![0; height];
    for r in 0..height {
        let s = fs.next().unwrap().into_bytes();
        for c in 0..width {
            if s[c] == b'#' {
                cs[r] |= 1i64 << c;
            }
        }
    }

                return;
            }
        }
    }
