impl Scanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }
    fn skip_whitespace(&mut self) {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
    }
    fn next_i32(&mut self) -> i32 {
        self.skip_whitespace();
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
    fn read_array_i32(&mut self, n: usize) -> Vec<i32> {
        let mut a = Vec::with_capacity(n);
        for _ in 0..n {
            a.push(self.next_i32());
        }
        a
    }
}

use std::io::{self, Read};

struct FastScanner {
    input: Vec<u8>,
    idx: usize,
}
impl FastScanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }
    fn skip_whitespace(&mut self) {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
    }
    fn next(&mut self) -> String {
        self.skip_whitespace();
        let start = self.idx;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        String::from_utf8(self.input[start..self.idx].to_vec()).unwrap()
    }
    fn next_i32(&mut self) -> i32 {
        self.next().parse::<i32>().unwrap()
    }
}

fn solve(in_sc: &mut FastScanner, out: &mut String) {
    let n = in_sc.next_i32() as usize;
    let s = in_sc.next();
    let ch: Vec<u8> = s.into_bytes();

    if n <= 2 {
        out.push_str("0\n");
        return;
    }

    let mut array: Vec<i32> = Vec::new();
    let mut prev = ch[0];
    let mut count: i32 = 1;
    for i in 1..n {
        if ch[i] == prev {
            count += 1;
        } else {
            array.push(count);
            count = 1;
            prev = ch[i];
        }
    }
    array.push(count);

    let mut extra: i32 = 0;
    let mut ssz = array.len();
    if ch[0] == ch[n - 1] && ssz != 1 {
        ssz -= 1;
        extra = array[ssz];
    }

    if ssz == 1 {
        let v = (array[0] - 1) / 3 + 1;
        out.push_str(&format!("{}\n", v));
        return;
    }

    let mut ans: i32 = 0;
    for i in 0..ssz {
        let mut x = array[i];
        if i == 0 {
            x += extra;
        }
        ans += x / 3;
    }
    out.push_str(&format!("{}\n", ans));
}

fn main() {
    let mut in_sc = FastScanner::new();
    let t = in_sc.next_i32();
    let mut out = String::new();
    for _ in 0..t {
        solve(&mut in_sc, &mut out);
    }

}
