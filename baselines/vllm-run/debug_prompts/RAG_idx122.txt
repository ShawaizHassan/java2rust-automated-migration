        print!("-1");
        return;

    let mut div: Vec<i64> = Vec::new();
    let mut h: i64 = 1;
    while (h as f64) <= (n as f64).sqrt() {
        if n % h == 0 {
            div.push(h);
            div.push(n / h);
        }
        h += 1;
    }

    let mut gcd: i64 = 1;

    let t1: i64 = if k % 2 == 0 { k / 2 } else { (k + 1) / 2 };
    let t2: i64 = if k % 2 == 0 { k + 1 } else { k };

    if n / t1 < t2 {
        print!("-1");
        return;
    }

    for &d in div.iter() {
        if d / t1 >= t2 {
            if n / d > gcd {
                gcd = n / d;
            }
        }
    }

    let mut sum: i64 = 0;

    let mut g: i64 = 1;
    while g < k {
        let i = g * gcd;
        out.push_str(&format!("{} ", i));
        sum += i;
        g += 1;

        Some(v) => v.parse().unwrap(),
        None => return,
    };
