use std::io::{self, Read};

struct FastScanner {
    input: Vec<u8>,
    idx: usize,
}
impl FastScanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }
    fn next_i32(&mut self) -> i32 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
    fn read_array_i32(&mut self, n: usize) -> Vec<i32> {
        let mut a = Vec::with_capacity(n);
        for _ in 0..n {
            a.push(self.next_i32());
        }
        a
    }
}

fn isvalid(ans: i32, arr: &[i32], lis: &[i32]) -> bool {
    for &ai in arr {
        let mut flag = false;
        for &lj in lis {
            let val: i64 = (ai & lj) as i64;
            if ((val | ans as i64) as i64) == ans as i64 {
                flag = true;
                break;
            }
        }
        if !flag {
            return false;
        }
    }
    true
}

fn main() {
    let mut sc = FastScanner::new();
    let n = sc.next_i32() as usize;
    let m = sc.next_i32() as usize;
    let arr = sc.read_array_i32(n);
    let lis = sc.read_array_i32(m);

    for ans in 0..(1 << 9) {
        if isvalid(ans, &arr, &lis) {
            println!("{}", ans);
            break;
        }
    }
}

use std::io::{self, Read};

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };
    let m_i32: i32 = it.next().unwrap().parse().unwrap();
    let m_usize: usize = m_i32 as usize;

    let mut arr: Vec<usize> = vec![0; n];
    let mut res = false;

    for i in 0..n {
        let mut v: i32 = it.next().unwrap().parse().unwrap();
        v %= m_i32;
        if v % m_i32 == 0 {
            res = true;
        }
        arr[i] = v as usize;
    }

    if n >= m_usize {
        res = true;
    }
    if res {
        print!("YES\n");
        return;
    }

    let mut can = vec![vec![false; m_usize + 1]; n + 1];

    for i in 1..=n {
        can[i][arr[i - 1]] = true;
        for j in 0..m_usize {
            if can[i - 1][j] {
                can[i][j] = true;
                let temp = (arr[i - 1] + j) % m_usize;
                can[i][temp] = true;
                if temp == 0 {
                    res = true;
                }
            }
        }
    }

    if res {
        print!("YES\n");
    } else {
        print!("NO\n");
