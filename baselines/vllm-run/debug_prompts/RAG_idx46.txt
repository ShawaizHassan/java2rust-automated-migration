use std::cmp::Ordering;
use std::io::{self, Read};

#[derive(Clone)]
struct Rect {
    parent: Option<usize>,
    w: i32,
    h: i32,
    i: i32,
    count: i32, // -1 means uncomputed
}

impl Rect {
    fn new(w: i32, h: i32, i: i32) -> Self {
        Self {
            parent: None,
            w,
            h,
            i,
            count: -1,
        }
    }
}

fn check(r0: &Rect, r: &Rect) -> bool {
    r0.h > r.h && r0.w > r.w
}

fn parents_str(idx: usize, rects: &Vec<Rect>) -> String {
    match rects[idx].parent {
        None => String::new(),
        Some(p) => format!("{} {}", rects[p].i, parents_str(p, rects)),
    }
}

fn count(idx: usize, rects: &mut Vec<Rect>) -> i32 {
    if rects[idx].count != -1 {
        return rects[idx].count;
    }
    let v = match rects[idx].parent {
        None => 0,
        Some(p) => count(p, rects) + 1,
    };
    rects[idx].count = v;
    v
}

fn best(idx: usize, card_idx: usize, n: usize, rects: &mut Vec<Rect>) -> i32 {
    if rects[idx].count != -1 {
        return rects[idx].count;
    }
    for i in (1..=n).rev() {
        let ok = {
            let r_i = &rects[i];
            let card = &rects[card_idx];
            let r = &rects[idx];
            check(r_i, card) && check(r_i, r)
        };
        if ok {
            let new_parent = i;
            let keep_old = match rects[idx].parent {
                None => false,
                Some(old_p) => {
                    let a = best(old_p, card_idx, n, rects);
                    let b = best(new_parent, card_idx, n, rects);
                    a > b
                }
            };
            if !keep_old {
                rects[idx].parent = Some(new_parent);
            }
        }
    }
    count(idx, rects)
}

struct Scanner {
    input: Vec<u8>,
    idx: usize,

impl Scanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self {
            input: input.into_bytes(),
            idx: 0,

    }
    fn next_i32(&mut self) -> i32 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {

            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
}
