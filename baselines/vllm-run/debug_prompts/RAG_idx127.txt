struct FastReader {
    input: Vec<u8>,
    idx: usize,
}

impl FastReader {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self {
            input: input.into_bytes(),
            idx: 0,
        }
    }

    fn next_token(&mut self) -> Option<String> {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        if self.idx >= self.input.len() {
            return None;
        }
        let start = self.idx;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        Some(String::from_utf8_lossy(&self.input[start..self.idx]).to_string())
    }

    fn next_i64(&mut self) -> i64 {
        self.next_token().unwrap().parse::<i64>().unwrap()
    }
}

fn main() {
    let mut reader = FastReader::new();

    let mut a: i64 = reader.next_i64();
    let b: i64 = reader.next_i64();

    let mut n: i32 = 0;
    while ((n as i64) * ((n as i64) + 1)) / 2 <= a + b {
        n += 1;
    }
    n -= 1;

    let nn = n as usize;
    let mut array = vec![false; nn];

    let mut p: i32 = 0;

    for i in (0..nn).rev() {
        let val = (i + 1) as i64;
        if a >= val {
            a -= val;
            array[i] = true;
            p += 1;
        }
