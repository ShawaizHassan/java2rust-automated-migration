use std::io::{self, Read};

const MOD: i64 = 1_000_000_007;

fn mod_pow(mut a: i64, mut e: i64) -> i64 {
    let mut r: i64 = 1;
    a %= MOD;
    while e > 0 {
        if (e & 1) == 1 {
            r = (r * a) % MOD;
        }
        a = (a * a) % MOD;
        e >>= 1;
    }
    r
}

fn egcd(a: i64, b: i64) -> (i64, i64, i64) {
    if b == 0 {
        (a, 1, 0)
    } else {
        let (g, x, y) = egcd(b, a % b);
        (g, y, x - (a / b) * y)
    }
}

fn mod_inv(a: i64) -> i64 {
    let (g, x, _) = egcd(a, MOD);
    if g != 1 && g != -1 {
        0
    } else {
        ((x % MOD) + MOD) % MOD
    }
}

fn floor_div_nonneg_big_by_i32(divisor: i64, digits: &[u8]) -> i64 {
    let mut rem: i64 = 0;
    let mut q_mod: i64 = 0;
    for &d in digits {
        rem = rem * 10 + d as i64;
        let qd = rem / divisor;
        rem %= divisor;
        q_mod = (q_mod * 10 + (qd % MOD)) % MOD;
    }
    q_mod
}
