        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }
    fn next_i32(&mut self) -> i32 {
        self.skip_whitespace();
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.input.len() {
            let c = self.input[self.idx];
            if c.is_ascii_digit() {
                num = num * 10 + (c - b'0') as i32;
                self.idx += 1;
            } else {
                break;
            }
        }
        num * sign
    }
    fn next_string(&mut self) -> Vec<u8> {
        self.skip_whitespace();
        let start = self.idx;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        self.input[start..self.idx].to_vec()
    }
    fn skip_whitespace(&mut self) {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
    }
}

use std::fs::File;
use std::io::{Read, Write};

fn bg(mut b: i32, mut g: i32, cont: i32) -> String {
    let mut out = String::new();
    for _ in 0..cont {
        if b > 0 {
            out.push('B');
            b -= 1;
        }
        if g > 0 {
            out.push('G');
            g -= 1;
        }
    }
    out
}

fn gb(mut b: i32, mut g: i32, cont: i32) -> String {
    let mut out = String::new();
    for _ in 0..cont {
        if g > 0 {
            out.push('G');
            g -= 1;
        }
        if b > 0 {
            out.push('B');
            b -= 1;
        }
    }
    out
}

fn output(str_in: &str) {
    let parts: Vec<&str> = str_in.split(' ').collect();
    let b: i32 = parts[0].parse().unwrap();
    let g: i32 = parts[1].parse().unwrap();

    let cont: i32 = if b > g { b } else { g };

    let ans = if b >= g { bg(b, g, cont) } else { gb(b, g, cont) };

    let mut out = File::create("output.txt").unwrap();
    out.write_all(ans.as_bytes()).unwrap();
    out.flush().unwrap();
}

fn main() {
    let mut input = String::new();
    if let Ok(mut f) = File::open("input.txt") {
        let _ = f.read_to_string(&mut input);
    }

    let mut last_line = String::new();
    for line in input.lines() {
        last_line = line.to_string();
    }

    output(&last_line);
}
