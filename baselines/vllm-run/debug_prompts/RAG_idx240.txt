use std::io::{self, Read};

struct Scanner {
    input: Vec<u8>,
    idx: usize,
}

impl Scanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }

    fn next_token(&mut self) -> Option<Vec<u8>> {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        if self.idx >= self.input.len() {
            return None;
        }
        let start = self.idx;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        Some(self.input[start..self.idx].to_vec())
    }

    fn next_i32(&mut self) -> i32 {
        let tok = self.next_token().unwrap();
        let s = std::str::from_utf8(&tok).unwrap();
        s.parse::<i32>().unwrap()
    }

    fn next_i64(&mut self) -> i64 {
        let tok = self.next_token().unwrap();
        let s = std::str::from_utf8(&tok).unwrap();
        s.parse::<i64>().unwrap()
    }
}

fn main() {
    let mut sc = Scanner::new();
    let t = sc.next_i32();

    let mut out = String::new();
    for _ in 0..t {
        let x1 = sc.next_i64();
        let y1 = sc.next_i64();
        let x2 = sc.next_i64();
        let y2 = sc.next_i64();

        let x = x2 - x1;

use std::io::{self, Read};

struct Scanner {
    input: Vec<u8>,
    idx: usize,
}
impl Scanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }
    fn next_i32(&mut self) -> i32 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
}

fn main() {
    let mut sc = Scanner::new();
    let t = sc.next_i32();
    let mut out = String::new();

    for _ in 0..(t as usize) {
        let n = sc.next_i32();
        let k = sc.next_i32();

        let nk = (n as i64) * (k as i64);
        let nk_usize = nk as usize;

        let mut arr: Vec<i32> = vec![0; nk_usize];
        for i in 0..nk_usize {
            arr[i] = sc.next_i32();
        }

        let mid = n / 2;
        let sub = mid + 1;

        let mut sum: i64 = 0;
        let mut count: i32 = 0;

        let mut i: i64 = (n as i64) * (k as i64) - (sub as i64);
        while i >= 0 {
            if count < k {
                sum += arr[i as usize] as i64;
            } else {
                break;
            }
            count += 1;
            i -= sub as i64;
        }

        out.push_str(&format!("{}\n", sum));
    }

    print!("{}", out);
}
