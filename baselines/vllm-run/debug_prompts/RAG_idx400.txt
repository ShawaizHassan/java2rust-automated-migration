use std::io::{self, Read};

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = match it.next() {
        Some(v) => v.parse::<usize>().unwrap(),
        None => return,
    };

    let mut d = vec![vec![0i64; n + 1]; n + 1];
    for i in 1..=n {
        for j in 1..=n {
            d[i][j] = it.next().unwrap().parse::<i64>().unwrap();
        }
    }

    let mut remove = vec![0usize; n + 1];
    for i in 1..=n {
        remove[i] = it.next().unwrap().parse::<usize>().unwrap();
    }

    let mut ans = vec![0i64; n];

    for rep in (1..=n).rev() {
        let cr = remove[rep];

        for i in 1..=n {
            let dicr = d[i][cr];
            for j in 1..=n {
                let cd = dicr + d[cr][j];
                if d[i][j] > cd {
                    d[i][j] = cd;
                }
            }
        }

        let mut sum: i64 = 0;
        for i in rep..=n {
            let ri = remove[i];
            for j in rep..=n {
                let rj = remove[j];
                sum += d[ri][rj];
            }
        }
        ans[rep - 1] = sum;
    }

    let mut out = String::new();
    for i in 0..n {
        out.push_str(&ans[i].to_string());
        out.push(' ');
    }
    out.push('\n');
    print!("{}", out);
