    let n: usize = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };

    let mut dis: Vec<i32> = Vec::with_capacity(n);
    for _ in 0..n {
        dis.push(it.next().unwrap().parse().unwrap());
    }

    let mut s: usize = it.next().unwrap().parse::<usize>().unwrap();
    let mut t: usize = it.next().unwrap().parse::<usize>().unwrap();

    if t < s {
        std::mem::swap(&mut s, &mut t);
    }

    let mut len1: i32 = 0;
    let mut len2: i32 = 0;

    for i in (s - 1)..(t - 1) {
        len1 += dis[i];
    }
    for i in (t - 1)..n {
        len2 += dis[i];
    }
    for i in 0..(s - 1) {
        len2 += dis[i];
