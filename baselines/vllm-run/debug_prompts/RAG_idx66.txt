        PrintWriter out = new PrintWriter(System.out);

        int n = in.nextInt();
        String number = in.next();

        char[] r = number.substring(n).toCharArray();

        boolean gt = true, lt = true;

impl FastScanner {

        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }
    fn next_i32(&mut self) -> i32 {

            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
}

fn check(a: &[i32], n: usize) -> bool {
    let mut l: i32 = -1;
    let mut r: i32 = n as i32;

    for i in 0..n {
        if a[i] < i as i32 {
            break;
        }
        l = i as i32;
    }

    for j in (0..n).rev() {
        if a[j] < (n - j - 1) as i32 {
            break;
        }
        r = j as i32;
    }
