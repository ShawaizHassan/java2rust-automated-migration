            out.println(1);

            out.println(2);

            long ans = 0;
            int lower = (int) (n <= 100 ? 1 : n - 100);
            for (int i = lower; i <= n; i++) {
                for (int j = lower; j <= n; j++) {
                    for (int k = lower; k <= n; k++) {

        return x * y / gcd(x, y);
    }

        io::stdin().read_to_end(&mut input).unwrap();
        Self { input, idx: 0 }
    }
    fn next_i32(&mut self) -> i32 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
}
