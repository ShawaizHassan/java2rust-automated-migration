fn dp(ind: i32, sum: i32, taken: bool) -> bool {
    unsafe {
        if taken && sum % M == 0 {
            return true;
        }
        let p = Pair { x: ind, y: sum };
        if let Some(mp) = MP.as_ref() {

        res
    }
}

fn binary_search(al: &Vec<Pair>, key: i32, mut low: i32, mut high: i32) -> i32 {
    while low <= high {
        let mid = ((low + high) as u32 >> 1) as i32; // emulate Java >>> 1 on int
        let p = &al[mid as usize];
        if p.x == key {
            return p.y;
        } else if p.x > key {

            low = mid + 1;
        }
    }

}
