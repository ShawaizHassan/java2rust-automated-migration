use std::cmp::Ordering;
use std::collections::BinaryHeap;
use std::io::{self, Read};

#[derive(Clone)]
struct Node {
    idx: usize,
    rating: i32,
}

impl Eq for Node {}

impl PartialEq for Node {
    fn eq(&self, other: &Self) -> bool {
        self.rating == other.rating && self.idx == other.idx
    }
}

impl Ord for Node {
    fn cmp(&self, other: &Self) -> Ordering {
        match self.rating.cmp(&other.rating) {
            Ordering::Equal => other.idx.cmp(&self.idx),
            ord => ord,
        }
    }
}

impl PartialOrd for Node {
    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
        Some(self.cmp(other))
    }
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };

    let mut ratings: Vec<i32> = vec![0; n];
    for i in 0..n {
        ratings[i] = it.next().unwrap().parse().unwrap();
    }

    let mut heap: BinaryHeap<Node> = BinaryHeap::with_capacity(n);
    for i in 0..n {
        heap.push(Node {
            idx: i,
            rating: ratings[i],
        });
    }

    let mut ranks: Vec<i32> = vec![0; n];
    let mut old_rating: i32 = -1;
    let mut current_rank: i32 = -1;

    for i in 0..n {
        let node = heap.pop().unwrap();
        if node.rating == old_rating {
            ranks[node.idx] = current_rank;
        } else {
            let r = (i as i32) + 1;
            ranks[node.idx] = r;
            current_rank = r;
            old_rating = node.rating;
        }
    }

    let mut out = String::new();
    for i in 0..n {
        if i != 0 {
            out.push(' ');
        }
        out.push_str(&ranks[i].to_string());
    }
    print!("{}", out);
}
