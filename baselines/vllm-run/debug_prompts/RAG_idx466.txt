use std::io::{self, Read};

struct FastScanner {
    input: Vec<u8>,
    idx: usize,
}
impl FastScanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }
    fn next(&mut self) -> String {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let start = self.idx;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        String::from_utf8(self.input[start..self.idx].to_vec()).unwrap()
    }
}

fn solve(fs: &mut FastScanner) -> i64 {
    let s1 = fs.next();
    let s2 = fs.next();
    let b1 = s1.as_bytes();
    let b2 = s2.as_bytes();
    let n = b1.len();
    let m = b2.len();

    let mut ans: i64 = 0;

    let lim = if n < m { n } else { m };
    for i in 0..lim {
        if b1[i] != b2[i] {
            break;
        }
        let k = i + 1;
        if m % k != 0 {
            continue;
        }
        if n % k != 0 {
            continue;
        }
        let mut ok = true;

        for j in k..n {
            if b1[j] != b1[j - k] {
                ok = false;
                break;
            }
        }
        if ok {
            for j in k..m {
                if b2[j] != b2[j - k] {
                    ok = false;
                    break;
                }
            }
        }

        if ok {
            ans += 1;
        }
    }

    ans
}

fn main() {
    let mut fs = FastScanner::new();
    let ans = solve(&mut fs);
    print!("{}", ans);
}
