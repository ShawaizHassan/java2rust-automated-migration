fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = match it.next() {
        Some(x) => x.parse().unwrap(),
        None => return,
    };
    let k: usize = it.next().unwrap().parse().unwrap();

    let mut edges: Vec<Vec<usize>> = vec![Vec::new(); n];
    for _ in 0..(n.saturating_sub(1)) {
        let u: usize = it.next().unwrap().parse::<usize>().unwrap() - 1;
        let v: usize = it.next().unwrap().parse::<usize>().unwrap() - 1;
        edges[u].push(v);
        edges[v].push(u);
    }

    let mut cities: Vec<City> = vec![
        City {
            dist: i32::MAX,
            children: 0
        };
        n
    ];

    // Iterative DFS to compute parent, depth, and traversal order
    let mut parent: Vec<usize> = vec![usize::MAX; n];
    let mut order: Vec<usize> = Vec::with_capacity(n);

    let mut stack: Vec<usize> = Vec::new();
    cities[0].dist = 0;
    parent[0] = 0;
    stack.push(0);

    while let Some(u) = stack.pop() {
        order.push(u);
        let du = cities[u].dist;
        for &v in &edges[u] {
            if cities[v].dist == i32::MAX {
                cities[v].dist = du + 1;
                parent[v] = u;
                stack.push(v);
            }
        }
    }

    // Post-order accumulation to compute subtree sizes and children counts
    let mut sub_size: Vec<i32> = vec![1; n];
    for &u in order.iter().rev() {
        if u == 0 {
            continue;
        }
        let p = parent[u];
        sub_size[p] += sub_size[u];
    }
    for i in 0..n {
        cities[i].children = sub_size[i] - 1;
    }

    // Sort by (dist - children) descending
    cities.sort_by(|a, b| {
        let va = a.dist - a.children;
        let vb = b.dist - b.children;
        vb.cmp(&va)
    });

    let mut ans: i64 = 0;
    let kk = k.min(n);
    for i in 0..kk {
        ans += (cities[i].dist - cities[i].children) as i64;
    }

    print!("{ans}\n");
}
