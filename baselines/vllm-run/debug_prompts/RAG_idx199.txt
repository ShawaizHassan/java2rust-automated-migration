    let mut it = input.split_whitespace();

    let s = match it.next() {
        Some(v) => v.to_string(),

    };

        Some(v) => v.to_string(),
        None => return,

    let mut mv = vec![vec![MAXV; N]; N];
    for i in 0..N {
        mv[i][i] = 0;
    }

    for _ in 0..n {
        let a_str = it.next().unwrap();
        let b_str = it.next().unwrap();
        let w = it.next().unwrap().parse::<i32>().unwrap();
        let a = (a_str.as_bytes()[0] - b'a') as usize;
        let b = (b_str.as_bytes()[0] - b'a') as usize;
        if w < mv[a][b] {
            mv[a][b] = w;
        }
    }

    if s.len() != t.len() {
        print!("-1\n");
        return;
    }

    for k in 0..N {
        for i in 0..N {
            for j in 0..N {
                let cand = mv[i][k].saturating_add(mv[k][j]);
                if cand < mv[i][j] {
                    mv[i][j] = cand;
                }
            }
        }
    }

    let sb = s.as_bytes();
    let tb = t.as_bytes();
    let mut cost: i32 = 0;
    let mut ans = String::with_capacity(sb.len());

        let p = (sb[idx] - b'a') as usize;
        let q = (tb[idx] - b'a') as usize;

        let mut tmp = MAXV;
        let mut tmpc = b'a';

        for c in 0..N {
            let val = mv[p][c].saturating_add(mv[q][c]);
            if val < tmp {
                tmp = val;
                tmpc = (b'a' + c as u8);
            }

        if cost < MAXV {
            cost = cost.saturating_add(tmp);
            ans.push(tmpc as char);
        } else {
            print!("-1\n");
            return;
        }
