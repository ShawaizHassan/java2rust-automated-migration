        Self { input, idx: 0 }
    }
    fn next_i32(&mut self) -> i32 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;

            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
}

fn ncr(n: i32, k: i32) -> i64 {
    let mut res: i64 = 1;
    for i in 1..=k {
        let numerator = (n - k + i) as i64;
        res = res * numerator / (i as i64);
    }
    res
}
