use std::io::{self, Read};

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };

    fn cat(s: &str) -> i32 {
        match s {
            "rat" => 1,
            "child" => 2,
            "woman" => 2,
            "man" => 3,
            "captain" => 4,
            _ => 0,
        }
    }

    let mut v: Vec<(i32, usize, String)> = Vec::with_capacity(n);
    for idx in 0..n {
        let name = it.next().unwrap().to_string();
        let role = it.next().unwrap();
        v.push((cat(role), idx, name));
    }

    v.sort_by_key(|&(c, idx, _)| (c, idx));

    let mut out = String::new();
    for (_, _, name) in v {
        out.push_str(&name);
        out.push('\n');
    }
    print!("{}", out);
}
