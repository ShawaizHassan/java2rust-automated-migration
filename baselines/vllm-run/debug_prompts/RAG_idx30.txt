struct Scanner {

        io::stdin().read_to_string(&mut input).unwrap();
        Self {
            input: input.into_bytes(),
            idx: 0,
        }
    }

    fn next_i32(&mut self) -> i32 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
}

    let mut it = input.split_whitespace();

    let n: i32 = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };
    let _k: i32 = it.next().unwrap().parse().unwrap();

    let mut map: HashMap<i32, i32> = HashMap::new();

        let drink_choice: i32 = it.next().unwrap().parse().unwrap();
        let entry = map.entry(drink_choice).or_insert(0);
        *entry += 1;
    }

    let drink_sets: i32 = n / 2 + (n % 2);

    let mut pairs: i32 = 0;
    let mut remainders: i32 = 0;

        pairs += *v / 2;
        if *v % 2 != 0 {
