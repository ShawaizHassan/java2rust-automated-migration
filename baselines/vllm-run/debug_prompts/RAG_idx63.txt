use std::io::{self, Read};

struct Scanner {
    input: Vec<u8>,
    idx: usize,
}

impl Scanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }

    fn next(&mut self) -> String {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let start = self.idx;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        String::from_utf8(self.input[start..self.idx].to_vec()).unwrap()
    }
}

fn main() {
    let mut sc = Scanner::new();
    let a1 = sc.next().into_bytes();
    let a2 = sc.next().into_bytes();

    let mut uno: i32 = 0;
    for &c in &a1 {
        if c == b'1' {
            uno += 1;
        }
    }

    let mut dos: i32 = 0;
    for &c in &a2 {
        if c == b'1' {
            dos += 1;
        }
    }

    if uno % 2 == 1 {
        uno += 1;
    }

    if uno >= dos {

    }
}
