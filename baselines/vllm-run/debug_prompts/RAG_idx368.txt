fn merge(a: &mut [i32], temp: &mut [i32], left: usize, mid: usize, right: usize) -> i64 {

    let mut i = left;
    let mut j = mid;
    let mut k = left;

    while i <= mid - 1 && j <= right {
        if a[i] <= a[j] {
            temp[k] = a[i];
            i += 1;
        } else {
            temp[k] = a[j];
            j += 1;
            cnt += (mid - i) as i64;
        }
        k += 1;
    }

    while i <= mid - 1 {
        temp[k] = a[i];
        i += 1;
        k += 1;
    }

    while j <= right {
        temp[k] = a[j];
        j += 1;
        k += 1;
    }

    for idx in left..=right {
        a[idx] = temp[idx];

    cnt
}

fn mergesort(a: &mut [i32], temp: &mut [i32], left: usize, right: usize) -> i64 {
    if right > left {
        let mid = (right + left) / 2;
        let mut cnt = mergesort(a, temp, left, mid);
        cnt += mergesort(a, temp, mid + 1, right);
        cnt += merge(a, temp, left, mid + 1, right);
        cnt
    } else {
        0
    }
}
