use std::io::{self, Read};

struct Scanner {
    input: Vec<u8>,
    idx: usize,
}
impl Scanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }
    fn next_i32(&mut self) -> i32 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
    fn next_arr_i32(&mut self, n: usize) -> Vec<i32> {
        let mut v = Vec::with_capacity(n);
        for _ in 0..n {
            v.push(self.next_i32());
        }
        v
    }
}

fn solve(out: &mut String, sc: &mut Scanner) {
    let n = sc.next_i32() as usize;
    let nums = sc.next_arr_i32(n);
    for i in 0..nums.len().saturating_sub(1) {
        if nums[i] <= nums[i + 1] {
            out.push_str("YES\n");
            return;
        }
    }
    out.push_str("NO\n");
}

fn main() {
    let mut sc = Scanner::new();
    let t = sc.next_i32();
    let mut out = String::new();
    for _ in 0..t {
        solve(&mut out, &mut sc);
    }
    print!("{}", out);
}
