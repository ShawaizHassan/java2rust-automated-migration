use std::io::{self, Read};

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();
    let n: i32 = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };
    let k: i32 = it.next().unwrap().parse().unwrap();

    let n_usize = n as usize;
    let mut matrix = vec![vec![false; n_usize]; n_usize];

    if (n as i64) * (n as i64) < k as i64 {
        print!("-1\n");
        return;
    }

    let mut tofill: i32 = k;
    let mut i: usize = 0;
    let mut j: usize = 0;

    while tofill > 0 {
        if tofill == 1 {
            tofill = 0;
            if !matrix[i][i] {
                matrix[i][i] = true;
            } else {
                matrix[i + 1][i + 1] = true;
            }
        } else if matrix[i][j] {
            j += 1;
            if j == n_usize {
                i += 1;
                j = 0;
            }
        } else if i == j {
            matrix[i][j] = true;
            tofill -= 1;
            j += 1;
        } else {
            matrix[i][j] = true;
            matrix[j][i] = true;
            tofill -= 2;
            j += 1;
            if j == n_usize {
                i += 1;
                j = 0;
            }
        }
    }

    let mut out = String::new();
    for i in 0..n_usize {
        for j in 0..n_usize {
            out.push_str(if matrix[i][j] { "1 " } else { "0 " });
        }
        out.push('\n');
    }
    print!("{out}");
}
