struct FastScanner {
    input: Vec<u8>,
    idx: usize,
}

impl FastScanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }

    fn next_i32(&mut self) -> i32 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
}

fn main() {
    let mut fs = FastScanner::new();
    let mut t = fs.next_i32();
    let mut out = String::new();

    while t > 0 {
        t -= 1;
        let n = fs.next_i32() as f64;
        let angle = (PI * (2.0 * n - 2.0)) / (4.0 * n);
        let v = angle.tan();
        out.push_str(&format!("{}\n", v));
    }

    print!("{out}");
}

    input: Vec<u8>,
    idx: usize,
}

impl FastScanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }

    fn next_i32(&mut self) -> i32 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
}

fn main() {
    let mut fs = FastScanner::new();

    let n = fs.next_i32() as usize;
    let k = fs.next_i32();

    let mut sum: i64 = 0;
    for _ in 0..n {
        let a = fs.next_i32();
        let b = fs.next_i32();
        sum += (a - b).abs() as i64 + 1;
    }

    for i in 0..=k {
        if (sum + i as i64) % k as i64 == 0 {
            println!("{}", i);
            return;
        }
    }
