    let n: i64 = match it.next() {

        None => return,

    let mut k: i64 = it.next().unwrap().parse().unwrap();

    let mut a: Vec<i32> = Vec::new();
    let binary_string = format!("{:b}", n);
    let len = binary_string.len() as i32;

    for (idx, ch) in binary_string.chars().enumerate() {
        if ch == '1' {
            a.push(len - 1 - idx as i32);
            k -= 1;
        }
    }

    if k < 0 {
        print!("No");
    } else {
        while k != 0 {
            let highest = a[0];
            let mut num_highest: usize = 1;
            while num_highest < a.len() && a[num_highest] == highest {
                num_highest += 1;
            }

            if (num_highest as i64) <= k {
                let rest: Vec<i32> = a[num_highest..].to_vec();
                let mut to_add: Vec<i32> = Vec::with_capacity(num_highest * 2 + rest.len());
                for _ in 0..num_highest {
                    to_add.push(highest - 1);
                    to_add.push(highest - 1);
                }
                to_add.extend_from_slice(&rest);
                a = to_add;
                k -= num_highest as i64;

                    let min_val = a.pop().unwrap();
                    a.push(min_val - 1);
                    a.push(min_val - 1);

            }
