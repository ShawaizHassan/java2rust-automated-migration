impl FastScanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }
    fn next_i32(&mut self) -> i32 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;

            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }

        io::stdin().read_to_string(&mut input).unwrap();

    fn next_i32(&mut self) -> i32 {

            self.idx += 1;

        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
}

fn main() {
    let mut sc = Scanner::new();
    let n = sc.next_i32() as usize;
    let m = sc.next_i32() as usize;

    let mut c: Vec<i32> = vec![0; n];
    for i in 0..n {
        c[i] = sc.next_i32();
    }

    let mut al: Vec<i32> = Vec::with_capacity(m);
    for _ in 0..m {
        let x = sc.next_i32();
        al.push(x);
    }

    let mut i: usize = 0;
    let mut count: i32 = 0;

    while !al.is_empty() && i < n {
        if count > 0 {
            i += 1;
        }
        let b = al[0];
        // remove first occurrence of value b (like al.remove(new Integer(b)) in Java)
        if let Some(pos) = al.iter().position(|&v| v == b) {
            al.remove(pos);
        }

        while i < n {
            if b >= c[i] {
                count += 1;
                break;
            }
            i += 1;
        }
    }

    let mut out = String::new();
    out.push_str(&format!("{}\n", count));
    let mut w = io::BufWriter::new(io::stdout());
    w.write_all(out.as_bytes()).unwrap();
