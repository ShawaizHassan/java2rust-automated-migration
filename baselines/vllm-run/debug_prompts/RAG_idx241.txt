struct Scanner {
    input: Vec<u8>,
    idx: usize,
}
impl Scanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }
    fn next(&mut self) -> String {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let start = self.idx;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        String::from_utf8(self.input[start..self.idx].to_vec()).unwrap()
    }
    fn next_i32(&mut self) -> i32 {
        self.next().parse::<i32>().unwrap()
    }
}

use std::io::{self, Read};

fn is_palindrome(buf: &[char]) -> bool {
    let mut i: usize = 0;
    let mut j: usize = buf.len().saturating_sub(1);
    while i < j {
        if buf[i] != buf[j] {
            return false;
        }
        i += 1;
        j -= 1;
    }
    true
}

fn check(buf: &[char], original: &[char]) -> bool {
    buf != original && is_palindrome(buf)
}

fn main() {
