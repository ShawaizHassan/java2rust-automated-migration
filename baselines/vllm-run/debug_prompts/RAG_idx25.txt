struct InputReader {
    buf: Vec<u8>,
    idx: usize,
}

impl InputReader {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { buf: input.into_bytes(), idx: 0 }
    }

    fn is_whitespace(c: u8) -> bool {
        c <= 32
    }

    fn next_i32(&mut self) -> i32 {
        while self.idx < self.buf.len() && Self::is_whitespace(self.buf[self.idx]) {
            self.idx += 1;
        }
        let mut sgn: i32 = 1;
        if self.idx < self.buf.len() && self.buf[self.idx] == b'-' {
            sgn = -1;
            self.idx += 1;
        }
        let mut ret: i32 = 0;
        while self.idx < self.buf.len() && !Self::is_whitespace(self.buf[self.idx]) {
            let c = self.buf[self.idx];
            if c < b'0' || c > b'9' {
                panic!("Invalid character");
            }
            ret = ret * 10 + (c - b'0') as i32;
            self.idx += 1;
        }
        ret * sgn
    }
}

use std::io::{self, Read};

struct FastScanner {
    data: Vec<u8>,
    idx: usize,
}

impl FastScanner {
    fn new(mut input: Box<dyn Read>) -> Self {
        let mut data = Vec::new();
        input.read_to_end(&mut data).unwrap();
        Self { data, idx: 0 }
    }

    fn next_int(&mut self) -> i32 {
        let mut negative = false;
        loop {
            if self.idx >= self.data.len() {
                panic!("EOF");
            }
            let b = self.data[self.idx];
            self.idx += 1;
            if b == b'-' {
                negative = true;
                continue;
            }
            if b >= b'0' && b <= b'9' {
                let mut r: i32 = (b - b'0') as i32;
                while self.idx < self.data.len() {
                    let d = self.data[self.idx];
                    if d >= b'0' && d <= b'9' {
                        self.idx += 1;
                        r = r * 10 + (d - b'0') as i32;
                    } else {
                        self.idx += 1;
                        break;
                    }
                }
                return if negative { -r } else { r };
            }
        }
    }
}
