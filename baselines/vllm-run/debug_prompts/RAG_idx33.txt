struct Scanner {

impl Scanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }
    fn next(&mut self) -> String {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let start = self.idx;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        String::from_utf8(self.input[start..self.idx].to_vec()).unwrap()
    }
}

use std::io::{self, Read};

struct LineReader {
    parts: Vec<String>,
    idx: usize,
}

impl LineReader {
    fn new(input: String) -> Self {
        let mut parts: Vec<String> = input
            .split('\n')
            .map(|s| {
                let s = if s.ends_with('\r') { &s[..s.len() - 1] } else { s };
                s.to_string()
            })
            .collect();

        // If input ends with '\n', split will include a final empty segment already.
        // If not, behavior matches typical line reading as well.
        LineReader { parts, idx: 0 }
    }

    fn next_line(&mut self) -> String {
        if self.idx >= self.parts.len() {
            return String::new();
        }
        let s = self.parts[self.idx].clone();
        self.idx += 1;
        s
    }
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut lr = LineReader::new(input);

    let first_line = lr.next_line();
    let n: usize = first_line
        .split_whitespace()
        .next()
        .unwrap_or("0")
        .parse::<usize>()
        .unwrap_or(0);

    let mut x: Vec<String> = Vec::with_capacity(n);
    for _ in 0..n {
