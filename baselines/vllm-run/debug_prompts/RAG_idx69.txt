            }
        }

use std::io::{self, Read};

struct Scanner {
    input: Vec<u8>,
    idx: usize,
}

impl Scanner {
    fn new() -> Self {
        let mut input = Vec::new();
        io::stdin().read_to_end(&mut input).unwrap();
        Self { input, idx: 0 }
    }

    fn next_i32(&mut self) -> i32 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
}

fn main() {
    let mut sc = Scanner::new();
    let n = sc.next_i32() as usize;
    let mut cnt = vec![0i32; n + 1];

    for _ in 0..(n.saturating_sub(1)) {
        let x = sc.next_i32() as usize;
        if x <= n {
            cnt[x] += 1;
        }
    }

    let mut out = String::new();
    for i in 1..=n {
        if cnt[i] == 0 {
            out.push_str(&format!("{}\n", i));
            break;
        }
    }
