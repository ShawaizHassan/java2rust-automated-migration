struct Scanner {
    input: Vec<u8>,
    idx: usize,
}
impl Scanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }
    fn next_i32(&mut self) -> i32 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
}

fn solve(scan: &mut Scanner, out: &mut String) {
    let mut answ: i64 = 0;
    let size = scan.next_i32() as usize;
    let nums = scan.next_i32() as usize;

    let total = size * nums;
    let mut arr: Vec<i32> = Vec::with_capacity(total);
    for _ in 0..total {
        arr.push(scan.next_i32());
    }

    let mut ind: usize = ((size + 1) / 2) - 1;
    let skip: usize = size - ind - 1;
    ind *= nums;

    let step = skip + 1;
    let mut i = ind;
    while i < total {
        answ += arr[i] as i64;
        i += step;
    }

    out.push_str(&format!("{}\n", answ));
}

fn main() {
    let mut scan = Scanner::new();
    let tests = scan.next_i32();
    let mut out = String::new();
    for _ in 0..tests {
        solve(&mut scan, &mut out);
    }
    print!("{}", out);
}

            self.idx += 1;
