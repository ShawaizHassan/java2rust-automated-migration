struct FastScanner<'a> {
    bytes: &'a [u8],
    idx: usize,
}

impl<'a> FastScanner<'a> {
    fn new(s: &'a str) -> Self {
        Self {
            bytes: s.as_bytes(),
            idx: 0,
        }
    }

    fn skip_whitespace(&mut self) {
        while self.idx < self.bytes.len() && self.bytes[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
