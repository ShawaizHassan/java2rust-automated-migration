use std::io::{self, Read};

struct Scanner {
    input: Vec<u8>,
    idx: usize,
}
impl Scanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }
    fn next<T: std::str::FromStr>(&mut self) -> T {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let start = self.idx;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        std::str::from_utf8(&self.input[start..self.idx])
            .unwrap()
            .parse::<T>()
            .ok()
            .unwrap()
    }
}

use std::io::{self, Read};

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let t: i32 = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };

    let mut out = String::new();

    for _ in 0..t {
        let n: usize = it.next().unwrap().parse::<i32>().unwrap() as usize;
        let x: i32 = it.next().unwrap().parse().unwrap();

        let mut a: Vec<i32> = vec![0; n];
        let mut b: Vec<i32> = vec![0; n];

        for i in 0..n {
            a[i] = it.next().unwrap().parse().unwrap();
        }
        for i in 0..n {
            b[i] = it.next().unwrap().parse().unwrap();
        }

        a.sort();
        b.sort_by(|u, v| v.cmp(u)); // descending

        let mut cnt: usize = 0;
        for i in 0..n {
            if a[i] + b[i] <= x {
                cnt += 1;
            }
        }

        if cnt == n {
            out.push_str("Yes\n");
        } else {
            out.push_str("No\n");
        }
    }

    print!("{}", out);
}
