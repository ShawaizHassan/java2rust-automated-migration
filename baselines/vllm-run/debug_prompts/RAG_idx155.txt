#[derive(Clone, Debug, PartialEq, Eq)]
struct BigInt {
    sign: i8,      // 0 for zero, 1 for positive
    digits: Vec<u32>, // base 1e9, little-endian
}

impl BigInt {
    const BASE: u32 = 1_000_000_000;

    fn zero() -> Self {
        BigInt { sign: 0, digits: Vec::new() }
    }

    fn normalize(&mut self) {
        while let Some(&last) = self.digits.last() {
            if last == 0 {
                self.digits.pop();
            } else {
                break;
            }
        }
        if self.digits.is_empty() {
            self.sign = 0;
        } else {
            self.sign = 1;
        }
    }

    fn is_zero(&self) -> bool {
        self.sign == 0
    }

    fn div_small(&mut self, d: u32) {
        if self.is_zero() {
            return;
        }
        let mut rem: u64 = 0;
        for i in (0..self.digits.len()).rev() {
            let cur = rem * (Self::BASE as u64) + self.digits[i] as u64;
            self.digits[i] = (cur / d as u64) as u32;
            rem = cur % d as u64;
        }
        self.normalize();
    }

    fn from_decimal_str(s: &str) -> Self {
        let s = s.trim();
        if s.is_empty() {
            return BigInt::zero();
        }
        let s = if s.starts_with('+') { &s[1..] } else { s };
        let s = s.trim_start_matches('0');
        if s.is_empty() {
            return BigInt::zero();
        }

        let mut digits = Vec::new();
        let mut i = s.len();
        while i > 0 {
            let start = if i >= 9 { i - 9 } else { 0 };
            let chunk = &s[start..i];
            digits.push(chunk.parse::<u32>().unwrap());
            i = start;
        }
        let mut bi = BigInt { sign: 1, digits };
        bi.normalize();
        bi
    }
}

    fn next<T: FromStr>(&mut self) -> T {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let start = self.idx;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        std::str::from_utf8(&self.input[start..self.idx])
            .unwrap()
            .parse::<T>()
            .ok()
            .unwrap()
    }

        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {

        }
        let start = self.idx;

            self.idx += 1;
        }
        String::from_utf8(self.input[start..self.idx].to_vec()).unwrap()
    }
}

use std::io::{self, Read};

const MOD: i64 = 1_000_000_007;

fn mod_pow(mut a: i64, mut e: i64) -> i64 {
    let mut r: i64 = 1;
    a %= MOD;
    while e > 0 {
        if (e & 1) == 1 {
            r = (r * a) % MOD;
        }
        a = (a * a) % MOD;
        e >>= 1;
    }
    r
}

fn egcd(a: i64, b: i64) -> (i64, i64, i64) {
    if b == 0 {
        (a, 1, 0)
    } else {
        let (g, x, y) = egcd(b, a % b);
        (g, y, x - (a / b) * y)
    }
}

fn mod_inv(a: i64) -> i64 {
    let (g, x, _) = egcd(a, MOD);
    if g != 1 && g != -1 {
        0
    } else {
        ((x % MOD) + MOD) % MOD
    }
}

fn floor_div_nonneg_big_by_i32(divisor: i64, digits: &[u8]) -> i64 {
    let mut rem: i64 = 0;
    let mut q_mod: i64 = 0;
    for &d in digits {
        rem = rem * 10 + d as i64;
        let qd = rem / divisor;
        rem %= divisor;
        q_mod = (q_mod * 10 + (qd % MOD)) % MOD;
    }
    q_mod
}

fn main() {

    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();
    let n: i32 = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };
    let k: i32 = it.next().unwrap().parse().unwrap();
    let m: usize = (n / k) as usize;

    let mut divs: Vec<i64> = vec![0; m];
    let mut digs: Vec<i32> = vec![0; m];
    for i in 0..m {
        divs[i] = it.next().unwrap().parse::<i32>().unwrap() as i64;
    }
    for i in 0..m {
        digs[i] = it.next().unwrap().parse::<i32>().unwrap();
    }

    let mut ans: i64 = 1;
    let ten_pow_k = mod_pow(10, k as i64);
    let hb_mod = (ten_pow_k - 1 + MOD) % MOD; // 10^k - 1

    for i in 0..m {
        let dv = divs[i];
        let inv_dv = mod_inv(dv % MOD);

        let dv_i64 = dv;

        // hb = 10^k - 1
        let hb_div_mod = (hb_mod * inv_dv) % MOD;

        // hbd = dig*10^(k-1) + (10^(k-1)-1)
        let ten_pow_km1 = mod_pow(10, (k as i64) - 1);
        let hbd_mod = (((digs[i] as i64 % MOD) * ten_pow_km1) % MOD + (ten_pow_km1 - 1 + MOD) % MOD) % MOD;

        // hbd_div_mod = hbd / dv (as BigInteger floor division) mod MOD
        // We'll compute exact quotient mod MOD via digit long division.
        let mut hbd_digits: Vec<u8> = Vec::with_capacity(k as usize);
        hbd_digits.push(digs[i] as u8);
        for _ in 0..(k - 1) {
            hbd_digits.push(9);
        }
        let hbd_div_mod_exact = floor_div_nonneg_big_by_i32(dv_i64, &hbd_digits);

            let factor = (hb_div_mod - hbd_div_mod_exact + MOD) % MOD;
            ans = (ans * factor) % MOD;

            ans = (ans * factor) % MOD;
        }
