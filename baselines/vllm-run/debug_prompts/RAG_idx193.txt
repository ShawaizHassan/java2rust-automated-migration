use std::io::{self, Read};

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    // First token is read and ignored
    let _ = it.next();

    let mut out = String::new();
    loop {
        let n_opt = it.next();
        if n_opt.is_none() {
            break;
        }
        let k_opt = it.next();
        if k_opt.is_none() {
            break;
        }

        let n: i64 = n_opt.unwrap().parse().unwrap();
        let k: i64 = k_opt.unwrap().parse().unwrap();

        let m = std::cmp::min(n - 1, k);
        let ans = (m * (m + 1) / 2) + if k >= n { 1 } else { 0 };
        out.push_str(&ans.to_string());
        out.push('\n');
    }

    print!("{}", out);
