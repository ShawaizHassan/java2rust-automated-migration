use std::collections::HashMap;
use std::io::{self, Read};

struct Scanner {
    input: Vec<u8>,
    idx: usize,
}
impl Scanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self {
            input: input.into_bytes(),
            idx: 0,
        }
    }
    fn next(&mut self) -> Option<String> {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        if self.idx >= self.input.len() {
            return None;
        }
        let start = self.idx;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        Some(String::from_utf8(self.input[start..self.idx].to_vec()).unwrap())
    }
    fn next_i32(&mut self) -> i32 {
        self.next().unwrap().parse::<i32>().unwrap()
    }
}

fn main() {
    let mut sc = Scanner::new();
    let n = sc.next_i32() as usize;

    let mut g: Vec<Vec<String>> = vec![Vec::new(), Vec::new(), Vec::new(), Vec::new()];

    let mut map: HashMap<String, i32> = HashMap::new();
    map.insert("rat".to_string(), 0);
    map.insert("woman".to_string(), 1);
    map.insert("child".to_string(), 1);
    map.insert("man".to_string(), 2);
    map.insert("captain".to_string(), 3);
