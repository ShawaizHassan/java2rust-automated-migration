fn main() {
    // Read entire input (matches Java nextLine() behavior by taking first line)
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();

    let mut s = input;
    if s.ends_with('\n') {
        s.pop();
        if s.ends_with('\r') {
            s.pop();
        }
    }

    let mut out = String::new();

    out.push_str("Mike\n");

    let bytes = s.as_bytes();
    if !bytes.is_empty() {
        let mut min_ch: u8 = bytes[0];
        let len: usize = bytes.len();
        for i in 1..len {
            let mut flag = true;
            let c = bytes[i];
            if c <= min_ch {
                flag = false;
                min_ch = c;
            }
            if flag {
                out.push_str("Ann\n");
            } else {
                out.push_str("Mike\n");
            }
        }
    }

    let mut stdout = io::BufWriter::new(io::stdout());
    stdout.write_all(out.as_bytes()).unwrap();

use std::io::{self, Read, Write};

struct FastScanner {
    input: Vec<u8>,
    idx: usize,
}
impl FastScanner {
    fn new() -> Self {
        let mut input = Vec::new();
        io::stdin().read_to_end(&mut input).unwrap();
        Self { input, idx: 0 }
    }
    fn next_i32(&mut self) -> i32 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
}

fn main() {
    let mut fs = FastScanner::new();

    let n = fs.next_i32() as usize;
    let mut relations = vec![0i32; n + 1];

    for i in 0..n {
        let p = fs.next_i32();
        let c = fs.next_i32();
        if c == 0 {
            relations[i + 1] = 1;
            if p != -1 {
                let pi = p as usize;
                if pi < relations.len() {
                    relations[pi] = 1;
                }
            }
        }
    }

    let mut out = String::new();
    let mut flag = false;
    for i in 1..=n {
        if relations[i] == 0 {
            flag = true;

        }

}
