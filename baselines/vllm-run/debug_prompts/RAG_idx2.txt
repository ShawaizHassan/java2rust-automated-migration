fn main() {

    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: i32 = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };
    let m: i32 = it.next().unwrap().parse().unwrap();

    let n_usize = n as usize;
    let mut points: Vec<Vec<i32>> = vec![Vec::new(); n_usize];

    for _ in 0..m {
        let s: i32 = it.next().unwrap().parse().unwrap();
        let e: i32 = it.next().unwrap().parse().unwrap();

        let s_idx = ((s % n) + n) % n;
        let e_idx = ((e % n) + n) % n;

        let d1 = (n + e - s) % n;
        let d2 = (n + s - e) % n;

        points[s_idx as usize].push(d1);
        points[e_idx as usize].push(d2);
    }

    for v in points.iter_mut() {
        v.sort();
    }

    // Unique prime factors of n
    let mut factors: Vec<i32> = Vec::new();
    let mut n_copy = n;
    let mut i: i32 = 2;
    while (i as i64) * (i as i64) <= n_copy as i64 {
        if n_copy % i == 0 {
            factors.push(i);
            while n_copy % i == 0 {
                n_copy /= i;
            }
        }
        i += 1;
    }
    if n_copy != 1 {
        factors.push(n_copy);
    }

    for &f in factors.iter() {
        let dist = n / f;
        let dist_usize = dist as usize;

        let mut possible = true;

fn hp(list: &Vec<i32>) -> i32 {
    if list.is_empty() {
        return 0;
    }
    let n = list.len();
    let mut dp = vec![0i32; n];
    dp[0] = 1;
    for i in 1..n {
        let mut val = 1i32;
        for j in 0..i {
            if ((list[i] - list[j]) % 2).abs() == 1 {
                val = val.max(1 + dp[j]);
            }
        }
        dp[i] = val;
    }
    dp[n - 1]
}
