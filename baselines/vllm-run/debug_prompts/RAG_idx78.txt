use std::io::{self, Read};

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    if input.trim().is_empty() {
        return;
    }

    let mut lines = input.lines();

    let first = match lines.next() {
        Some(v) => v,
        None => return,
    };
    let mut count: i32 = match first.trim().parse() {
        Ok(v) => v,
        Err(_) => return,
    };

    let mut rats: Vec<String> = Vec::new();
    let mut woman_and_child: Vec<String> = Vec::new();
    let mut man: Vec<String> = Vec::new();
    let mut captain: String = String::new();

    while count > 0 {
        if let Some(line) = lines.next() {
            let mut it = line.split_whitespace();
            let name = it.next().unwrap_or("").to_string();
            let role = it.next().unwrap_or("");

            match role {
                "rat" => rats.push(name),
                "woman" | "child" => woman_and_child.push(name),
                "man" => man.push(name),
                "captain" => captain = name,
                _ => {}
            }
        } else {
            break;
        }
        count -= 1;
    }

    let mut out = String::new();
    for item in rats {
        out.push_str(&item);
        out.push('\n');
    }
    for item in woman_and_child {
        out.push_str(&item);
        out.push('\n');
    }
    for item in man {
        out.push_str(&item);
        out.push('\n');
    }
    out.push_str(&captain);
    out.push('\n');

    print!("{out}");
}
