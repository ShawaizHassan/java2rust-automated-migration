use std::io::{self, Read};

fn cross(a: i32, s: i32, d: i32, f: i32) -> i32 {
    a * f - s * d
}

struct Scanner {
    input: Vec<u8>,
    idx: usize,
}
impl Scanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }
    fn next_i32(&mut self) -> i32 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
}

fn main() {
    let mut scan = Scanner::new();
    let n = scan.next_i32() as usize;
    let mut x = vec![0i32; n];
    let mut y = vec![0i32; n];

    for q in 0..n {
        x[q] = scan.next_i32();
        y[q] = scan.next_i32();
    }

    let mut max_val: i32 = 0;
    for q in 0..n {
        for w in (q + 1)..n {
            let mut a: i32 = -1;
            let mut s: i32 = -1;

            for e in 0..n {
                if e != q && e != w {
                    let d = cross(
                        x[w] - x[q],
                        y[w] - y[q],
                        x[e] - x[q],
                        y[e] - y[q],
                    );
                    if d >= 0 {
                        if a < d {
                            a = d;
                        }
                    }
                    if d <= 0 {
                        if s < -d {
                            s = -d;
                        }
                    }
                }
            }

            if a + s > max_val && a >= 0 && s >= 0 {
                max_val = a + s;
            }
        }
    }

    if max_val % 2 == 0 {
        println!("{}.000000", max_val / 2);
