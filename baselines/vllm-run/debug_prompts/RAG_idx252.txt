use std::io::{self, Read};

fn is_square(n: i64) -> i64 {
    let sq = (n as f64).sqrt();
    let s = sq as i64;
    if s * s == n { s } else { -1 }
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: usize = match it.next() {
        Some(v) => v.parse::<i32>().unwrap() as usize,
        None => return,
    };

    let mut pluses = vec![0i64; n];
    let mut on_screen: i64 = 2;
    let mut k: i32 = 2;
    pluses[0] = 2;
    let mut nextsq: i64 = 2;

    while k < (n as i32) + 1 {
        let kk = k as i64;
        let kp1 = (k + 1) as i64;
        pluses[(k - 1) as usize] = kp1 * kp1 * kk - (kk - 1);

        k += 1;
        on_screen = nextsq;
        let _sq = is_square(k as i64);
        nextsq = (k as i64) * ((k + 1) as i64);
    }

    let mut out = String::new();
    for i in 0..n {
        out.push_str(&(pluses[i].abs().to_string()));
        out.push('\n');
    }
    print!("{}", out);
}

use std::io::{self, Read};

struct FastScanner {
    input: Vec<u8>,
    idx: usize,
}
impl FastScanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }
    fn next_i32(&mut self) -> i32 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
}

fn is_square(n: i64) -> i64 {
    let sq = (n as f64).sqrt();
    let s = sq as i64;
    if s * s == n { s } else { -1 }
}

fn main() {
    let mut fs = FastScanner::new();
    let n = fs.next_i32() as i64;

    let mut pluses: Vec<i64> = vec![0; n as usize];
    let mut on_screen: i64 = 2;
    let mut k: i32 = 2;
    if n >= 1 {
        pluses[0] = 2;

        pluses[(k - 1) as usize] = nextsq * ((k + 1) as i64) - ((k - 1) as i64);

        out.push_str(&val.to_string());
        out.push('\n');
    }
    print!("{}", out);
}
