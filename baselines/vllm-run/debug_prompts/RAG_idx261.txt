struct InputReader {
    buf: Vec<u8>,
    idx: usize,
}

impl InputReader {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { buf: input.into_bytes(), idx: 0 }
    }

    fn is_whitespace(c: u8) -> bool {
        c <= 32
    }

    fn next_i32(&mut self) -> i32 {
        while self.idx < self.buf.len() && Self::is_whitespace(self.buf[self.idx]) {
            self.idx += 1;
        }
        let mut sgn: i32 = 1;
        if self.idx < self.buf.len() && self.buf[self.idx] == b'-' {
            sgn = -1;
            self.idx += 1;
        }
        let mut ret: i32 = 0;
        while self.idx < self.buf.len() && !Self::is_whitespace(self.buf[self.idx]) {
            let c = self.buf[self.idx];
            if c < b'0' || c > b'9' {
                panic!("Invalid character");
            }
            ret = ret * 10 + (c - b'0') as i32;
            self.idx += 1;
        }
        ret * sgn
    }
}

    let mut input = InputReader::new();
    let n = input.next_i32() as i64;
    let mut sum: i64 = 0;
    for _ in 0..(n - 1) {

    }
    let ans = n * (n + 1) / 2 - sum;
