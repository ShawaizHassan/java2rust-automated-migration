fn power(mut x: i64, mut y: i64, p: i64) -> i64 {
    let mut res: i64 = 1;
    x %= p;
    while y > 0 {
        if (y & 1) == 1 {
            res = (res * x) % p;
        }
        y >>= 1;
        x = (x * x) % p;
    }
    res
}
