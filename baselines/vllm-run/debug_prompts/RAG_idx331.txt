use std::collections::HashMap;
use std::io::{self, Read};

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    input = input.replace("\r\n", "\n").replace('\r', "\n");
    let mut parts: Vec<&str> = input.split('\n').collect();
    let mut idx: usize = 0;

    fn next_line<'a>(parts: &'a [&'a str], idx: &mut usize) -> &'a str {
        if *idx >= parts.len() {
            ""
        } else {
            let s = parts[*idx];
            *idx += 1;
            s
        }
    }

    let t_line = next_line(&parts, &mut idx);
    let t: i32 = t_line.trim().parse().unwrap_or(0);

    let mut out = String::new();

    for _ in 0..t {
        let n_line = next_line(&parts, &mut idx);
        let n: i32 = n_line.trim().parse().unwrap();

        let mut freqs: HashMap<char, i32> = HashMap::new();

        for _ in 0..n {
            let s = next_line(&parts, &mut idx);
            for c in s.chars() {
                *freqs.entry(c).or_insert(0) += 1;
            }
        }

        let mut res = true;
        for &count in freqs.values() {
            if count % n != 0 {
                res = false;
                break;
            }
        }

        out.push_str(if res { "YES\n" } else { "NO\n" });
    }

    print!("{out}");
}

    let mut two: i32 = 0;
