use std::io::{self, Read};

fn is_all_lowercase_letters(s: &str) -> bool {
    s.bytes().all(|b| b >= b'a' && b <= b'z')
}

fn replace_token(tok: &str) -> String {
    if is_all_lowercase_letters(tok) {
        if tok.ends_with("lios") {
            return "_A".to_string();
        }
        if tok.ends_with("liala") {
            return "_a".to_string();
        }
        if tok.ends_with("etr") {
            return "_N".to_string();
        }
        if tok.ends_with("etra") {
            return "_n".to_string();
        }
        if tok.ends_with("initis") {
            return "_V".to_string();
        }
        if tok.ends_with("inites") {
            return "_v".to_string();
        }
    }
    tok.to_string()
}

fn has_two_consecutive_letters(s: &str) -> bool {
    let mut prev_is_letter = false;
    for b in s.bytes() {
        let is_letter = (b >= b'a' && b <= b'z') || (b >= b'A' && b <= b'Z');
        if prev_is_letter && is_letter {
            return true;
        }
        prev_is_letter = is_letter;
    }
    false
}

fn matches_female(cat: &str) -> bool {
    let bytes = cat.as_bytes();
    let mut i = 0usize;
    while i < bytes.len() && bytes[i] == b'a' {
        i += 1;
    }
    if i >= bytes.len() || bytes[i] != b'n' {
        return false;
    }
    i += 1;
    while i < bytes.len() && bytes[i] == b'v' {
        i += 1;
    }
    i == bytes.len()
}

fn matches_male(cat: &str) -> bool {

        return false;
    }
    i += 1;
    while i < bytes.len() && bytes[i] == b'V' {
        i += 1;
    }
    i == bytes.len()
}

            System.out.println("impossible");

        println!("impossible");

        for ch in word.chars() {
            set.insert(ch);
        }
        let distinct = set.len() as i32;
        if distinct >= k {
            println!("0");
        } else {
            println!("{}", k - distinct);
        }
    }
}
