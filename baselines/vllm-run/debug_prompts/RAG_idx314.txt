struct FastScanner {
    input: Vec<u8>,
    idx: usize,
}
impl FastScanner {
    fn new() -> Self {
        let mut input = Vec::new();
        io::stdin().read_to_end(&mut input).unwrap();
        Self { input, idx: 0 }
    }
    fn skip_whitespace(&mut self) {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
    }
    fn next(&mut self) -> String {
        self.skip_whitespace();
        let start = self.idx;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        String::from_utf8(self.input[start..self.idx].to_vec()).unwrap()
    }
    fn next_i32(&mut self) -> i32 {
        self.next().parse::<i32>().unwrap()
    }
}

    let mut sc = FastScanner::new();
    let t = sc.next_i32() as usize;
    let mut out = String::new();

                let left = if i > 0 { s[i - 1] } else { 0 };
                let right = if i + 1 < n { s[i + 1] } else { 0 };

                let mut chosen = b'a';
