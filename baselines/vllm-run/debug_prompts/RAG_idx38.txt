use std::io::{self, Read};

fn int_of_alphabet(ch: u8) -> i32 {
    // "-ABCDEFGHIJKLMNOPQRSTUVWXYZ"
    if ch >= b'A' && ch <= b'Z' {
        (ch - b'A' + 1) as i32
    } else {
        0
    }
}

fn char_of_integer(i: i32) -> u8 {
    // 1..=26 -> 'A'..'Z'
    (b'A' + (i as u8) - 1) as u8
}

fn is_rc_format(s: &str) -> bool {
    // matches "R[0-9]+C[0-9]+"
    let b = s.as_bytes();
    if b.len() < 4 || b[0] != b'R' {
        return false;
    }
    let mut i = 1usize;
    if i >= b.len() || !b[i].is_ascii_digit() {
        return false;
    }
    while i < b.len() && b[i].is_ascii_digit() {
        i += 1;
    }
    if i >= b.len() || b[i] != b'C' {
        return false;
    }
    i += 1;
    if i >= b.len() || !b[i].is_ascii_digit() {
        return false;
    }
    while i < b.len() && b[i].is_ascii_digit() {
        i += 1;
    }
    i == b.len()
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: i32 = match it.next() {
        Some(x) => x.parse().unwrap(),
        None => return,
    };

    let mut out = String::new();

    for _ in 0..n {
        let strv = it.next().unwrap();

        if is_rc_format(strv) {
            let bytes = strv.as_bytes();
            let c_pos = bytes.iter().position(|&ch| ch == b'C').unwrap();

            let excel_row = &strv[1..c_pos];
            let mut excel_column: i32 = strv[c_pos + 1..].parse().unwrap();

            let mut result_bytes: Vec<u8> = Vec::new();

            loop {
                if excel_column <= 26 {
                    result_bytes.push(char_of_integer(excel_column));
                    break;
                } else if excel_column % 26 == 0 {
                    result_bytes.push(char_of_integer(26));
                    excel_column -= 1;
                    excel_column /= 26;
                } else {
                    result_bytes.push(char_of_integer(excel_column % 26));
                    excel_column /= 26;
                }
            }

            result_bytes.reverse();
            let mut result = String::from_utf8(result_bytes).unwrap();
            result.push_str(excel_row);

            out.push_str(&result);
            out.push('\n');
        } else {
            // openColum = str.split("[0-9]+")[0]; => prefix letters until first digit
            let b = strv.as_bytes();
            let mut k = 0usize;
            while k < b.len() && !b[k].is_ascii_digit() {
                k += 1;
            }
            let open_colum = &strv[..k];
            let open_row = &strv[k..];

            let col_bytes = open_colum.as_bytes();
            let mut column: i32 = 0;

            // reversed, sum pow(26,i) * intOfAlphabet(ch)
            for (i, &ch) in col_bytes.iter().rev().enumerate() {
                let pow = 26_i32.pow(i as u32);
                column += pow * int_of_alphabet(ch);
            }

            out.push('R');
            out.push_str(open_row);
            out.push('C');
            out.push_str(&column.to_string());
            out.push('\n');
        }
    }

    print!("{out}");
}

        Some(v) => v.parse().unwrap(),
