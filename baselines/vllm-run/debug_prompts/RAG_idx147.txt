use std::io::{self, Read};

fn cal(x: i64, d: i64) -> i64 {
    x + (d / (x + 1)) + if d % (x + 1) != 0 { 1 } else { 0 }
}

fn bs(mut l: i64, mut r: i64, d: i64) -> i64 {
    let mut res = cal(l, d);
    while l <= r {
        let mid = (l + r) / 2;
        let v = cal(mid, d);
        if v < res {
            res = v;
            l = mid + 1;
        } else {
            r = mid - 1;
        }
    }
    res
}
