struct FastScanner {
    input: Vec<u8>,
    idx: usize,
}

impl FastScanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self {
            input: input.into_bytes(),
            idx: 0,
        }
    }

    fn next_i64(&mut self) -> i64 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i64;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i64;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.input[self.idx] - b'0') as i64;
            self.idx += 1;
        }
        num * sign
    }

    fn next_i32(&mut self) -> i32 {
        self.next_i64() as i32
    }
}

fn check(a: i64, b: i64, l: i64, r: i64, t: i64, m: i64) -> bool {
    if a + b * r > t {
        return false;
    }

    let len = r - l + 1;
    let sum: i128 = (a as i128) * (len as i128)
        + ((b as i128) * (len as i128) * ((r + l) as i128)) / 2;

    sum <= (t as i128) * (m as i128)
}
