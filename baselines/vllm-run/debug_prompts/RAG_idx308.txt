struct FastScanner {
    input: Vec<u8>,
    idx: usize,
}
impl FastScanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self {
            input: input.into_bytes(),
            idx: 0,
        }
    }
    fn skip_whitespace(&mut self) {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
    }
    fn next_i32(&mut self) -> i32 {
        self.skip_whitespace();
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
}

    let mut sc = FastScanner::new();

    let mut freq = vec![0i64; 100005];

use std::io::{self, Read};

struct FastScanner {
    input: Vec<u8>,
    idx: usize,
}

impl FastScanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }

    fn next(&mut self) -> String {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let start = self.idx;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        String::from_utf8(self.input[start..self.idx].to_vec()).unwrap()
    }

    fn next_i32(&mut self) -> i32 {
        self.next().parse::<i32>().unwrap()
    }
}

fn solve(in_sc: &mut FastScanner, out: &mut String) {
    let n: i32 = in_sc.next_i32();
    let nf = n as f64;

    let val = nf * nf - 4.0 * nf;

    if val < 0.0 {
        out.push_str("N\n");
        return;
    }

    let root = val.sqrt();
    let a = (nf + root) / 2.0;
    let b = (nf - root) / 2.0;

    out.push_str("Y ");
    out.push_str(&format!("{} {}", a, b));
    out.push('\n');
}

fn main() {
    let mut in_sc = FastScanner::new();
    let t = in_sc.next_i32();
    let mut out = String::new();
    for _ in 0..t {
        solve(&mut in_sc, &mut out);
    }
    print!("{}", out);
}
