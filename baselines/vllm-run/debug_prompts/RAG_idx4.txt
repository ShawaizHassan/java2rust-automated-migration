        let cc: i32 = b * aa + bb;

            out.push_str("Waste\n");
        } else {
            let lo = match cod.binary_search_by(|x| {
                if *x <= cc {
                    Ordering::Less

                    Ordering::Greater
                }
            }) {
                Ok(pos) => pos + 1,
                Err(pos) => pos,
            };
            let idx = ((cc - lo as i32) % 3) as usize;
            out.push_str(res[idx]);
            out.push('\n');
        }
    }

use std::io::{self, Read};

#[derive(Clone, Debug)]
struct BigInt {
    sign: i8,        // 0 for zero, 1 for positive, -1 for negative
    digits: Vec<u32>, // base 1e9, little-endian
}

impl BigInt {
    const BASE: u32 = 1_000_000_000;

    fn zero() -> Self {
        BigInt {
            sign: 0,
            digits: Vec::new(),
        }
    }

    fn from_i64(x: i64) -> Self {
        if x == 0 {
            return Self::zero();
        }
        let mut v = if x < 0 { -x } else { x } as u64;
        let mut digits = Vec::new();
        while v > 0 {
            digits.push((v % (Self::BASE as u64)) as u32);
            v /= Self::BASE as u64;
        }
        BigInt {
            sign: if x < 0 { -1 } else { 1 },
            digits,
        }
    }

    fn normalize(&mut self) {
        while let Some(&last) = self.digits.last() {
            if last == 0 {
                self.digits.pop();
            } else {
                break;
            }
        }
        if self.digits.is_empty() {
            self.sign = 0;
        }
    }

    fn cmp_abs(&self, other: &Self) -> i32 {
        if self.digits.len() != other.digits.len() {
            return if self.digits.len() < other.digits.len() { -1 } else { 1 };
        }
        for i in (0..self.digits.len()).rev() {
            if self.digits[i] != other.digits[i] {
                return if self.digits[i] < other.digits[i] { -1 } else { 1 };
            }
        }
        0
    }

    fn compare_to(&self, other: &Self) -> i32 {
        if self.sign != other.sign {
            return if self.sign < other.sign { -1 } else { 1 };
        }
        match self.sign {
            0 => 0,
            1 => self.cmp_abs(other),
            -1 => -self.cmp_abs(other),
            _ => 0,
        }
    }

    fn mul_small(&self, m: u32) -> Self {
        if self.sign == 0 || m == 0 {
            return Self::zero();
        }
        let mut res = Vec::with_capacity(self.digits.len() + 1);
        let mut carry: u64 = 0;
        for &d in &self.digits {
            let prod = (d as u64) * (m as u64) + carry;
            res.push((prod % (Self::BASE as u64)) as u32);
            carry = prod / (Self::BASE as u64);
        }
        if carry > 0 {
            res.push(carry as u32);
        }
        BigInt { sign: self.sign, digits: res }
    }

    fn mul(&self, other: &Self) -> Self {
        if self.sign == 0 || other.sign == 0 {
            return Self::zero();
        }
        let n = self.digits.len();
        let m = other.digits.len();
        let mut res = vec![0u64; n + m];
        for i in 0..n {
            let a = self.digits[i] as u64;
            let mut carry = 0u64;
            for j in 0..m {
                let idx = i + j;
                let cur = res[idx] + a * (other.digits[j] as u64) + carry;
                res[idx] = cur % (Self::BASE as u64);
                carry = cur / (Self::BASE as u64);
            }
            let mut idx = i + m;
            while carry > 0 {
                let cur = res[idx] + carry;
                res[idx] = cur % (Self::BASE as u64);
                carry = cur / (Self::BASE as u64);
                idx += 1;
                if idx >= res.len() && carry > 0 {
                    res.push(0);
                }
            }
        }
        let mut digits: Vec<u32> = res.into_iter().map(|x| x as u32).collect();
        let mut out = BigInt {
            sign: self.sign * other.sign,
            digits,
        };
        out.normalize();
        out
    }

    fn pow(&self, exp: i32) -> Self {
        if exp == 0 {
            return BigInt::from_i64(1);
        }
        if self.sign == 0 {
            return Self::zero();
        }
        let mut e = exp as u32;
        let mut base = self.clone();
        let mut result = BigInt::from_i64(1);
        while e > 0 {
            if (e & 1) == 1 {
                result = result.mul(&base);
            }
            e >>= 1;
            if e > 0 {
                base = base.mul(&base);
            }
        }
        result
    }

    fn to_string(&self) -> String {
        if self.sign == 0 {
            return "0".to_string();
        }
        let mut s = String::new();
        if self.sign < 0 {
            s.push('-');
        }

            s.push_str(&first.to_string());
            for &d in it {
                s.push_str(&format!("{:09}", d));
            }
        }
        s
    }
}
