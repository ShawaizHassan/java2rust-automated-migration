struct FastScanner {
    input: Vec<u8>,
    idx: usize,
}
impl FastScanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }
    fn next_i32(&mut self) -> i32 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
}

fn main() {
    let mut sc = FastScanner::new();
    let t = sc.next_i32();
    let mut out = String::new();

    for _ in 0..t {
        let n = sc.next_i32() as usize;
        let k = sc.next_i32() as usize;

        let mut set: BTreeSet<i32> = BTreeSet::new();
        for _ in 0..n {
            let v = sc.next_i32();
            set.insert(v);
        }

        if set.len() > k {
            out.push_str("-1\n");
            continue;
        }

        let mut list: Vec<i32> = set.into_iter().collect();
        if list.len() < k {
            let d = k - list.len();
            for _ in 0..d {

            }
        }

        out.push('\n');

                out.push(' ');
            }
        }

    input: Vec<u8>,
    idx: usize,
}

impl FastScanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self {
            input: input.into_bytes(),
            idx: 0,
        }
    }

    fn next_i32(&mut self) -> i32 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
}

fn main() {
    let mut fs = FastScanner::new();
    let mut t = fs.next_i32();
    let mut out = String::new();

    while t > 0 {
        t -= 1;
        let n_i32 = fs.next_i32();
        let n = n_i32 as usize;

        let mut arr: Vec<i32> = Vec::with_capacity(n);
        for _ in 0..n {
            arr.push(fs.next_i32());
        }
        arr.sort();

        let mut list: Vec<i32> = Vec::with_capacity(n);
        for i in 0..=n / 2 {
            if i == n / 2 {
                list.push(arr[i]);
            } else {
                list.push(arr[i]);
                list.push(arr[n - i - 1]);
            }

            out.push_str(&list[i].to_string());
            out.push(' ');
        }
        out.push('\n');
    }
