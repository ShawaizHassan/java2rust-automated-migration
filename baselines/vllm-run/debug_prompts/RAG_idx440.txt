fn binary(a: &Vec<i32>, k: i32) -> i32 {
    if k < a[0] {
        return 0;
    }
    if k >= a[a.len() - 1] {
        return a.len() as i32;
    }
    let mut l: i32 = 0;
    let mut r: i32 = a.len() as i32 - 1;
    let mut ans: i32 = 0;
    while l <= r {
        let m = (l + r) / 2;
        match k.cmp(&a[m as usize]) {
            Ordering::Greater | Ordering::Equal => {
                ans = m + 1;
                l = m + 1;
            }
            Ordering::Less => {
                r = m - 1;
            }
        }
    }
    ans
}
