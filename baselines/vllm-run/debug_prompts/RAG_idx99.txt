    }

use std::collections::HashSet;
use std::io::{self, Read};

struct Scanner {
    input: Vec<u8>,
    idx: usize,
}
impl Scanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }
    fn next_i32(&mut self) -> i32 {
        self.skip_whitespace();
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.input.len() {
            let c = self.input[self.idx];
            if c.is_ascii_digit() {
                num = num * 10 + (c - b'0') as i32;
                self.idx += 1;
            } else {
                break;
            }
        }
        sign * num
    }
    fn next_string(&mut self) -> String {
        self.skip_whitespace();
        let start = self.idx;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        String::from_utf8(self.input[start..self.idx].to_vec()).unwrap()
    }
    fn skip_whitespace(&mut self) {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
    }
}

fn main() {
    let mut sc = Scanner::new();
    let n = sc.next_i32();
    let mut set: HashSet<u32> = HashSet::new();

    for _ in 0..n {
        let s = sc.next_string();
        let mut mask: u32 = 0;
        for b in s.bytes() {
            if (b'a'..=b'z').contains(&b) {
                let bit = (b - b'a') as u32;
                mask |= 1u32 << bit;
            }
        }
        set.insert(mask);
    }

    println!("{}", set.len());
}
