use std::io::{self, Read};

struct Scanner {
    input: Vec<u8>,
    idx: usize,
}

impl Scanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }

    fn next(&mut self) -> Option<String> {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        if self.idx >= self.input.len() {
            return None;
        }
        let start = self.idx;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        Some(String::from_utf8_lossy(&self.input[start..self.idx]).to_string())
    }

    fn next_i32(&mut self) -> i32 {
        self.next().unwrap().parse::<i32>().unwrap()
    }

    fn next_i64(&mut self) -> i64 {
        self.next().unwrap().parse::<i64>().unwrap()
    }
}

fn main() {
    let mut sc = Scanner::new();
    let t = sc.next_i32();
    let mut out = String::new();

    for _ in 0..t {
        let mut n = sc.next_i64();
        let mut count: i32 = 0;

        while n % 2 == 0 || n % 3 == 0 || n % 5 == 0 {
            if n % 2 == 0 {
                n /= 2;
            } else if n % 3 == 0 {
                n = 2 * (n / 3);
            } else {
                n = 4 * (n / 5);
            }
            count += 1;
        }

        if n == 1 {
            out.push_str(&format!("{}\n", count));
        } else {
            out.push_str("-1\n");
        }
    }

    print!("{}", out);
}
