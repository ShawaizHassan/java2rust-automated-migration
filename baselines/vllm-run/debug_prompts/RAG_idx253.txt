struct FastScanner {
    input: Vec<u8>,
    idx: usize,
}

impl FastScanner {
    fn new() -> Self {

        io::stdin().read_to_string(&mut input).unwrap();
        Self {
            input: input.into_bytes(),
            idx: 0,
        }
    }

    fn next_i32(&mut self) -> i32 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
}

    let mut fs = FastScanner::new();
    let n = fs.next_i32() as usize;
    let k = fs.next_i32();

    let mut out = String::new();

        for j in 0..n {

use std::io::{self, Read};

struct FastScanner {
    input: Vec<u8>,
    idx: usize,
}

impl FastScanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }

    fn next_token(&mut self) -> String {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let start = self.idx;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        String::from_utf8(self.input[start..self.idx].to_vec()).unwrap()
    }

    fn next_i32(&mut self) -> i32 {
        self.next_token().parse::<i32>().unwrap()
    }
}

fn main() {
    let mut sc = FastScanner::new();
    let n = sc.next_i32() as usize;
    let mut mixture: Vec<i32> = vec![0; n];
    let mut sum: i32 = 0;

    for i in 0..n {
        mixture[i] = sc.next_i32();
    }
    for i in 0..n {
        sum += mixture[i];
    }

    let avg = (sum as f64) / (n as f64);
    println!("{}", avg);
}
