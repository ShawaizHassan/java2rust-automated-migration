use std::io::{self, Read};

struct Scanner {
    input: Vec<u8>,
    idx: usize,
}
impl Scanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }
    fn next(&mut self) -> String {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let start = self.idx;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        String::from_utf8(self.input[start..self.idx].to_vec()).unwrap()
    }
    fn next_i32(&mut self) -> i32 {
        self.next().parse::<i32>().unwrap()
    }
}

fn main() {
    let mut sc = Scanner::new();
    let n: i32 = sc.next_i32();
    let k: i32 = sc.next_i32();

    let n_usize = n as usize;
    let mut a: Vec<i32> = vec![0; n_usize + 1];
    for i in 0..n_usize {
        a[i] = sc.next_i32();
    }

    let mut ans: i32 = 0;
    for i in 0..n_usize {
        for j in 0..n_usize {
            if i != j {
                if (a[i] - a[j]).abs() <= k {
                    ans += 1;
                }
            }
        }
    }

    print!("{ans}\n");
}
