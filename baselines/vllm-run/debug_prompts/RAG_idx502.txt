fn sum(vs: &[i32]) -> i32 {
    vs.iter().copied().sum()
}

fn mcnt(n: usize, vs: &[i32]) -> i32 {
    let w = sum(vs);
    if w == n as i32 || w == 0 {
        return 0;
    }

    let mut res = w;
    for i in 0..n {
        let j = (i + w as usize) % n;

        let mut rr = w;
        let mut k = i;
        while k != j {
            rr -= vs[k];
            k = (k + 1) % n;
        }
        if rr < res {
            res = rr;
        }
    }
    res
}
