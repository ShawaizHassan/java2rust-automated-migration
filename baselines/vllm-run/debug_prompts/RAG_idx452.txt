class MatrixUtils {

    public static int[][] mul(int[][] a, int[][] b, int MOD) {
        int n = a.length;
        int m = a[0].length;
        int k = b[0].length;
        int[][] r = new int[n][k];
        for (int i = 0; i < n; ++i) {
            for (int j = 0; j < k; ++j) {
                for (int p = 0; p < m; ++p) {
                    r[i][j] += (a[i][p] * 1L * b[p][j]) % MOD;
                    if (r[i][j] >= MOD) r[i][j] -= MOD;
                }
            }
        }
        return r;
    }

    public static int[][] pow(int[][] aa, long p, int MOD) {
        int[][] a = aa.clone();
        int n = a.length;
        int[][] r = buildOne(n);
        for (long i = p; i > 0; i /= 2) {
            if (i % 2 == 1) r = mul(r, a, MOD);
            a = mul(a, a, MOD);
        }
        return r;
    }

    private static int[][] buildOne(int n) {
        int[][] r = new int[n][n];
        for (int i = 0; i < n; ++i) {
            r[i][i] = 1;
        }
        return r;
    }
