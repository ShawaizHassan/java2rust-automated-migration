struct Scanner {
    input: Vec<u8>,
    idx: usize,
}
impl Scanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }
    fn next_i32(&mut self) -> i32 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
    fn next_usize(&mut self) -> usize {
        self.next_i32() as usize
    }
}

fn solution(mut array: Vec<i32>) -> &'static str {
    if array.len() == 1 {
        return "YES";
    }
    array.sort();
    for i in 0..array.len() - 1 {
        let diff = array[i + 1] - array[i];
        if diff > 1 {
            return "NO";
        }
    }
    "YES"
}
