use std::io::{self, Read};

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: i32 = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };
    let mut x: i64 = it.next().unwrap().parse::<i32>().unwrap() as i64;

    let mut count: i32 = 0;

    for _ in 0..n {
        let c = it.next().unwrap().as_bytes()[0] as char;
        let o: i32 = it.next().unwrap().parse().unwrap();

        if c == '+' {
            x += o as i64;
        } else if c == '-' {
            if (o as i64) > x {
                count += 1;
            } else {
                x -= o as i64;
            }
        }
    }

    println!("{} {}", x, count);
}
