    let n: usize = match it.next() {
        Some(x) => x.parse().unwrap(),
        None => return,
    };

    let mut v: Vec<i64> = vec![0; n];
    let mut d: Vec<i64> = vec![0; n];
    let mut p: Vec<i64> = vec![0; n];

        v[i] = it.next().unwrap().parse::<i64>().unwrap();
        d[i] = it.next().unwrap().parse::<i64>().unwrap();
        p[i] = it.next().unwrap().parse::<i64>().unwrap();
    }

    let mut result: Vec<usize> = Vec::new();
    let mut ch: Vec<bool> = vec![false; n];

    for i in 0..n {
        if p[i] >= 0 {
            result.push(i + 1);
            let mut current = v[i];

            for x in ch.iter_mut() {
                *x = false;
            }

                if p[j] >= 0 {
                    p[j] -= current;
                    current -= 1;
                    if p[j] < 0 {
                        ch[j] = true;
                    }

                j += 1;

            }
        }
