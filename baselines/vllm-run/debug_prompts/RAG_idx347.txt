    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self {
            input: input.into_bytes(),
            idx: 0,
        }
    }

    fn next_i32(&mut self) -> i32 {
        self.next_string().parse::<i32>().unwrap()
    }

    fn next_string(&mut self) -> String {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let start = self.idx;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        String::from_utf8(self.input[start..self.idx].to_vec()).unwrap()
    }
}

use std::io::{self, Read};

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let str1 = match it.next() {
        Some(s) => s.as_bytes().to_vec(),
        None => return,
    };
    let str2 = match it.next() {
        Some(s) => s.as_bytes().to_vec(),
        None => return,
    };

    let n = str1.len();
    let mut dp = vec![vec![0i32; 26]; n + 1];
    let mut temp = vec![-1i32; 26];

    for i in (0..n).rev() {
        dp[i + 1].clone_from_slice(&temp);
        let c = (str1[i] - b'a') as usize;
        temp[c] = (i as i32) + 1;
    }
    dp[0] = temp;

    for &ch in &str2 {
        let idx = (ch - b'a') as usize;
        if dp[0][idx] == -1 {
            print!("-1\n");
            return;
        }
    }

    let mut i_pos: usize = 0;
    let mut count: i32 = 1;
    let mut m: usize = 0;

    while m < str2.len() {
        let idx = (str2[m] - b'a') as usize;
        let next = dp[i_pos][idx];
        if next == -1 {
            count += 1;
            i_pos = 0;
        } else {
            i_pos = next as usize;
            m += 1;
        }
    }

    print!("{}\n", count);
