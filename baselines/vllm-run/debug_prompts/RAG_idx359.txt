use std::io::{self, Read};

const MOD: i64 = 1_000_000_007;

fn mod_pow(mut a: i64, mut e: i64) -> i64 {
    let mut r: i64 = 1;
    a %= MOD;
    while e > 0 {
        if (e & 1) == 1 {
            r = (r * a) % MOD;
        }
        a = (a * a) % MOD;
        e >>= 1;
    }
    r
}

fn egcd(a: i64, b: i64) -> (i64, i64, i64) {
    if b == 0 {
        (a, 1, 0)
    } else {
        let (g, x, y) = egcd(b, a % b);
        (g, y, x - (a / b) * y)
    }
}

fn mod_inv(a: i64) -> i64 {
    let (g, x, _) = egcd(a, MOD);
    if g != 1 && g != -1 {
        0
    } else {
        ((x % MOD) + MOD) % MOD
    }
}

fn floor_div_nonneg_big_by_i32(divisor: i64, digits: &[u8]) -> i64 {
    let mut rem: i64 = 0;
    let mut q_mod: i64 = 0;
    for &d in digits {
        rem = rem * 10 + d as i64;
        let qd = rem / divisor;
        rem %= divisor;
        q_mod = (q_mod * 10 + (qd % MOD)) % MOD;
    }
    q_mod
}

use std::io::{self, Read};

#[derive(Clone, Debug, PartialEq, Eq)]
struct BigNum {
    // non-negative, base 2^32 limbs, little-endian
    limbs: Vec<u32>,
}

impl BigNum {
    fn zero() -> Self {
        Self { limbs: Vec::new() }
    }

    fn one() -> Self {
        Self { limbs: vec![1] }
    }

    fn is_zero(&self) -> bool {
        self.limbs.is_empty()
    }

    fn normalize(&mut self) {
        while let Some(&last) = self.limbs.last() {
            if last == 0 {
                self.limbs.pop();
            } else {
                break;
            }
        }
    }

    fn cmp(&self, other: &Self) -> std::cmp::Ordering {
        if self.limbs.len() != other.limbs.len() {
            return self.limbs.len().cmp(&other.limbs.len());
        }
        for i in (0..self.limbs.len()).rev() {
            if self.limbs[i] != other.limbs[i] {
                return self.limbs[i].cmp(&other.limbs[i]);
            }
        }
        std::cmp::Ordering::Equal
    }

    fn max(a: Self, b: Self) -> Self {
        if a.cmp(&b) >= std::cmp::Ordering::Equal {
            a
        } else {
            b
        }
    }

    fn add(&self, other: &Self) -> Self {
        let n = self.limbs.len().max(other.limbs.len());
        let mut res = Vec::with_capacity(n + 1);
        let mut carry: u64 = 0;
        for i in 0..n {
            let a = if i < self.limbs.len() { self.limbs[i] as u64 } else { 0 };
            let b = if i < other.limbs.len() { other.limbs[i] as u64 } else { 0 };
            let sum = a + b + carry;
            res.push((sum & 0xFFFF_FFFF) as u32);
            carry = sum >> 32;
        }
        if carry != 0 {
            res.push(carry as u32);
        }
        let mut out = Self { limbs: res };
        out.normalize();
        out
    }

    fn pow2(k: i32) -> Self {
        if k < 0 {
            return Self::zero();
        }
        let k = k as usize;
        let limb_idx = k / 32;
        let bit_idx = k % 32;
        let mut limbs = vec![0u32; limb_idx + 1];
        limbs[limb_idx] = 1u32 << bit_idx;
        let mut out = Self { limbs };
        out.normalize();
        out
    }

    fn to_decimal_string(&self) -> String {
        if self.is_zero() {
            return "0".to_string();
        }
        const BASE10: u64 = 1_000_000_000;

        let mut tmp = self.limbs.clone();
        let mut parts: Vec<u32> = Vec::new();

        while !tmp.is_empty() {
            let mut carry: u64 = 0;
            for i in (0..tmp.len()).rev() {
                let cur = (carry << 32) | (tmp[i] as u64);
                tmp[i] = (cur / BASE10) as u32;
                carry = cur % BASE10;
            }
            parts.push(carry as u32);
            while let Some(&last) = tmp.last() {
                if last == 0 {
                    tmp.pop();
                } else {
                    break;
                }
            }
        }

        let mut s = String::new();
        if let Some(&last) = parts.last() {
            s.push_str(&last.to_string());
        }
        for &p in parts.iter().rev().skip(1) {
            s.push_str(&format!("{:09}", p));
        }
        s
    }
}

struct FastScanner {
    data: Vec<u8>,
    idx: usize,
}

impl FastScanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self {
            data: input.into_bytes(),
            idx: 0,
        }
    }

    fn skip_whitespace(&mut self) {
        while self.idx < self.data.len() && self.data[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
    }

    fn next_i32(&mut self) -> i32 {
        self.skip_whitespace();
        let mut sign = 1i32;
        if self.idx < self.data.len() && self.data[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut val: i32 = 0;
        while self.idx < self.data.len() {
            let c = self.data[self.idx];
            if !c.is_ascii_digit() {
                break;
            }
            val = val.saturating_mul(10).saturating_add((c - b'0') as i32);
            self.idx += 1;
        }
        val * sign
    }

    fn next_string(&mut self) -> String {
        self.skip_whitespace();
        let start = self.idx;
        while self.idx < self.data.len() && !self.data[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        String::from_utf8(self.data[start..self.idx].to_vec()).unwrap()
    }
}
