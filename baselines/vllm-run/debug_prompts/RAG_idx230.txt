    idx: usize,
}
impl FastScanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { buf: input.into_bytes(), idx: 0 }
    }
    fn next_i32(&mut self) -> i32 {
        while self.idx < self.buf.len() && self.buf[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.buf.len() && self.buf[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.buf.len() && !self.buf[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.buf[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
}

fn possible(n: usize, k: i64, a: &[i64], b: &[i64], mid: i64) -> bool {
    let mut left: i64 = k;
    for i in 0..n {
        let target: i64 = a[i] * mid;
        if b[i] < target {
            left -= target - b[i];
        }
    }
    left >= 0
}
