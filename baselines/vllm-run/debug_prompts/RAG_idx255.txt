fn bad(s: &[u8]) -> bool {
    let mut min_c: u8 = 255;
    let mut max_c: u8 = 0;
    for &c in s {
        if c < min_c {
            min_c = c;
        }
        if c > max_c {
            max_c = c;
        }
    }
    if min_c == max_c {
        return true;
    }
    for &c in s {
        if c != min_c && c != max_c {
            return false;
        }
    }
    let mut cnt0 = 0i32;
    let mut cnt1 = 0i32;
    for &c in s {
        if c == min_c {
            cnt0 += 1;
        } else {
            cnt1 += 1;
        }
    }
    cnt0 == 1 || cnt1 == 1
}

fn palindrome(s: &[u8]) -> bool {
    let n = s.len();
    for i in 0..n {
        if s[i] != s[n - i - 1] {
            return false;
        }
    }
    true
}
