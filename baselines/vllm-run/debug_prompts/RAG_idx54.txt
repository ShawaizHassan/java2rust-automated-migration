}

fn dnc(lo: i32, hi: i32, a: i32, b: i32) -> i64 {
    if lo > hi {
        return 0;
    }
    let ceil = first(lo);
    let floor = last(hi);

    if lo == hi {
        if floor < ceil {
            return a as i64;
        } else {
            return (hi as i64 - lo as i64 + 1)
                * (floor as i64 - ceil as i64 + 1)
                * (b as i64);
        }
    }

    if floor < ceil {
        return a as i64;
    }

    let total =
        (floor as i64 - ceil as i64 + 1) * (hi as i64 - lo as i64 + 1) * (b as i64);

    let mid = ((lo + hi) as u32 >> 1) as i32;
    let mut divided = dnc(lo, mid, a, b);
    divided += dnc(mid + 1, hi, a, b);

    if total < divided { total } else { divided }
}
