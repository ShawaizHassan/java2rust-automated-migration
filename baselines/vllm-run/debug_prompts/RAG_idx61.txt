const LIMIT: usize = 1_000_001;

fn primelist() -> Vec<u8> {
    let mut prime = vec![0u8; LIMIT];
    prime[0] = 1;
    prime[1] = 1;
    for i in 2..LIMIT {
        if prime[i] == 0 {
            let ii = (i as u64) * (i as u64);
            if ii >= LIMIT as u64 {
                continue;
            }
            let mut j = ii as usize;
            while j < LIMIT {
                prime[j] = 1;
                j += i;
            }
        }
    }
    prime
}

use std::io::{self, Read};

fn gcd(a: i32, b: i32) -> i32 {
    if a == 0 {
        b
    } else {
        gcd(b % a, a)
    }
}

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let n: i32 = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };
    let mut m: i32 = it.next().unwrap().parse().unwrap();

    let mut arr: Vec<String> = Vec::new();

    if m < (n - 1) {
        print!("Impossible\n");
    } else {
        let mut i = 1;
        while i <= n && m > 0 {
            let mut j = i + 1;
            while j <= n && m > 0 {
                if gcd(j, i) == 1 {
                    arr.push(format!("{} {}", i, j));
                    m -= 1;
                }
                j += 1;
            }
            i += 1;
        }

        if m == 0 {
            let mut out = String::new();
            out.push_str("Possible\n");
            for s in arr {
                out.push_str(&s);
                out.push('\n');
            }
            print!("{}", out);
        } else {
            print!("Impossible\n");
        }
    }
}
