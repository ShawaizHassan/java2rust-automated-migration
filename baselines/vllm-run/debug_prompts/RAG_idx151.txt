use std::io::{self, Read};

fn main() {
    // Read all input
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.lines();

    let n: i32 = it.next().unwrap().trim().parse().unwrap();

    if n == 1 {
        // Read and ignore the single string
        let _s = it.next().unwrap_or("");
        println!("0");
        return;
    }

    let n_usize = n as usize;

    let mut parity_strings: Vec<String> = vec![String::new(); n_usize];

    for i in 0..n_usize {
        let s = it.next().unwrap_or("");
        let mut c = [0i32; 26];
        for b in s.as_bytes() {
            let idx = (b - b'a') as usize;
            c[idx] += 1;
        }
        let mut out = String::new();
        for j in 0..26 {
            if (c[j] & 1) == 1 {
                out.push((b'a' + j as u8) as char);
            }
        }
        parity_strings[i] = out;
    }

    let mut hm: HashMap<String, i32> = HashMap::new();
    for s in &parity_strings {
        if let Some(v) = hm.get_mut(s) {
            *v += 1;
        } else {
            hm.insert(s.clone(), 1);
        }
    }

    let mut res: i64 = 0;

    for &x in hm.values() {
        if x > 1 {
            let xi = x as i64;
            res += (xi * (xi - 1)) / 2;
        }
    }

    // Iterate over keys snapshot to avoid borrow issues (matches Java semantics)
    let keys: Vec<String> = hm.keys().cloned().collect();
    for x in keys {
        let l = x.len();
        for i in 0..l {
            // build string with char at position i removed (ASCII letters only)
            let mut ss = String::with_capacity(l.saturating_sub(1));
            ss.push_str(&x[..i]);
            ss.push_str(&x[i + 1..]);
            if let Some(&cnt_ss) = hm.get(&ss) {
                let cnt_x = *hm.get(&x).unwrap();
                res += (cnt_ss as i64) * (cnt_x as i64);
            }
        }
    }

    println!("{res}");
}

use std::io::{self, Read};

fn count_difference(ss: &[u8], s: &[u8]) -> i32 {
    let mut cnt: i32 = 0;
    for i in 0..ss.len() {
        if ss[i] != s[i] {
            cnt += 1;
        }
    }
    cnt / 2
}

    let s = s_str.as_bytes();
    let n = s.len();

    let mut c_h: usize = 0;
    for &ch in s {
        if ch == b'H' {
            c_h += 1;
        }
    }

    let mut ss: Vec<u8> = Vec::with_capacity(n);
    ss.extend(std::iter::repeat(b'H').take(c_h));
    ss.extend(std::iter::repeat(b'T').take(n - c_h));

    let mut res: i32 = i32::MAX;
    for _ in 0..n {
        let cur = count_difference(&ss, s);
        if cur < res {
            res = cur;
        }
        if n > 0 {

            ss.rotate_left(1);
            ss[n - 1] = first;
