use std::io::{self, Read};

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let mut out = String::new();

    loop {
        let n = match it.next() {
            Some(v) => v.parse::<i32>().unwrap(),
            None => break,
        };
        let k = it.next().unwrap().parse::<i32>().unwrap();
        let c = it.next().unwrap().parse::<i32>().unwrap();

        let mut day = vec![false; (n as usize) + 1];
        for _ in 0..c {
            if let Some(v) = it.next() {
                let d = v.parse::<usize>().unwrap();
                if d < day.len() {
                    day[d] = true;
                }
            }
        }

        let mut cnt: i32 = 0;
        let mut ck: i32 = 0;

        for i in 1..day.len() {
            if day[i] {
                ck = 0;
                cnt += 1;
            } else {
                ck += 1;
                if ck == k {
                    ck = 0;
                    day[i] = true;
                    cnt += 1;
                }
            }
        }

        out.push_str(&format!("{}\n", cnt));
    }

    print!("{}", out);
}
