use std::collections::HashMap;
use std::io::{self, Read};

struct Scanner {
    input: Vec<u8>,
    idx: usize,
}

impl Scanner {
    fn new() -> Self {
        let mut input = Vec::new();
        io::stdin().read_to_end(&mut input).unwrap();
        Self { input, idx: 0 }
    }

    fn next_i32(&mut self) -> i32 {
        self.skip_whitespace();
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.input.len() {
            let c = self.input[self.idx];
            if c.is_ascii_digit() {
                num = num * 10 + (c - b'0') as i32;
                self.idx += 1;
            } else {
                break;
            }
        }
        sign * num
    }

    fn next_string(&mut self) -> String {
        self.skip_whitespace();
        let start = self.idx;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        String::from_utf8(self.input[start..self.idx].to_vec()).unwrap()
    }

    fn skip_whitespace(&mut self) {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
    }
}

fn main() {
    let mut sc = Scanner::new();
    let t = sc.next_i32();

    let mut out = String::new();

    for _ in 0..t {
        let n = sc.next_i32();
        let strv = sc.next_string();

        if n == 1 {
            out.push_str("-1\n");
            continue;
        }

        let bytes = strv.as_bytes();

        let mut map: HashMap<(i32, i32), i32> = HashMap::new();
        map.insert((0, 0), 0);

        let mut min_len: i32 = i32::MAX;
        let mut start: i32 = -1;
        let mut end: i32 = -1;

        let mut x: i32 = 0;
        let mut y: i32 = 0;

        for j in 0..(n as usize) {
            match bytes[j] {
                b'L' => x -= 1,
                b'R' => x += 1,
                b'U' => y += 1,
                _ => y -= 1,
            }

            let key = (x, y);
            let step = (j as i32) + 1;

            if let Some(&prev) = map.get(&key) {
                let len = step - prev;
                if len < min_len {
                    min_len = len;
                    start = prev + 1;
                    end = step;
                }
                map.insert(key, step);
            } else {
                map.insert(key, step);
            }
        }

        if start == -1 && end == -1 {
            out.push_str("-1\n");
        } else {
            out.push_str(&format!("{} {}\n", start, end));
        }
    }

    print!("{}", out);

use std::io::{self, Read};

fn main() {
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace();

    let t: i32 = match it.next() {
        Some(v) => v.parse().unwrap(),
        None => return,
    };

    let mut out = String::new();

    for _ in 0..t {
        let n: usize = it.next().unwrap().parse::<i32>().unwrap() as usize;
        let s = it.next().unwrap().as_bytes();

        let mut map: HashMap<(i32, i32), i32> = HashMap::new();
        let mut x: i32 = 0;
        let mut y: i32 = 0;
        map.insert((0, 0), 0);

        let mut re: i32 = 1 << 30;
        let mut from: i32 = -1;
        let mut to: i32 = -1;

        for i in 0..n {
            match s[i] as char {
                'L' => x -= 1,
                'R' => x += 1,
                'U' => y -= 1,
                'D' => y += 1,
                _ => {}
            }

            let key = (x, y);
            if let Some(&prev) = map.get(&key) {
                let length = i as i32 - prev;
                if re > length {
                    re = length;
                    from = prev;
                    to = i as i32;
                }
            }
            map.insert(key, i as i32 + 1);
        }

        if re >= (1 << 30) {
            out.push_str("-1\n");
        } else {
            out.push_str(&(from + 1).to_string());
            out.push(' ');
            out.push_str(&(to + 1).to_string());
            out.push('\n');
        }
    }

    print!("{}", out);
}
