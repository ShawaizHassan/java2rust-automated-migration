    let u_line = parts.get(1).copied().unwrap_or("").trim_end_matches('\r');

    let s: Vec<u16> = s_line.encode_utf16().collect();
    let u: Vec<u16> = u_line.encode_utf16().collect();

    let s_len = s.len() as i32;
    let u_len = u.len() as i32;

    let mut ans: i32 = u_len;

    for start in (-u_len + 1)..s_len {
        let mut count: i32 = 0;
        for i in start..(start + u_len) {
            if i < 0 || i >= s_len {
                count += 1;
            } else {
                let si = s[i as usize];
                let ui = u[(i - start) as usize];
                if si != ui {
                    count += 1;
                }
            }
        }
        if count < ans {
            ans = count;
        }
    }

    print!("{}", ans);
    if !ans.to_string().ends_with('\n') {

fn main() {

            ans += 1;
        }
    }
