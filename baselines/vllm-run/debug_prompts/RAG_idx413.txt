    fn new(input: R) -> Self {
        Self { input }

    fn rl(&mut self) -> String {
        let mut s = String::new();
        let n = self.input.read_line(&mut s).unwrap();
        if n == 0 {
            return String::new();

        while s.ends_with('\n') || s.ends_with('\r') {
