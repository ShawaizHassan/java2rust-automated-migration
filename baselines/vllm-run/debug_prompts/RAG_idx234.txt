use std::io::{self, Read};

fn bad(s: &[u8]) -> bool {
    let mut min_c: u8 = 255;
    let mut max_c: u8 = 0;
    for &c in s {
        if c < min_c {
            min_c = c;
        }
        if c > max_c {
            max_c = c;
        }
    }
    if min_c == max_c {
        return true;
    }
    for &c in s {
        if c != min_c && c != max_c {
            return false;
        }
    }
    let mut cnt0 = 0i32;
    let mut cnt1 = 0i32;
    for &c in s {
        if c == min_c {
            cnt0 += 1;
        } else {
            cnt1 += 1;
        }
    }
    cnt0 == 1 || cnt1 == 1
}

fn palindrome(s: &[u8]) -> bool {
    let n = s.len();
    for i in 0..n {
        if s[i] != s[n - i - 1] {
            return false;
        }
    }
    true
}

fn main() {
    let mut input = String::new();

    let mut it = input.lines();

    let t: i32 = match it.next() {
        Some(x) => x.trim().parse().unwrap(),
        None => return,
    };

    for _ in 0..t {
        let n: i32 = it.next().unwrap_or("").trim().parse().unwrap();
        let s = it.next().unwrap_or("").trim_end_matches(&['\r', '\n'][..]).to_string();
        let bytes = s.as_bytes();

        if n % 2 == 0 {
            let mut i: usize = 1;
            while i < bytes.len() {
                if bytes[i] % 2 == 0 {
                    println!("2");
                    break;

                i += 2;
            }

                println!("1");

                    println!("1");
                    break;

                i += 2;
