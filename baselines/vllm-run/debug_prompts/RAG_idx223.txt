use std::io::{self, Read};

struct Scanner {
    input: Vec<u8>,
    idx: usize,
}
impl Scanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }
    fn next_i32(&mut self) -> i32 {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            num = num * 10 + (self.input[self.idx] - b'0') as i32;
            self.idx += 1;
        }
        num * sign
    }
}

fn binary_search(a: &Vec<i32>, l: i32, r: i32, n: i32) -> i32 {
    let m = (l + r) / 2;
    let mi = m as usize;
    if a[mi] == n || (a[mi] < n && a[mi + 1] > n) {
        m + 1
    } else if a[mi] < n {
        binary_search(a, m + 1, r, n)
    } else {
        binary_search(a, l, m, n)
    }
}
