        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }
    fn is_whitespace(b: u8) -> bool {
        b == b' ' || b == b'\n' || b == b'\r' || b == b'\t' || b == 0x0B || b == 0x0C
    }
    fn next_token(&mut self) -> Option<String> {
        while self.idx < self.input.len() && Self::is_whitespace(self.input[self.idx]) {
            self.idx += 1;

            return None;

        self.next_token().map(|s| s.parse::<i32>().unwrap())
