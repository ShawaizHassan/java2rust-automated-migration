fn generate_line(n: i32, m: i32) {
    let mut p: i32 = 0;
    let spaces: i32 = m - n;

    for _ in 0..spaces {
        print!("  ");
    }
    for _ in 0..=n {
        if p == n && n == 0 {
            print!("{}", p);
        } else {
            print!("{} ", p);
        }
        p += 1;
    }
    p -= 2;
    while p > 0 {
        print!("{} ", p);
        p -= 1;
    }
    if n != 0 {
        print!("0");
    }
    print!("\n");
}
