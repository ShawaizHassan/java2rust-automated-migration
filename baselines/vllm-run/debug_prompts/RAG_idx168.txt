use std::io::{self, Read};

fn bad(s: &[u8]) -> bool {
    let mut min_c: u8 = 255;
    let mut max_c: u8 = 0;
    for &c in s {
        if c < min_c {
            min_c = c;
        }
        if c > max_c {
            max_c = c;
        }
    }
    if min_c == max_c {
        return true;
    }
    for &c in s {
        if c != min_c && c != max_c {
            return false;
        }
    }
    let mut cnt0 = 0i32;
    let mut cnt1 = 0i32;
    for &c in s {
        if c == min_c {
            cnt0 += 1;
        } else {
            cnt1 += 1;
        }
    }
    cnt0 == 1 || cnt1 == 1
}

fn palindrome(s: &[u8]) -> bool {
    let n = s.len();
    for i in 0..n {
        if s[i] != s[n - i - 1] {
            return false;
        }
    }
    true
}

use std::io::{self, Read};

struct Scanner {
    input: Vec<u8>,
    idx: usize,
}
impl Scanner {
    fn new() -> Self {
        let mut input = String::new();
        io::stdin().read_to_string(&mut input).unwrap();
        Self { input: input.into_bytes(), idx: 0 }
    }
    fn next_i32(&mut self) -> i32 {
        self.skip_whitespace();
        let mut sign = 1i32;
        if self.idx < self.input.len() && self.input[self.idx] == b'-' {
            sign = -1;
            self.idx += 1;
        }
        let mut num = 0i32;
        while self.idx < self.input.len() {
            let c = self.input[self.idx];
            if c < b'0' || c > b'9' {
                break;
            }
            num = num * 10 + (c - b'0') as i32;
            self.idx += 1;
        }
        sign * num
    }
    fn next_string(&mut self) -> String {
        self.skip_whitespace();
        let start = self.idx;
        while self.idx < self.input.len() && !self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
        String::from_utf8(self.input[start..self.idx].to_vec()).unwrap()
    }
    fn skip_whitespace(&mut self) {
        while self.idx < self.input.len() && self.input[self.idx].is_ascii_whitespace() {
            self.idx += 1;
        }
    }
}

fn main() {
    let mut scn = Scanner::new();
    let mut t = scn.next_i32();
    let mut out = String::new();

    while t > 0 {
        t -= 1;
        let n = scn.next_i32() as usize;
        let s = scn.next_string();
        let bytes = s.as_bytes();

        let mut i: usize = 0;
        let mut ans: i32 = 0;
        let mut al: Vec<i32> = Vec::new();

        while i < bytes.len() {
            let mut cnt: i32 = 1;
            while i + 1 < bytes.len() && bytes[i] == bytes[i + 1] {
                cnt += 1;
                i += 1;
            }

            if cnt as usize == n {
                ans += ((cnt + 2) / 3) as i32; // ceil(cnt/3)
                break;
            } else {
                if i == bytes.len() - 1 && bytes[0] == bytes[n - 1] {
                    // Mirrors Java behavior; will panic if al is empty, same as Java exception.
                    let v0 = al[0] + cnt;
                    al[0] = v0;
                } else {
                    al.push(cnt);
                }
            }
            i += 1;
        }

        if al.is_empty() {
            out.push_str(&format!("{}\n", ans));
            continue;
        }
        for &val in &al {
            ans += val / 3;
        }
        out.push_str(&format!("{}\n", ans));
    }

    print!("{}", out);
}
